1//! # One number per transaction: where its coins came from, on average
2//!
3//! [`main`](../coloring_bt_transactions/index.html) prints a transaction's
4//! **colour** in full — every block its coins descend from, each with its share
5//! of the value under `--weighted`.  That is the whole answer, and it is also
6//! why a line of it can run to tens of thousands of bytes: a colour is a *set*,
7//! and a set does not fit in a column.
8//!
9//! This binary prints the same colouring collapsed to one `f64` per record:
10//!
11//! ```text
12//!     mean  =  sum_b  b . weight(b)
13//! ```
14//!
15//! the **weighted mean block id** — the centre of mass of the blocks the coins
16//! came from.  A coinbase minted in block `b` prints exactly `b`; a transaction
17//! spending half its value from block 0 and half from block 3 prints `1.5`.  So
18//! the number reads on the same scale as a block id, and the distance between
19//! two of them is a distance along the chain.
20//!
21//! ```text
22//! tx-mean [<record-limit>|all] < records
23//! ```
24//!
25//! Output is `<tx-id>,<mean>` a line, in the order the records arrive, and each
26//! line is written the moment that record's colour is finished — nothing is held
27//! back to the end, so this is usable in a pipe over a chain that has not
28//! stopped arriving.  (Standard output is still buffered a megabyte at a time;
29//! that is an I/O detail, not a batching one.)
30//!
31//! # Why this needs the weighted backend
32//!
33//! A mean needs weights to be a mean.  The unweighted backends give every block
34//! in a colour a coefficient of 1, so the best they could offer is the *plain*
35//! mean of the block ids — which counts a block that one satoshi passed through
36//! the same as the block the whole balance was minted in.  So the store here is
37//! [`weighted::WeightedSets`] and there is no flag to change it: this program
38//! asks a question only that backend can answer.
39//!
40//! What that costs is the caveat the main driver states about `--weighted` — a
41//! weight decays by roughly a factor per hop of ancestry, so a colour's older
42//! terms carry very little.  It costs this program *less* than it costs printing
43//! the terms, though: a weight too small to show in six decimals still moves the
44//! mean by its full share, because the mean sums the weights rather than
45//! rounding each one on its own.
46//!
47//! # The loop
48//!
49//! It is `main::run`'s, with the emit step replaced.  A transaction's colour is
50//! the fold of its inputs' colours, each scaled by the fraction of value it
51//! carries, and an entry is dropped when its last unspent output is spent — so
52//! the colours in flight track the UTXO set rather than the chain, exactly as
53//! they do in the main driver.  That bookkeeping is the reason a run over a
54//! million records fits in memory, and it is why it is copied here rather than
55//! simplified away.

88/// How many decimal places the mean is printed to.
89///
90/// The same six as `main`'s weights, and for the same reason: enough to separate
91/// two transactions that differ, few enough to hold a column.  Six is also well
92/// inside what an `f64` can say about a number the size of a block id — a mean
93/// near a million still has nine significant digits to spare.
94const PLACES: u32 = 6;
95
96fn main() -> ExitCode {
97    // No limit unless one is asked for.  `main`'s default stops at 1,000,001
98    // records because that is where the Scheme it ports stops and the recorded
99    // output has to be reproducible; this program has no output to reproduce, so
100    // the useful default is "all of them".
101    let mut limit = usize::MAX;
102
103    for arg in std::env::args().skip(1) {
104        match arg.as_str() {
105            "all" => limit = usize::MAX,
106            _ => match arg.parse::<usize>() {
107                Ok(n) => limit = n,
108                Err(_) => {
109                    eprintln!("{}", USAGE);
110                    return ExitCode::FAILURE;
111                }
112            },
113        }
114    }
115
116    let out = io::BufWriter::with_capacity(1 << 20, io::stdout().lock());
117    match run(io::stdin().lock(), out, limit) {
118        Ok(()) => ExitCode::SUCCESS,
119        // Downstream went away (`| head`); that is not our failure.
120        Err(e) if e.kind() == io::ErrorKind::BrokenPipe => ExitCode::SUCCESS,
121        Err(e) => {
122            eprintln!("tx-mean: {}", e);
123            ExitCode::FAILURE
124        }
125    }
126}

128/// Colour at most `limit` records off `input`, writing a line to `out` per
129/// record as each colour is finished.
130///
131/// The writer is a parameter rather than standard output taken here, so that a
132/// test can read back what a run produced; `main` hands it the buffered stdout
133/// it would otherwise have opened.
134fn run(input: impl io::Read, mut out: impl Write, limit: usize) -> io::Result<()> {
135    let mut reader = sexp::Reader::new(input);
136
137    let mut store = WeightedSets::new();
138    // A transaction's colour, and how many of its outputs are still unspent.
139    // The count is the whole memory-management story: spending the last one
140    // drops the entry, and with it the colour.
141    let mut colors: HashMap<usize, (Color, usize)> = HashMap::new();
142    let mut inputs: Vec<sexp::Input> = Vec::new();
143    // Reused across records so the formatting does not allocate per line.
144    let mut line: Vec<u8> = Vec::new();
145
146    let mut records: usize = 0;
147    while records < limit {
148        let record = match reader.next_record(&mut inputs)? {
149            Some(r) => r,
150            None => break,
151        };
152
153        let color = if inputs.is_empty() {
154            // Coinbase: the block that minted it holds all of the weight, so the
155            // mean of this one comes out as the block id itself.
156            store.singleton(record.block_id)
157        } else {
158            // `foldr` over the inputs, right to left.  The order does not change
159            // the sum -- the merge is sorted and addition is commutative -- but
160            // it decides which input hits an entry's last unspent output, which
161            // is the order `main` folds in.
162            let mut accumulator: Option<Color> = None;
163
164            // Each input contributes its ancestor's colour in proportion to the
165            // amount it spends.  A total of zero has no proportions to speak of,
166            // and nothing forbids a zero-value input, so those share equally
167            // rather than dividing by it.
168            let total: f64 = inputs.iter().map(|i| i.amount as f64).sum();
169            let share = |input: &sexp::Input| -> f64 {
170                if total > 0.0 {
171                    input.amount as f64 / total
172                } else {
173                    1.0 / inputs.len() as f64
174                }
175            };
176
177            for i in (0..inputs.len()).rev() {
178                let previous = inputs[i].prev_tx_id;
179                let weight = share(&inputs[i]);
180                let entry = colors.get_mut(&previous).ok_or_else(|| {
181                    io::Error::new(
182                        io::ErrorKind::InvalidData,
183                        format!(
184                            "transaction {} spends unknown transaction {}",
185                            record.tx_id, previous
186                        ),
187                    )
188                })?;
189                let unspent = entry.1;
190
191                if unspent > 1 {
192                    // Others can still reach this colour, so it has to survive
193                    // the fold: merge from a borrow.
194                    entry.1 = unspent - 1;
195                    let held = &entry.0;
196                    accumulator = Some(match accumulator.take() {
197                        // First step: the accumulator *is* this input's share of
198                        // it.  A single-input transaction has weight 1 and
199                        // `scale` shares rather than rebuilding, so the
200                        // commonest shape costs nothing.
201                        None => store.scale(held, weight),
202                        // The accumulator already carries its own share, so it
203                        // comes in at full strength.
204                        Some(acc) => {
205                            let combined = store.combine(held, weight, &acc, 1.0);
206                            store.release(acc);
207                            combined
208                        }
209                    });
210                } else {
211                    // The last unspent output is being spent right now, so
212                    // nobody can reach this colour again and it may be taken
213                    // outright.
214                    let (root, _) = colors.remove(&previous).expect("just looked it up");
215                    accumulator = Some(match accumulator.take() {
216                        None => {
217                            if weight == 1.0 {
218                                root
219                            } else {
220                                let scaled = store.scale(&root, weight);
221                                store.release(root);
222                                scaled
223                            }
224                        }
225                        Some(acc) => {
226                            let combined = store.combine(&root, weight, &acc, 1.0);
227                            store.release(acc);
228                            store.release(root);
229                            combined
230                        }
231                    });
232                }
233            }
234            accumulator.expect("inputs is non-empty, so the fold ran at least once")
235        };
236
237        line.clear();
238        push_int(&mut line, record.tx_id);
239        line.push(b',');
240        push_fixed(&mut line, mean(&store, &color));
241        line.push(b'\n');
242        out.write_all(&line)?;
243
244        if record.outputs > 0 {
245            if let Some((displaced, _)) = colors.insert(record.tx_id, (color, record.outputs)) {
246                store.release(displaced);
247            }
248        } else {
249            store.release(color);
250        }
251
252        records += 1;
253    }
254
255    out.flush()
256}

258/// `sum_b b . weight(b)`, the block id a colour sits at on average.
259///
260/// Divided by the weights rather than trusting them to sum to 1.  They do sum to
261/// 1 -- that is the invariant `--stats` reports drift against -- but the
262/// division is one instruction per *record* against a walk of the whole colour,
263/// and it makes the answer a weighted mean by construction instead of by
264/// argument.
265///
266/// A colour with no terms cannot come out of the fold above: every colour starts
267/// as a single block and a union never empties one.  Answering 0 rather than a
268/// NaN is what to do with a case that does not arise.
269fn mean(store: &WeightedSets, color: &Color) -> f64 {
270    let (mut moment, mut total) = (0.0f64, 0.0f64);
271    store.for_each_term(color, |block, weight| {
272        moment += block as f64 * weight;
273        total += weight;
274    });
275    if total > 0.0 {
276        moment / total
277    } else {
278        0.0
279    }
280}

282/// A non-negative value to [`PLACES`] fixed decimals, straight into the line
283/// buffer.
284///
285/// Fixed rather than shortest-round-trip so the column lines up, and done in
286/// integers for the reason [`push_int`] exists: `write!`'s formatting machinery
287/// is a real cost once it runs a million times.  Scaling by a power of ten and
288/// printing two integers is one multiply and one rounding.
289///
290/// The scaled value has to fit a `u64`, which at six places leaves room for
291/// block ids up to about 1.8e13 -- some ten million times the length of the
292/// chain, so the cast is not a limit anything will reach.
293fn push_fixed(out: &mut Vec<u8>, value: f64) {
294    let scale = 10u64.pow(PLACES);
295    let units = (value * scale as f64).round() as u64;
296
297    push_int(out, (units / scale) as usize);
298    out.push(b'.');
299
300    // Zero-padded to a fixed width, which `push_int` will not do -- it prints 5
301    // as "5" where this needs "000005".
302    let mut fraction = units % scale;
303    let mut digits = [b'0'; PLACES as usize];
304    let mut i = digits.len();
305    while fraction > 0 {
306        i -= 1;
307        digits[i] = b'0' + (fraction % 10) as u8;
308        fraction /= 10;
309    }
310    out.extend_from_slice(&digits);
311}
312
313/// Decimal, straight into the line buffer.  `main`'s, for the same reason it has
314/// one.
315fn push_int(out: &mut Vec<u8>, value: usize) {
316    let mut digits = [0u8; 20];
317    let mut i = digits.len();
318    let mut magnitude = value;
319    loop {
320        i -= 1;
321        digits[i] = b'0' + (magnitude % 10) as u8;
322        magnitude /= 10;
323        if magnitude == 0 {
324            break;
325        }
326    }
327    out.extend_from_slice(&digits[i..]);
328}

334    /// A record in the shape [`sexp`] reads: the header's block and tx id, then
335    /// `(prev-tx, amount)` an input, then how many outputs.
336    fn record(block: usize, tx: usize, spends: &[(usize, usize)], outputs: usize) -> String {
337        let inputs: String = spends
338            .iter()
339            .map(|(prev, amount)| format!("(7 {} {} 0)", amount, prev))
340            .collect();
341        let outs: String = (0..outputs).map(|_| "(7 1 0)".to_string()).collect();
342        format!("((1 {} {} 0 0 0 0) ({}) ({}))\n", block, tx, inputs, outs)
343    }
344
345    fn means(records: &str, limit: usize) -> Vec<String> {
346        let mut out = Vec::new();
347        run(records.as_bytes(), &mut out, limit).expect("the records are well formed");
348        String::from_utf8(out)
349            .expect("the output is decimal digits and punctuation")
350            .lines()
351            .map(str::to_string)
352            .collect()
353    }

355    /// Nothing was spent, so the block that minted it holds all of the weight
356    /// and the mean is that block, exactly.
357    #[test]
358    fn a_coinbase_sits_on_its_own_block() {
359        assert_eq!(means(&record(7, 0, &[], 1), usize::MAX), ["0,7.000000"]);
360    }

362    /// Half the value from block 0 and half from block 3 puts the mean halfway
363    /// between them -- the case the whole program exists to compute.
364    #[test]
365    fn two_equal_inputs_land_between_their_blocks() {
366        let records = record(0, 0, &[], 1) + &record(3, 1, &[], 1) + &record(5, 2, &[(0, 50), (1, 50)], 1);
367        assert_eq!(means(&records, usize::MAX).last().unwrap(), "2,1.500000");
368    }

370    /// Weighting is by amount, not by input count: nine tenths of the value
371    /// coming from block 0 pulls the mean nine tenths of the way to it.
372    #[test]
373    fn the_mean_follows_the_value_not_the_inputs() {
374        let records = record(0, 0, &[], 1) + &record(10, 1, &[], 1) + &record(11, 2, &[(0, 90), (1, 10)], 1);
375        assert_eq!(means(&records, usize::MAX).last().unwrap(), "2,1.000000");
376    }

378    /// A chain of single-input transactions carries its ancestor's mean along
379    /// unchanged: one input takes all of the weight, so there is nothing to mix
380    /// with.
381    #[test]
382    fn a_single_input_inherits_the_mean() {
383        let mut records = record(0, 0, &[], 1) + &record(4, 1, &[], 1) + &record(5, 2, &[(0, 50), (1, 50)], 1);
384        for tx in 3..8 {
385            records += &record(5 + tx, tx, &[(tx - 1, 100)], 1);
386        }
387        let lines = means(&records, usize::MAX);
388        assert!(lines[3..].iter().all(|l| l.ends_with(",2.000000")), "{:?}", lines);
389    }

391    /// Nothing forbids a record whose inputs are all worth nothing, and it must
392    /// not divide by the total: the inputs share equally instead.

396        assert_eq!(means(&records, usize::MAX).last().unwrap(), "2,3.000000");

399    /// The limit is a record count, and it stops the reader rather than the
400    /// printing -- so a short run is a prefix of a long one.
401    #[test]
402    fn the_limit_takes_a_prefix() {
403        let records = record(0, 0, &[], 1) + &record(1, 1, &[], 1) + &record(2, 2, &[], 1);
404        assert_eq!(means(&records, 2), ["0,0.000000", "1,1.000000"]);
405        assert!(means(&records, 0).is_empty());
