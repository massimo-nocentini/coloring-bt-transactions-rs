1//! A streaming reader for the transaction records the driver eats.
2//!
3//! The Scheme reads them with plain `(read)`, so the format is whitespace
4//! agnostic and a record may be split across lines; this reader is too, rather
5//! than assuming the one-record-per-line the sample files happen to use.
6//!
7//! Records look like
8//!
9//! ```text
10//! ((timestamp block-id tx-id _ _ _ _) (input ...) (output ...))
11//! input  = (addr-id amount prev-tx-id vout)
12//! output = (addr-id amount _)
13//! ```
14//!
15//! Only five things are ever used: the header's `block-id` and `tx-id`, each
16//! input's `amount` and `prev-tx-id`, and how many outputs there are.  Fields
17//! are picked out by position and the rest of each list is skipped, so a record
18//! with extra trailing fields still reads — unlike the Scheme's fixed-arity
19//! `match/first`.  Nothing is allocated per record.
20//!
21//! The `amount` is there for `--weighted`, which shares each input's color out
22//! in proportion to it; the unweighted driver reads it and ignores it.
23//!
24//! The buffer is ours rather than a `BufReader`, because at 149 GB the hot path
25//! is "look at one byte" and it needs to inline down to a bounds check.
26//!
27//! ## Skipped means skipped
28//!
29//! Those four values are a small minority of what a record spells out: the
30//! header gives 2 of its 7 fields to the driver, an input 2 of its 4, an output
31//! none of its 3 — an output is read only for existing.  So the wanted
32//! positions are named in a bitmask ([`HEADER_FIELDS`] and friends) and
33//! everything else goes through [`Reader::skip_int`], which walks the digits
34//! without converting them.  Turning a number into a `usize` costs a multiply
35//! and an add per digit, and there is no point paying it for a number that is
36//! dropped on the next line.
37//!
38//! ## Digits are scanned wide
39//!
40//! Finding where a run of digits ends is the one thing this reader does to
41//! nearly every byte, and it is a poor fit for a loop and a good fit for a
42//! vector compare.  [`simd::digit_run`] answers it sixteen bytes at a time, and
43//! [`simd::eight_digits`] folds eight digits into a value in three
44//! multiply-shift steps, so [`Reader::read_int`] checks for overflow once per
45//! eight digits rather than once per digit.  [`Reader::digit_run`] is the seam
46//! between those kernels and this buffer: it is what handles a run of digits
47//! that straddles a refill, so neither kernel has to know the buffer exists.
48
49use crate::simd;
50use std::io::{self, Read};
51
52const BUF_SIZE: usize = 1 << 20;
53
54/// Positions the record header is read for: the block id at 1, the tx id at 2.
55const HEADER_FIELDS: u32 = (1 << 1) | (1 << 2);
56
57/// Positions an input is read for: the amount at 1 and the previous tx id at 2.
58///
59/// The amount is only wanted by `--weighted`, but it is read either way. Making
60/// the mask depend on the mode would push a branch into the innermost parsing
61/// loop to save one field of four on runs that do not use it, and the
62/// measurement that matters here is that reading a field costs a `digit_run` and
63/// eight-digit folding, not a per-field decision.
64const INPUT_FIELDS: u32 = (1 << 1) | (1 << 2);
65
66/// An output is read for nothing at all — only for there being one.
67const OUTPUT_FIELDS: u32 = 0;
68
69pub struct Record {
70    pub block_id: usize,
71    pub tx_id: usize,
72    pub outputs: usize,
73}

75/// One input of a record: the transaction it spends, and for how much.
76///
77/// The amount is what `--weighted` shares a color out in proportion to; the
78/// unweighted driver ignores it.
79#[derive(Clone, Copy, Debug, PartialEq, Eq)]
80pub struct Input {
81    pub prev_tx_id: usize,
82    pub amount: usize,
83}

85pub struct Reader<R: Read> {
86    inner: R,
87    buf: Box<[u8]>,
88    pos: usize,
89    end: usize,
90    /// Bytes consumed strictly before `buf[pos]`, for error messages.
91    consumed: usize,
92}
93
94impl<R: Read> Reader<R> {
95    pub fn new(inner: R) -> Self {
96        Reader {
97            inner,
98            buf: vec![0u8; BUF_SIZE].into_boxed_slice(),
99            pos: 0,
100            end: 0,
101            consumed: 0,
102        }
103    }
104
105    #[inline]
106    fn peek(&mut self) -> io::Result<Option<u8>> {
107        if self.pos < self.end {
108            return Ok(Some(self.buf[self.pos]));
109        }
110        self.refill()
111    }
112
113    #[cold]
114    fn refill(&mut self) -> io::Result<Option<u8>> {
115        self.consumed += self.end;
116        self.pos = 0;
117        self.end = 0;
118        loop {
119            match self.inner.read(&mut self.buf) {
120                Ok(0) => return Ok(None),
121                Ok(n) => {
122                    self.end = n;
123                    return Ok(Some(self.buf[0]));
124                }
125                Err(e) if e.kind() == io::ErrorKind::Interrupted => continue,
126                Err(e) => return Err(e),
127            }
128        }
129    }
130
131    fn offset(&self) -> usize {
132        self.consumed + self.pos
133    }
134
135    fn err<T>(&self, msg: String) -> io::Result<T> {
136        Err(io::Error::new(
137            io::ErrorKind::InvalidData,
138            format!("{} at byte offset {}", msg, self.offset()),
139        ))
140    }
141
142    /// The next non-whitespace byte, left unconsumed.
143    #[inline]
144    fn skip_ws(&mut self) -> io::Result<Option<u8>> {
145        loop {
146            match self.peek()? {
147                Some(b) if b.is_ascii_whitespace() => self.pos += 1,
148                other => return Ok(other),
149            }
150        }
151    }
152
153    fn expect(&mut self, want: u8) -> io::Result<()> {
154        match self.skip_ws()? {
155            Some(b) if b == want => {
156                self.pos += 1;
157                Ok(())
158            }
159            Some(b) => self.err(format!(
160                "expected {:?}, found {:?}",
161                want as char, b as char
162            )),
163            None => self.err(format!("expected {:?}, found end of input", want as char)),
164        }
165    }
166
167    /// Walk the run of digits at the cursor, handing each contiguous slice of it
168    /// to `absorb`, and answer how many digits there were altogether.
169    ///
170    /// `absorb` is called more than once when the run straddles a refill.  The
171    /// buffer is a window onto the stream and [`simd::digit_run`] can only speak
172    /// for what is inside it, so a run reaching the end of the window is not
173    /// known to have ended: refill and scan again.  Keeping that here is what
174    /// lets the vector kernel work in whole 16-byte blocks without any caller
175    /// having to think about the buffer boundary.
176    #[inline]
177    fn digit_run(&mut self, mut absorb: impl FnMut(&[u8])) -> io::Result<usize> {
178        let mut digits = 0;
179        loop {
180            let run = simd::digit_run(&self.buf[self.pos..self.end]);
181            if run > 0 {
182                absorb(&self.buf[self.pos..self.pos + run]);
183                self.pos += run;
184                digits += run;
185            }
186            if self.pos < self.end {
187                // Stopped on a byte that is not a digit: the run is over.
188                return Ok(digits);
189            }
190            // The window is spent.  `peek` refills it, or says the stream has
191            // ended and the run with it.
192            if self.peek()?.is_none() {
193                return Ok(digits);
194            }
195        }
196    }
197
198    /// Read one non-negative integer.  Nothing in these records is ever
199    /// negative — the fields are timestamps, ids, counts and satoshi amounts —
200    /// so a leading `-` is a malformed record rather than a number to accept.
201    ///
202    /// The digits are folded eight at a time, which is also how often the
203    /// overflow check runs; the remainder is finished off one at a time.
204    fn read_int(&mut self) -> io::Result<usize> {
205        let first = match self.skip_ws()? {
206            Some(b) => b,
207            None => return self.err("expected an integer, found end of input".into()),
208        };
209        if first == b'-' {
210            return self.err("negative integer, but every field here is unsigned".into());
211        }
212
213        let mut value: usize = 0;
214        let mut overflowed = false;
215        let digits = self.digit_run(|run| {
216            if overflowed {
217                return;
218            }
219            let mut chunks = run.chunks_exact(8);
220            for chunk in &mut chunks {
221                let eight = simd::eight_digits(u64::from_le_bytes(chunk.try_into().unwrap()));
222                match value
223                    .checked_mul(100_000_000)
224                    .and_then(|v| v.checked_add(eight as usize))
225                {
226                    Some(v) => value = v,
227                    None => {
228                        overflowed = true;
229                        return;
230                    }
231                }
232            }
233            for &b in chunks.remainder() {
234                match value
235                    .checked_mul(10)
236                    .and_then(|v| v.checked_add((b - b'0') as usize))
237                {
238                    Some(v) => value = v,
239                    None => {
240                        overflowed = true;
241                        return;
242                    }
243                }
244            }
245        })?;
246
247        if overflowed {
248            return self.err("integer does not fit in a usize".into());
249        }
250        if digits == 0 {
251            return self.err(format!("expected an integer, found {:?}", first as char));
252        }
253
254        Ok(value)
255    }
256
257    /// Consume one integer without working out what it is.
258    ///
259    /// The cheapest way to read a number nobody wants is not to convert it, and
260    /// most of the numbers in a record are numbers nobody wants — see the module
261    /// docs.  Every malformed-input check that does not need the value is still
262    /// made: a leading `-` is refused, and so is an empty field.
263    ///
264    /// The one check given up is the overflow test, which cannot be made without
265    /// doing the arithmetic this exists to avoid.  A field too wide for a
266    /// `usize` therefore passes here where [`Reader::read_int`] would reject it.
267    /// Nothing downstream ever sees the value, so that is a rejection the run
268    /// did not depend on — and satoshi amounts, the widest fields in the format,
269    /// have four digits of headroom against a 64-bit `usize` anyway.
270    fn skip_int(&mut self) -> io::Result<()> {
271        let first = match self.skip_ws()? {
272            Some(b) => b,
273            None => return self.err("expected an integer, found end of input".into()),
274        };
275        if first == b'-' {
276            return self.err("negative integer, but every field here is unsigned".into());
277        }
278        if self.digit_run(|_| {})? == 0 {
279            return self.err(format!("expected an integer, found {:?}", first as char));
280        }
281        Ok(())
282    }
283
284    /// Read `( int int ... )`, handing the elements whose position is set in
285    /// `wanted` to `visit`, skipping the rest, and answering how many elements
286    /// there were in total.
287    ///
288    /// `wanted` is a bitmask over positions: position `n` is wanted when bit `n`
289    /// is set.  Positions from 32 up are never wanted, which is the existing
290    /// rule that fields beyond the named ones are ignored, arriving at the same
291    /// place by a cheaper route.
292    fn read_flat_list(
293        &mut self,
294        wanted: u32,
295        mut visit: impl FnMut(usize, usize),
296    ) -> io::Result<usize> {
297        self.expect(b'(')?;
298        let mut n = 0;
299        loop {
300            match self.skip_ws()? {
301                Some(b')') => {
302                    self.pos += 1;
303                    return Ok(n);
304                }
305                Some(_) => {
306                    if n < 32 && wanted & (1 << n) != 0 {
307                        let v = self.read_int()?;
308                        visit(n, v);
309                    } else {
310                        self.skip_int()?;
311                    }
312                    n += 1;
313                }
314                None => return self.err("unterminated list".into()),
315            }
316        }
317    }
318
319    /// The next record, or `None` at end of input.  `inputs` is cleared and
320    /// refilled with one [`Input`] per input of the record; pass the same vector
321    /// every time to keep this allocation-free.
322    pub fn next_record(&mut self, inputs: &mut Vec<Input>) -> io::Result<Option<Record>> {
323        inputs.clear();
324
325        if self.skip_ws()?.is_none() {
326            return Ok(None);
327        }
328        self.expect(b'(')?;
329
330        let (mut block_id, mut tx_id) = (None, None);
331        let fields = self.read_flat_list(HEADER_FIELDS, |i, v| match i {
332            1 => block_id = Some(v),
333            2 => tx_id = Some(v),
334            _ => {}
335        })?;
336        let (block_id, tx_id) = match (block_id, tx_id) {
337            (Some(b), Some(t)) => (b, t),
338            _ => {
339                return self.err(format!(
340                    "record header has {} fields, expected the block id at position 1 and the tx id at position 2",
341                    fields
342                ))
343            }
344        };
345
346        self.expect(b'(')?;
347        loop {
348            match self.skip_ws()? {
349                Some(b')') => {
350                    self.pos += 1;
351                    break;
352                }
353                Some(b'(') => {
354                    let (mut amount, mut prev_tx_id) = (None, None);
355                    let fields = self.read_flat_list(INPUT_FIELDS, |i, v| match i {
356                        1 => amount = Some(v),
357                        2 => prev_tx_id = Some(v),
358                        _ => {}
359                    })?;
360                    match (amount, prev_tx_id) {
361                        (Some(amount), Some(prev_tx_id)) => inputs.push(Input {
362                            prev_tx_id,
363                            amount,
364                        }),
365                        _ => {
366                            return self.err(format!(
367                                "input has {} fields, expected the amount at position 1 and \
368                                 the previous tx id at position 2",
369                                fields
370                            ))
371                        }
372                    }
373                }
374                _ => return self.err("expected an input".into()),
375            }
376        }
377
378        self.expect(b'(')?;
379        let mut outputs = 0;
380        loop {
381            match self.skip_ws()? {
382                Some(b')') => {
383                    self.pos += 1;
384                    break;
385                }
386                Some(b'(') => {
387                    self.read_flat_list(OUTPUT_FIELDS, |_, _| {})?;
388                    outputs += 1;
389                }
390                _ => return self.err("expected an output".into()),
391            }
392        }
393
394        self.expect(b')')?;
395        Ok(Some(Record {
396            block_id,
397            tx_id,
398            outputs,
399        }))
400    }
401}

417    /// A `Read` that hands back one byte per call, so the reader refills its
418    /// window at every single position in the stream.
419    ///
420    /// This is the point of the whole test module.  [`Reader::digit_run`] has to
421    /// stitch a run of digits back together when it straddles a refill, and with
422    /// a 1 MiB buffer that happens about once per million bytes — far too rare
423    /// for a corpus to be trusted to cover it, and the corpus cannot say *which*
424    /// digit the split fell on.  Reading a byte at a time makes every number in
425    /// the input take the straddling path, at every offset within itself.
426    struct Dribble<'a>(&'a [u8]);
427
428    impl Read for Dribble<'_> {
429        fn read(&mut self, out: &mut [u8]) -> io::Result<usize> {
430            if self.0.is_empty() || out.is_empty() {
431                return Ok(0);
432            }
433            out[0] = self.0[0];
434            self.0 = &self.0[1..];
435            Ok(1)
436        }
437    }
438
439    fn read_all(inner: impl Read) -> io::Result<Vec<Parsed>> {
440        let mut reader = Reader::new(inner);
441        let mut inputs = Vec::new();
442        let mut records = Vec::new();
443        while let Some(r) = reader.next_record(&mut inputs)? {
444            records.push((r.block_id, r.tx_id, inputs.clone(), r.outputs));
445        }
446        Ok(records)
447    }

449    /// Read `text` both in one gulp and a byte at a time, insist the two agree,
450    /// and answer what they said.
451    fn parse(text: &str) -> io::Result<Vec<Parsed>> {
452        let whole = read_all(text.as_bytes());
453        let dribbled = read_all(Dribble(text.as_bytes()));
454        match (&whole, &dribbled) {
455            (Ok(a), Ok(b)) => assert_eq!(a, b, "buffered and dribbled reads disagree"),
456            (Err(a), Err(b)) => assert_eq!(
457                a.to_string(),
458                b.to_string(),
459                "buffered and dribbled reads disagree on the error"
460            ),
461            _ => panic!("one read succeeded and the other did not: {:?}", whole),
462        }
463        whole
464    }

611    /// A long run of records with fields of mixed width, read both ways.  The
612    /// widths are chosen so digit runs land on every offset relative to the
613    /// eight-digit stride, and the whole thing is long enough that the buffered
614    /// read does several thousand records between refills while the dribbled one
615    /// refills constantly.
616    #[test]
617    fn a_long_mixed_width_stream_reads_the_same_both_ways() {
618        let mut text = String::new();
619        let mut expected = Vec::new();
620        for i in 0..400usize {
