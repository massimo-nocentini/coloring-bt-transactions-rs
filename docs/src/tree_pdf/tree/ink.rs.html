1//! # The ink the pages share
2//!
3//! The palette and the few sizes that decide how a page looks: what colour a
4//! node with children is, how small a circle may get, how much paper is kept
5//! round the drawing.  `tree-pdf` and `block-pdf` draw different things — one
6//! a spanning tree, the other the bipartite gadgets of the graph's blocks —
7//! and they must draw them in the *same* ink, so that a reader who has learnt
8//! that orange means "the cut took something here" from one figure can read
9//! the next one without a second legend.  One file both compile in is how the
10//! two are kept from drifting.
11//!
12//! It is a module rather than a library for the reason `forest` and `pdf`
13//! are: the crate's `src/*.rs` belong to the main binary, so the drawing
14//! binaries reach it by `#[path]` from `src/bin/`.
15
16// Every binary uses most of these and none uses all of them; the surface is
17// the point, so the lint goes rather than the surface.
18#![allow(dead_code)]
19
20/// A colour as the page takes them, and the tones of the viewers, plus one.
21pub type Rgb = (f64, f64, f64);
22pub const PAPER: Rgb = (1.0, 1.0, 1.0);
23pub const INNER: Rgb = (0.0, 0.0, 0.0);
24pub const LEAF: Rgb = (0.5, 0.5, 0.5);
25pub const LINK: Rgb = (0.78, 0.78, 0.78);
26/// A node whose successors the cut left out: drawn apart, so that the pruned
27/// frontier cannot pass for a fringe of true leaves.
28pub const CUT: Rgb = (0.82, 0.24, 0.10);
29
30/// The arcs a spanning tree *cannot* draw: the second parents it dropped,
31/// restored to the page behind everything, in a tone a reader reads as
32/// absent.  A tree of this graph hides three arcs in four
33/// (`sum_b (|I_b| - 1) |O_b|`, three quarters of every arc there is), and a
34/// figure that draws only what the tree kept is a figure of the renderer, not
35/// of the graph; drawn in this ink the same page is both --- the tree in
36/// [`LINK`] and, under it, the bipartite block the tree turned into a star.
37pub const GHOST: Rgb = (0.62, 0.72, 0.86);
38
39/// The dash a ghost arc is stroked with, in points at the page's own scale:
40/// on and off.  Dashed rather than merely paler because a reduction in
41/// lightness survives neither a photocopier nor a colour-blind reader,
42/// whereas a broken line stays broken.
43pub const GHOST_DASH: (f64, f64) = (1.6, 1.3);
44
45/// A node the caller named with `--mark`: the subject of the drawing --- a
46/// stolen output, a payout leg --- inked apart from everything structural.
47pub const MARK: Rgb = (0.16, 0.47, 0.84);
48
49/// The ink labels are written in, when `--labels` asks for them.
50pub const LABEL: Rgb = (0.32, 0.32, 0.32);
51
52/// How wide the drawing may be, in points, when the caller does not say.
53/// A little under a typical `\textwidth`, so the figure drops straight in.
54pub const DEFAULT_WIDTH: f64 = 420.0;
55
56/// How tall it may grow before the height binds instead.
57pub const DEFAULT_MAX_HEIGHT: f64 = 620.0;
58
59/// Nodes the walk may place when the caller does not say.  A page is legible
60/// into the tens of thousands of nodes; past that the raster of `tree-jp2` is
61/// the better picture anyway.
62pub const DEFAULT_MAX_NODES: usize = 100_000;
63
64/// Clear paper kept around the drawing, in points.
65pub const MARGIN: f64 = 4.0;
66
67/// Below this radius a circle would be finer than any press: the floor the
68/// nodes are held at, whatever the scale.
69pub const MIN_RADIUS: f64 = 0.3;
70
71/// How much of its box a node inks.  The layout puts neighbouring levels edge
72/// to edge, so circles drawn at the full box fuse into a bar along every
73/// chain; a shade under it leaves a sliver of paper between the beads.
74pub const INK: f64 = 0.85;
75
76/// And above this radius a circle is a balloon: a drawing of a dozen nodes
77/// scaled to a page would ink them shoulder to shoulder, where holding the
78/// circles down turns the same page into an airy diagram whose edges do the
79/// talking.
80pub const MAX_RADIUS: f64 = 5.0;
81
82/// A leaf is hollow — its rim inked, paper inside — from this radius up, as in
83/// the windowed viewer; below it a ring closes into a smudge and the leaf is
84/// filled like everything else.
85pub const MIN_HOLLOW: f64 = 1.0;
86
87/// How many shapes go into one path before it is painted.  One fill per
88/// colour is the idea; a bound per path is kindness to readers that walk a
89/// path recursively.
90pub const BATCH: usize = 4096;
