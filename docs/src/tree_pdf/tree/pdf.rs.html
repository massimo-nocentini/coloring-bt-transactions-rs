1//! # A page, written by hand
2//!
3//! One PDF page of vector ink — lines and circles — produced with no Cairo, no
4//! toolkit, no C library at all.  `tree-pdf` wants to run where nothing is
5//! installed but a Rust toolchain, and a single page of strokes and fills is a
6//! small enough corner of PDF to write directly, the way [`image`](../../image.rs)
7//! writes its PNGs: a header, four objects, a cross-reference table, and a
8//! content stream deflated by the `flate2` this crate already carries.
9//!
10//! # What a page holds
11//!
12//! A PDF content stream is a program for a very small machine: operands, then
13//! an operator.  The five this file ever emits are
14//!
15//! - `w`, `RG`, `rg` — line width, stroke colour, fill colour;
16//! - `m`/`l` … `S` — a polyline path, stroked;
17//! - `m`/`c` … `f` — a path of cubic Béziers, filled (`b` closes, fills and
18//!   strokes in one, for the hollow rings).
19//!
20//! A circle is not a PDF primitive, so [`Page::circle`] spells one as four
21//! cubic Béziers with the usual magic handle length `k = 0.5523` of the
22//! radius, which bows each quarter-arc to under 0.03% of a true circle —
23//! far below anything a printer resolves.
24//!
25//! Batching is the caller's affair and the reason the path operators are
26//! separate from the paint ones: naming a colour once and filling ten thousand
27//! circles in one `f` is what keeps a page of a whole subtree in the hundreds
28//! of kilobytes.
29//!
30//! # Coordinates
31//!
32//! PDF measures a page in points from the *bottom-left* corner, y running up.
33//! This file takes what it is given: the caller owns the flip from whatever
34//! screen-like frame its camera thinks in, so that this file has one job.
35//!
36//! # Why the stream is deflated
37//!
38//! The operators are decimal text, and a drawing of `n` nodes is `n` circles
39//! of four Béziers each — sixty-odd bytes of digits a curve.  `/FlateDecode`
40//! is the compression every PDF reader must implement, `flate2` is already a
41//! dependency (the PNGs go through it), and digits compress well; the pages
42//! this writes come out several times smaller than they are spelled.
43

52/// One page being drawn: its size in points, and the content stream so far.
53pub struct Page {
54    width: f64,
55    height: f64,
56    ops: String,
57}
58
59/// Numbers are written with two decimals — a fiftieth of a point, which is a
60/// two-hundredth of a millimetre — and the trailing noise of `f64` printing
61/// kept out of the file.
62fn num(v: f64) -> String {
63    let s = format!("{v:.2}");
64    // "0.50" not "0.5000000001", but also "3" not "3.00": the digits are most
65    // of the stream, so the ones that say nothing go.
66    let s = s.trim_end_matches('0').trim_end_matches('.');
67    if s.is_empty() || s == "-" || s == "-0" {
68        "0".to_string()
69    } else {
70        s.to_string()
71    }
72}

74/// The Bézier handle that makes four quarter-arcs a circle.
75const K: f64 = 0.552_284_749_830_793_4;

77impl Page {
78    pub fn new(width: f64, height: f64) -> Page {

90    pub fn set_line_width(&mut self, w: f64) {
91        let _ = writeln!(self.ops, "{} w", num(w));
92    }

94    pub fn set_stroke(&mut self, (r, g, b): (f64, f64, f64)) {
95        let _ = writeln!(self.ops, "{} {} {} RG", num(r), num(g), num(b));
96    }
97
98    pub fn set_fill(&mut self, (r, g, b): (f64, f64, f64)) {
99        let _ = writeln!(self.ops, "{} {} {} rg", num(r), num(g), num(b));
100    }

102    /// Adds one segment to the path being gathered.
103    pub fn segment(&mut self, x0: f64, y0: f64, x1: f64, y1: f64) {
104        let _ = writeln!(
105            self.ops,
106            "{} {} m {} {} l",
107            num(x0),
108            num(y0),
109            num(x1),
110            num(y1)
111        );
112    }
113
114    /// Adds one circle to the path being gathered: four cubic Béziers.
115    pub fn circle(&mut self, cx: f64, cy: f64, r: f64) {
116        let k = K * r;
117        let _ = writeln!(
118            self.ops,
119            "{} {} m {} {} {} {} {} {} c {} {} {} {} {} {} c {} {} {} {} {} {} c {} {} {} {} {} {} c h",
120            num(cx + r), num(cy),
121            num(cx + r), num(cy + k), num(cx + k), num(cy + r), num(cx), num(cy + r),
122            num(cx - k), num(cy + r), num(cx - r), num(cy + k), num(cx - r), num(cy),
123            num(cx - r), num(cy - k), num(cx - k), num(cy - r), num(cx), num(cy - r),
124            num(cx + k), num(cy - r), num(cx + r), num(cy - k), num(cx + r), num(cy),
125        );
126    }
127
128    /// Strokes the gathered path and starts a new one.
129    pub fn stroke(&mut self) {
130        self.ops.push_str("S\n");
131    }
132
133    /// Fills the gathered path and starts a new one.
134    pub fn fill(&mut self) {
135        self.ops.push_str("f\n");
136    }
137
138    /// Fills the gathered path in the fill colour and strokes it in the stroke
139    /// colour, in one operator: what a hollow ring with a rim is.
140    pub fn fill_and_stroke(&mut self) {
141        self.ops.push_str("B\n");
142    }

144    /// Writes `s` at `(x, y)` --- the baseline's left end --- in `size` points
145    /// of Helvetica, in the current fill colour.
146    ///
147    /// Helvetica because it is one of the fourteen fonts every PDF reader
148    /// carries, so a label costs the file nothing but its own characters; the
149    /// font resource is declared on the page whether or not any text is drawn,
150    /// which is a few dozen constant bytes.  The three characters PDF strings
151    /// quote --- backslash and the parentheses --- are escaped; everything
152    /// this crate labels with is digits, but a writer that only *usually*
153    /// writes well-formed files is not one.
154    pub fn text(&mut self, x: f64, y: f64, size: f64, s: &str) {
155        let mut escaped = String::with_capacity(s.len());
156        for c in s.chars() {
157            if matches!(c, '\\' | '(' | ')') {
158                escaped.push('\\');
159            }
160            escaped.push(c);
161        }
162        let _ = writeln!(
163            self.ops,
164            "BT /F1 {} Tf {} {} Td ({escaped}) Tj ET",
165            num(size),
166            num(x),
167            num(y)
168        );
169    }
170
171    /// Writes the finished page to `path`: the four objects, the deflated
172    /// stream, and a cross-reference table whose offsets are measured off the
173    /// very bytes being written.
174    pub fn write(self, path: &str) -> Result<(), String> {
175        let mut encoder = ZlibEncoder::new(Vec::new(), Compression::default());
176        encoder
177            .write_all(self.ops.as_bytes())
178            .and_then(|()| encoder.finish())
179            .map_err(|e| format!("{path}: deflating the page ({e})"))
180            .and_then(|stream| {
181                let mut out: Vec<u8> = Vec::with_capacity(stream.len() + 512);
182                out.extend_from_slice(b"%PDF-1.4\n");
183                // A comment of bytes over 127, telling transports the file is
184                // binary -- the convention every writer follows.
185                out.extend_from_slice(b"%\xE2\xE3\xCF\xD3\n");
186
187                let mut offsets = [0usize; 4];
188
189                offsets[0] = out.len();
190                out.extend_from_slice(b"1 0 obj\n<< /Type /Catalog /Pages 2 0 R >>\nendobj\n");
191
192                offsets[1] = out.len();
193                out.extend_from_slice(b"2 0 obj\n<< /Type /Pages /Kids [3 0 R] /Count 1 >>\nendobj\n");
194
195                offsets[2] = out.len();
196                out.extend_from_slice(
197                    format!(
198                        "3 0 obj\n<< /Type /Page /Parent 2 0 R /MediaBox [0 0 {} {}] \
199                         /Contents 4 0 R /Resources << /Font << /F1 << /Type /Font \
200                         /Subtype /Type1 /BaseFont /Helvetica >> >> >> >>\nendobj\n",
201                        num(self.width),
202                        num(self.height)
203                    )
204                    .as_bytes(),
205                );
206
207                offsets[3] = out.len();
208                out.extend_from_slice(
209                    format!(
210                        "4 0 obj\n<< /Length {} /Filter /FlateDecode >>\nstream\n",
211                        stream.len()
212                    )
213                    .as_bytes(),
214                );
215                out.extend_from_slice(&stream);
216                out.extend_from_slice(b"\nendstream\nendobj\n");
217
218                let xref = out.len();
219                out.extend_from_slice(b"xref\n0 5\n0000000000 65535 f \n");
220                for offset in offsets {
221                    // Twenty bytes an entry, exactly: ten digits, five digits,
222                    // a keyword and a two-byte line end, as the format demands.
223                    out.extend_from_slice(format!("{offset:010} 00000 n \n").as_bytes());
224                }
225                out.extend_from_slice(
226                    format!(
227                        "trailer\n<< /Size 5 /Root 1 0 R >>\nstartxref\n{xref}\n%%EOF\n"
228                    )
229                    .as_bytes(),
230                );
231
232                std::fs::write(path, &out).map_err(|e| format!("{path}: {e}"))
233            })
234    }
235}

242    /// The stream, un-deflated back out of a written file: the operators the
243    /// page was given, if nothing was lost on the way.
244    fn content_of(bytes: &[u8]) -> String {
245        let start = bytes
246            .windows(7)
247            .position(|w| w == b"stream\n")
248            .expect("a stream")
249            + 7;
250        let end = bytes
251            .windows(10)
252            .position(|w| w == b"\nendstream")
253            .expect("an endstream");
254
255        let mut decoder = flate2::read::ZlibDecoder::new(&bytes[start..end]);
256        let mut ops = String::new();
257        decoder.read_to_string(&mut ops).expect("a zlib stream");
258        ops
259    }
260
261    fn scratch(name: &str) -> String {
262        std::env::temp_dir()
263            .join(format!("tree-pdf-{}-{name}.pdf", std::process::id()))
264            .to_string_lossy()
265            .into_owned()
266    }

268    /// What goes in comes back: the operators survive the deflate round trip,
269    /// and the page's size is on the page.
270    #[test]
271    fn the_operators_survive_the_file() {
272        let mut page = Page::new(300.0, 200.0);
273        page.set_line_width(0.5);
274        page.set_stroke((0.78, 0.78, 0.78));
275        page.segment(10.0, 20.0, 30.0, 40.0);
276        page.stroke();
277        page.set_fill((0.0, 0.0, 0.0));
278        page.circle(50.0, 60.0, 2.0);
279        page.fill();
280
281        let path = scratch("ops");
282        page.write(&path).unwrap();
283
284        let bytes = std::fs::read(&path).unwrap();
285        assert!(bytes.starts_with(b"%PDF-1.4"));
286        assert!(bytes.ends_with(b"%%EOF\n"));
287
288        let ops = content_of(&bytes);
289        assert!(ops.contains("0.5 w\n"), "{ops}");
290        assert!(ops.contains("0.78 0.78 0.78 RG\n"));
291        assert!(ops.contains("10 20 m 30 40 l\nS\n"));
292        assert!(ops.contains("52 60 m"), "the circle starts at (cx + r, cy)");
293        assert!(ops.contains("c h\nf\n"), "four curves, closed and filled");
294
295        let media = String::from_utf8_lossy(&bytes);
296        assert!(media.contains("/MediaBox [0 0 300 200]"));
297
298        std::fs::remove_file(&path).ok();
299    }

301    /// The cross-reference table points where the objects are: each offset in
302    /// it lands on the `N 0 obj` it claims to.
303    #[test]
304    fn the_xref_points_at_the_objects() {
305        let mut page = Page::new(100.0, 100.0);
306        page.circle(50.0, 50.0, 10.0);
307        page.fill();
308
309        let path = scratch("xref");
310        page.write(&path).unwrap();
311        let bytes = std::fs::read(&path).unwrap();
312        let text = String::from_utf8_lossy(&bytes);
313
314        let xref_at: usize = text
315            .rsplit_once("startxref\n")
316            .and_then(|(_, rest)| rest.split_whitespace().next())
317            .and_then(|n| n.parse().ok())
318            .expect("a startxref");
319        assert!(bytes[xref_at..].starts_with(b"xref"));
320
321        // Everything from the table on is plain ASCII; the deflated stream
322        // before it is not, so the byte offset is honoured before the text is.
323        let table = String::from_utf8_lossy(&bytes[xref_at..]).into_owned();
324        for (entry, object) in table.lines().skip(3).take(4).zip(1..) {
325            let offset: usize = entry[..10].parse().unwrap();
326            let expected = format!("{object} 0 obj");
327            assert!(
328                bytes[offset..].starts_with(expected.as_bytes()),
329                "object {object} is not at {offset}"
330            );
331        }
332
333        std::fs::remove_file(&path).ok();
334    }

336    /// A label lands in the stream as a text object in the declared font, and
337    /// the characters PDF strings quote come out escaped.
338    #[test]
339    fn text_is_written_and_escaped() {
340        let mut page = Page::new(100.0, 100.0);
341        page.set_fill((0.32, 0.32, 0.32));
342        page.text(10.0, 20.0, 5.0, "123456789");
343        page.text(10.0, 30.0, 5.0, r"a(b)c\d");
344
345        let path = scratch("text");
346        page.write(&path).unwrap();
347        let bytes = std::fs::read(&path).unwrap();
348
349        let ops = content_of(&bytes);
350        assert!(ops.contains("BT /F1 5 Tf 10 20 Td (123456789) Tj ET"), "{ops}");
351        assert!(ops.contains(r"(a\(b\)c\\d) Tj"), "{ops}");
352        assert!(
353            String::from_utf8_lossy(&bytes).contains("/BaseFont /Helvetica"),
354            "the font the text names is declared on the page"
355        );
356
357        std::fs::remove_file(&path).ok();
358    }
