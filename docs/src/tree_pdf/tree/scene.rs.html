1//! # The drawing, flattened
2//!
3//! Between the layout and the window there is one more shape to give the tree.
4//! The arena the layout works in is built for the layout: every node carries a
5//! level, a child number, contour links, a `Vec` of children.  A window wants
6//! none of that.  It wants, a hundred times a second, "which nodes are on
7//! screen", and then for each of them a place, a colour, and something to say
8//! when it is clicked.
9//!
10//! So the arena is walked once, into flat arrays, and dropped.  What is kept is
11//! two vectors and a [`Quadtree`] over them:
12//!
13//! - `xy`, a centre per node, which is what the index and the ink both want and
14//!   the only thing either of them wants often.
15//! - `placed`, a node number and two links per node, touched only for the few
16//!   nodes a frame actually draws.
17//!
18//! The split is not tidiness.  A coarse frame reads *every* visible node's
19//! position and none of its metadata, so keeping the positions apart is the
20//! difference between walking 16 bytes a node and walking 32.
21//!
22//! # Pre-order, and what it buys
23//!
24//! The walk is depth first, and a node's descendants are therefore the entries
25//! that follow it, contiguously, up to [`Placed::subtree_end`].  A subtree is a
26//! *range*.  That is the whole of the machinery behind selecting one: framing a
27//! subtree is a scan of a slice, asking whether a node is in the selection is
28//! two comparisons, and neither needs the arena, which by then is gone.
29//!
30//! Ranges also mean the ends can be filled in by one backward sweep instead of
31//! by the walk: in pre-order a parent always precedes its children, so passing
32//! down the array from the end and pushing each node's end up to its parent
33//! settles every one of them.
34//!
35//! # The node the graph does not have
36//!
37//! [`forest`](super) may add a root standing for nothing, to give the layout the
38//! single root it needs.  It is not in the drawing, so it is not in the scene:
39//! the walk starts at its children instead, which become parentless nodes here
40//! exactly as the several roots of the graph are.
41//!
42//! # The walk carries its own stack
43//!
44//! A chain of transactions spending each other is a tree a million levels deep,
45//! and a depth-first walk that recurses is a million stack frames.  The stack
46//! here is a `Vec`, which has no such ceiling — the same reason `forest` asks
47//! the layout crate for its flat entry point.
48

61/// What a root has instead of a parent.
62pub const NO_PARENT: u32 = u32::MAX;
63
64/// Everything about a node that is not its position.
65#[derive(Clone, Copy, Debug)]
66pub struct Placed {
67    /// The node's number in the graph that was read.
68    pub graph: u32,
69    /// The scene index of the node's parent, or [`NO_PARENT`].
70    pub parent: u32,
71    /// One past the last scene index of this node's subtree.
72    ///
73    /// Equal to the node's own index plus one exactly when it is a leaf, which
74    /// is how [`Scene::is_leaf`] answers without looking at any children.
75    pub subtree_end: u32,
76}
77
78/// A laid-out drawing, ready to be looked at.
79pub struct Scene {
80    xy: Vec<[f64; 2]>,
81    placed: Vec<Placed>,
82    index: Quadtree,
83    bounds: Rect,
84    radius: f64,
85}
86
87impl Scene {
88    /// Flattens `arena`, laid out and rooted at `root`, and indexes it.
89    ///
90    /// Refused rather than truncated when the drawing has more nodes than a
91    /// `u32` can name: the links here are indices, and a silently wrapped one
92    /// would draw a picture of the wrong tree.
93    pub fn of(arena: &Arena, root: NodeId) -> Result<Scene, String> {
94        if arena.len() > u32::MAX as usize {
95            return Err(format!(
96                "{} nodes is more than this can index; the limit is {}",
97                arena.len(),
98                u32::MAX
99            ));
100        }
101
102        let mut xy: Vec<[f64; 2]> = Vec::with_capacity(arena.len());
103        let mut placed: Vec<Placed> = Vec::with_capacity(arena.len());
104
105        // (node, the scene index of the nearest drawn ancestor).
106        let mut stack: Vec<(NodeId, u32)> = vec![(root, NO_PARENT)];
107
108        while let Some((id, parent)) = stack.pop() {
109            let node = &arena[id];
110
111            let parent = if node.isdummy {
112                // It stands for nothing, so it is neither drawn nor a parent:
113                // its children inherit whatever it inherited.
114                parent
115            } else {
116                let here = placed.len() as u32;
117                xy.push([node.x + node.w / 2.0, node.y + node.h / 2.0]);
118                placed.push(Placed {
119                    // `forest` numbers the arena one-based, keeping 0 for the
120                    // node standing for nothing, which never reaches here.
121                    graph: (node.idx - 1) as u32,
122                    parent,
123                    subtree_end: here + 1,
124                });
125                here
126            };
127
128            // Reversed, so that the first child is the first one off the stack
129            // and the array comes out in the order the tree reads.
130            for &child in node.children().iter().rev() {
131                stack.push((child, parent));
132            }
133        }
134
135        // A parent precedes its children, so one pass from the end settles every
136        // subtree's extent: each node hands its own end up, already final.
137        for i in (0..placed.len()).rev() {
138            let parent = placed[i].parent;
139            if parent != NO_PARENT {
140                let end = placed[i].subtree_end;
141                let up = &mut placed[parent as usize].subtree_end;
142                *up = (*up).max(end);
143            }
144        }
145
146        // Every real node is the same box, so one radius describes them all; it
147        // is read off the drawing rather than assumed so that this file does not
148        // have to agree with `forest` about a constant.
149        let radius = arena
150            .iter()
151            .find(|n| !n.isdummy)
152            .map_or(0.5, |n| n.w.max(n.h) / 2.0);
153
154        let mut bounds = Rect::nothing();
155        for p in &xy {
156            bounds.add(p[0], p[1]);
157        }
158        let bounds = bounds.grown(radius);
159
160        let index = Quadtree::over(&xy);
161
162        Ok(Scene { xy, placed, index, bounds, radius })
163    }
164
165    /// How many nodes are drawn.
166    pub fn len(&self) -> usize {
167        self.placed.len()
168    }
169
170    pub fn is_empty(&self) -> bool {
171        self.placed.is_empty()
172    }
173
174    /// The whole drawing, circles included, which is what to frame to see it all.
175    pub fn bounds(&self) -> Rect {
176        self.bounds
177    }
178
179    /// Half the side of a node's box: a circle's radius, in node units.
180    pub fn radius(&self) -> f64 {
181        self.radius
182    }
183
184    /// The index, for a caller that wants to report on it.
185    pub fn index(&self) -> &Quadtree {
186        &self.index
187    }
188
189    pub fn at(&self, i: u32) -> [f64; 2] {
190        self.xy[i as usize]
191    }
192
193    pub fn node(&self, i: u32) -> Placed {
194        self.placed[i as usize]
195    }
196
197    /// Whether the node has children, told from the extent of its subtree.
198    pub fn is_leaf(&self, i: u32) -> bool {
199        self.placed[i as usize].subtree_end == i + 1
200    }
201
202    /// The nodes of `i`'s subtree, `i` itself first: a contiguous range, which
203    /// is what the pre-order walk was for.
204    pub fn subtree(&self, i: u32) -> std::ops::Range<u32> {
205        i..self.placed[i as usize].subtree_end
206    }
207
208    /// The box holding a subtree's circles, for a camera to frame.
209    pub fn subtree_bounds(&self, i: u32) -> Rect {
210        let mut bounds = Rect::nothing();
211        let subtree = self.subtree(i);
212        for p in &self.xy[subtree.start as usize..subtree.end as usize] {
213            bounds.add(p[0], p[1]);
214        }
215        bounds.grown(self.radius)
216    }
217
218    /// Hands `patch` the nodes inside `seen`, summarising cells narrower than
219    /// `resolution`.  See [`Quadtree::visit`].
220    pub fn visit(&self, seen: Rect, resolution: f64, patch: &mut impl FnMut(Patch<'_>)) {
221        self.index.visit(seen, resolution, patch);
222    }
223
224    /// The node nearest `(x, y)` within `within` of it: what a click asks.
225    pub fn pick(&self, x: f64, y: f64, within: f64) -> Option<u32> {
226        self.index.nearest(&self.xy, x, y, within)
227    }
228}

235    /// Builds and lays out an arena from `(parent, child)` pairs over nodes
236    /// `0..n`, with `root` the one nothing points at.  Node `i` of the graph is
237    /// `idx` `i + 1`, as `forest` numbers them.
238    fn laid_out(n: usize, arcs: &[(usize, usize)], roots: &[usize]) -> (Arena, NodeId) {
239        let mut arena = Arena::new();
240        let ids: Vec<NodeId> = (0..n).map(|v| arena.add_node(v + 1, 1.0, 1.0, 1.0, false)).collect();
241        for &(p, c) in arcs {
242            arena.push_child(ids[p], ids[c]);
243        }
244
245        let root = if roots.len() == 1 {
246            ids[roots[0]]
247        } else {
248            let r = arena.add_node(0, 0.0, 0.0, 0.0, true);
249            arena.set_children(r, &roots.iter().map(|&v| ids[v]).collect::<Vec<_>>());
250            r
251        };
252
253        let mut input = LayoutInput::new(root);
254        input.vertically = false;
255        layout(&mut arena, &input);
256        (arena, root)
257    }

259    /// The walk is pre-order, and subtrees are the ranges that follow.
260    #[test]
261    fn a_subtree_is_a_range() {
262        //        0
263        //      /   \
264        //     1     4
265        //    / \
266        //   2   3
267        let (arena, root) = laid_out(5, &[(0, 1), (0, 4), (1, 2), (1, 3)], &[0]);
268        let scene = Scene::of(&arena, root).unwrap();
269
270        assert_eq!(scene.len(), 5);
271        let order: Vec<u32> = (0..5).map(|i| scene.node(i).graph).collect();
272        assert_eq!(order, [0, 1, 2, 3, 4], "depth first, eldest child first");
273
274        assert_eq!(scene.subtree(0), 0..5, "the root is the whole drawing");
275        assert_eq!(scene.subtree(1), 1..4, "1 and the two under it");
276        assert_eq!(scene.subtree(2), 2..3, "a leaf is itself");
277        assert_eq!(scene.subtree(4), 4..5);
278
279        assert!(!scene.is_leaf(0) && !scene.is_leaf(1));
280        assert!(scene.is_leaf(2) && scene.is_leaf(3) && scene.is_leaf(4));
281
282        assert_eq!(scene.node(0).parent, NO_PARENT);
283        assert_eq!(scene.node(2).parent, 1);
284        assert_eq!(scene.node(4).parent, 0);
285    }

352    /// A chain a hundred thousand levels deep, walked on an ordinary stack.
353    ///
354    /// The shape a run of transactions spending each other's output has, and the
355    /// reason the walk here holds its frontier in a `Vec`.
356    #[test]
357    fn a_chain_deeper_than_any_stack() {
358        let n = 100_000;
359        let mut arena = Arena::with_capacity(n);
360        let ids: Vec<NodeId> = (0..n).map(|v| arena.add_node(v + 1, 1.0, 1.0, 1.0, false)).collect();
361        for v in 0..n - 1 {
362            arena.push_child(ids[v], ids[v + 1]);
363        }
364        let root = ids[0];
365
366        let mut input = LayoutInput::new(root);
367        input.vertically = false;
368        non_layered_tidy_trees::flat::layout_flat(&mut arena, &input);
369
370        let scene = Scene::of(&arena, root).unwrap();
371
372        assert_eq!(scene.len(), n);
373        assert_eq!(scene.subtree(0), 0..n as u32, "the whole chain hangs off the top");
374        assert_eq!(scene.subtree(n as u32 - 1), n as u32 - 1..n as u32);
375        assert_eq!(scene.node(n as u32 - 1).parent, n as u32 - 2);
376    }
