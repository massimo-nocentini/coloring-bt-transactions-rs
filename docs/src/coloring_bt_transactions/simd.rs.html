1//! # Wide byte scanning for the record reader
2//!
3//! [`sexp`](crate::sexp) walks 149 GB one byte at a time.  Two of the things it
4//! does to every byte are shaped exactly like vector work, so they are done here
5//! instead, sixteen or eight bytes at a stride:
6//!
7//! - [`digit_run`] — *how long is the run of digits starting here?*  One
8//!   compare against a 16-byte load answers what a scalar loop asks sixteen
9//!   times.  This is a true vector operation and is written against NEON on
10//!   aarch64 and SSE2 on x86-64, both of which are baseline for their target, so
11//!   no runtime feature detection and no dispatch is needed.
12//!
13//! - [`eight_digits`] — *what number do these eight digits spell?*  This one is
14//!   SWAR, "SIMD within a register": the eight bytes ride in one `u64` and three
15//!   multiply-shift-mask steps fold them pairwise into a single value.  It is
16//!   plain integer arithmetic, so it is portable, and it is the right tool
17//!   anyway — the answer has to end up in a scalar register regardless, and no
18//!   field here is longer than 20 digits.
19//!
20//! ## The merge's kernels
21//!
22//! The rest of this file serves the weighted merge in [`crate::weighted`]:
23//! the elementwise scaling loops that the compiler vectorises on its own
24//! ([`scale_into_uninit`], [`scale_add_into_uninit`], and their initialised
25//! twins that the tests hold them against) and its two run finders
26//! ([`leading_below`], [`common_prefix`]).  [`common_prefix`] is the file's
27//! second hand-written kernel and, like [`digit_run`], it is written twice —
28//! NEON and SSE2 — because both are baseline and neither needs dispatch.
29//!
30//! ## Why not the rest of the parser
31//!
32//! `skip_ws` is left scalar on purpose.  These records separate tokens with a
33//! single space, so it skips zero or one bytes essentially every time, and a
34//! 16-byte load to find that out would cost more than the loop it replaces.
35//! Vectors pay off over runs, and whitespace here does not run.
36//!
37//! ## Correctness
38//!
39//! Every vector path has a scalar twin ([`digit_run_scalar`]) and the tests at
40//! the bottom of this file assert the two agree — across alignments, across
41//! lengths either side of the 16-byte stride, and on random bytes.  That is the
42//! only real defence against a hand-written vector kernel: the scalar version is
43//! obviously right, so make the fast one prove it matches.
44

47/// How many bytes at the front of `bytes` are ASCII digits.
48///
49/// A return of `bytes.len()` means the run was not seen to end — the caller is
50/// looking at a window, and the run may continue into whatever comes next.
51#[inline]
52pub fn digit_run(bytes: &[u8]) -> usize {
53    #[cfg(target_arch = "aarch64")]
54    {
55        // SAFETY: NEON is baseline for every aarch64 target, so these intrinsics
56        // are always available; `digit_run_neon` reads only within `bytes`.
57        unsafe { digit_run_neon(bytes) }
58    }
59    #[cfg(target_arch = "x86_64")]
60    {
61        // SAFETY: SSE2 is baseline for every x86-64 target, as above.
62        unsafe { digit_run_sse2(bytes) }
63    }
64    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
65    {
66        digit_run_scalar(bytes)
67    }
68}

70/// The obviously-correct version, and the oracle the vector paths are tested
71/// against.  Also the tail handler for both of them.
72#[inline]
73pub fn digit_run_scalar(bytes: &[u8]) -> usize {
74    let mut i = 0;
75    while i < bytes.len() && bytes[i].is_ascii_digit() {
76        i += 1;
77    }
78    i
79}

81/// `digit_run` on NEON.
82///
83/// A byte is a digit exactly when `b - b'0'` is 9 or less as an *unsigned*
84/// byte — one subtract and one compare, no range pair.  Locating the first
85/// failure is the only fiddly part: aarch64 has no `movemask`, so the usual
86/// stand-in is `vshrn` by 4, which narrows the 16 lanes of 0x00/0xFF into 16
87/// nibbles of a single `u64`.  The first non-digit is then at nibble
88/// `trailing_zeros() / 4`.
89#[cfg(target_arch = "aarch64")]
90#[inline]
91unsafe fn digit_run_neon(bytes: &[u8]) -> usize {
92    use core::arch::aarch64::*;
93
94    let n = bytes.len();
95    let mut i = 0;
96    while i + 16 <= n {
97        let chunk = vld1q_u8(bytes.as_ptr().add(i));
98        let shifted = vsubq_u8(chunk, vdupq_n_u8(b'0'));
99        // 0xFF in every lane that is *not* a digit.
100        let bad = vcgtq_u8(shifted, vdupq_n_u8(9));
101        let nibbles = vget_lane_u64::<0>(vreinterpret_u64_u8(vshrn_n_u16::<4>(
102            vreinterpretq_u16_u8(bad),
103        )));
104        if nibbles != 0 {
105            return i + (nibbles.trailing_zeros() >> 2) as usize;
106        }
107        i += 16;
108    }
109    i + digit_run_scalar(&bytes[i..])
110}

112/// `digit_run` on SSE2.
113///
114/// Same `b - b'0' <= 9` test, but SSE2 compares are signed only, so the
115/// subtracted byte is biased by 0x80 first: `x ^ 0x80 < 10 ^ 0x80` as signed is
116/// `x < 10` as unsigned.  `movemask` then hands back the 16 lane results as 16
117/// bits directly, which is the part aarch64 has to fake.
118#[cfg(target_arch = "x86_64")]
119#[inline]
120unsafe fn digit_run_sse2(bytes: &[u8]) -> usize {
121    use core::arch::x86_64::*;
122
123    let n = bytes.len();
124    let mut i = 0;
125    while i + 16 <= n {
126        let chunk = _mm_loadu_si128(bytes.as_ptr().add(i) as *const __m128i);
127        let shifted = _mm_sub_epi8(chunk, _mm_set1_epi8(b'0' as i8));
128        let biased = _mm_xor_si128(shifted, _mm_set1_epi8(-128));
129        // 10 ^ 0x80 == 0x8A == -118.
130        let good = _mm_cmplt_epi8(biased, _mm_set1_epi8(-118));
131        // One bit per lane, set where the byte *is* a digit.
132        let bad = !(_mm_movemask_epi8(good) as u32) & 0xFFFF;
133        if bad != 0 {
134            return i + bad.trailing_zeros() as usize;
135        }
136        i += 16;
137    }
138    i + digit_run_scalar(&bytes[i..])
139}

141/// The value of eight ASCII digits packed into `chunk`, the first digit in the
142/// lowest-addressed byte.  Callers get there with [`u64::from_le_bytes`], which
143/// keeps this correct on a big-endian target too.
144///
145/// Three rounds of divide-and-conquer, each folding neighbouring lanes into
146/// lanes twice as wide.  The multiplier in each round is two set fields — `1`
147/// and the power of ten that weights the lane above — so one multiply does the
148/// whole `hi * 10^k + lo` for every pair at once, and the shift-and-mask picks
149/// the combined halves back out:
150///
151/// ```text
152///  '1' '2' '3' '4' '5' '6' '7' '8'   eight bytes
153///   \_/     \_/     \_/     \_/      x 2561      = 10<<8   | 1
154///   12      34      56      78       four u16
155///     \____/          \____/         x 6553601   = 100<<16 | 1
156///     1234            5678           two u32
157///        \____________/              x 42949672960001 = 10000<<32 | 1
158///           12345678                 one u64
159/// ```
160///
161/// No lane can carry into its neighbour on the way: the largest intermediate any
162/// round produces is 99, 9999, 99999999, each one inside its lane.
163///
164/// The caller is responsible for the bytes actually being digits — feeding this
165/// anything else yields a meaningless number rather than an error.
166#[inline]
167pub fn eight_digits(chunk: u64) -> u64 {
168    let value = (chunk & 0x0F0F_0F0F_0F0F_0F0F).wrapping_mul(2561) >> 8;
169    let value = (value & 0x00FF_00FF_00FF_00FF).wrapping_mul(6_553_601) >> 16;
170    (value & 0x0000_FFFF_0000_FFFF).wrapping_mul(42_949_672_960_001) >> 32
171}

173/// `dst[k] = src[k] * factor`, for as many elements as both slices hold.
174///
175/// This is the piece of the weighted merge that vectorises, and it is left to
176/// the compiler rather than written by hand.  A flat elementwise multiply over
177/// two non-overlapping slices is exactly the shape LLVM's loop vectoriser is
178/// built for: `zip` gives it the length up front and rules out the aliasing
179/// question, so it emits `fmul` over two `f64` lanes on NEON and four on AVX
180/// without being asked.  The tests below check the arithmetic; the disassembly
181/// is what checks the vectorisation, and there is a `make asm-check` for it.
182///
183/// Hand-written intrinsics were tried for the *merge* itself and lost — see
184/// [`digit_run`] for the shape that does pay off, and the merge in
185/// [`crate::weighted`] for why the comparison loop does not.
186///
187/// Since the merge started writing into fresh allocations it goes through
188/// [`scale_into_uninit`], and this is the plain twin the tests hold that one
189/// against — hence the allowance below outside of test builds.
190#[cfg_attr(not(test), allow(dead_code))]
191#[inline]
192pub fn scale_into(dst: &mut [f64], src: &[f64], factor: f64) {
193    for (out, &value) in dst.iter_mut().zip(src) {
194        *out = value * factor;
195    }
196}

198/// `dst[k] = a[k] * fa + b[k] * fb`, for as many elements as all three hold.
199///
200/// The blocks that both colors carry, once the merge has lined them up.  Same
201/// reasoning as [`scale_into`]: written plainly so the vectoriser can take it.
202///
203/// It does *not* fuse into a multiply-add, and the doc used to say it did.
204/// aarch64 has `fmla` and LLVM declines to use it: contracting `x * fa + y * fb`
205/// changes the rounding, and nothing in this crate turns on the fast-math flag
206/// that would licence that.  The cross-built aarch64 binary of 2026-09-07
207/// (rustc 1.98.0, `--release`, `aarch64-unknown-linux-gnu`) contains zero `fmla`
208/// and zero `fmls` in its whole text, jemalloc and all — 0 matches in 203,593
209/// disassembled lines.  On the default x86-64 target the question does not
210/// arise; that is plain SSE2 and has no FMA to fuse into.  What the vectoriser
211/// does take is the separate two-lane multiply and add, which is what
212/// `make asm-check` counts.
213#[cfg_attr(not(test), allow(dead_code))]
214#[inline]
215pub fn scale_add_into(dst: &mut [f64], a: &[f64], fa: f64, b: &[f64], fb: f64) {
216    for ((out, &x), &y) in dst.iter_mut().zip(a).zip(b) {
217        *out = x * fa + y * fb;
218    }
219}

221/// [`scale_into`] writing into memory that has not been initialised yet.
222///
223/// The merge in [`crate::weighted`] used to build into a zero-filled staging
224/// buffer and copy the result into its final allocation, which at depth in the
225/// 2022 chain was half its time: the zero fill, the copy, and a `malloc` per
226/// colour.  Writing straight into the fresh allocation needs a kernel that is
227/// allowed to see uninitialised destination memory, which is the whole
228/// difference between this and [`scale_into`] — the loop body is the same store
229/// of the same product and vectorises the same way.  Every element of `dst` that
230/// is covered by `src` is initialised on return.
231#[inline]
232pub fn scale_into_uninit(dst: &mut [MaybeUninit<f64>], src: &[f64], factor: f64) {
233    for (out, &value) in dst.iter_mut().zip(src) {
234        out.write(value * factor);
235    }
236}

238/// [`scale_add_into`] writing into uninitialised memory, as
239/// [`scale_into_uninit`] does for [`scale_into`].
240#[inline]
241pub fn scale_add_into_uninit(
242    dst: &mut [MaybeUninit<f64>],
243    a: &[f64],
244    fa: f64,
245    b: &[f64],
246    fb: f64,
247) {
248    for ((out, &x), &y) in dst.iter_mut().zip(a).zip(b) {
249        out.write(x * fa + y * fb);
250    }
251}

253/// How many leading elements `a` and `b` have in common, position for position.
254///
255/// The merge's other run finder.  Two colours that share an ancestor share whole
256/// stretches of its support, and once the merge has lined one up it has to find
257/// where it ends: a scalar loop does that a block at a time, one dependent
258/// compare per element, and at the depths where colours run to a hundred
259/// thousand blocks that loop was a fifth of the merge.  Four `u32` lanes at a
260/// time is the same compare four times wider, and finding the first lane that
261/// differs is one mask and a count of trailing bits.
262///
263/// Until 2026-09-07 only x86-64 had a kernel here and aarch64 fell through to
264/// [`common_prefix_scalar`] — the paragraph above says what that costs, and it
265/// said it while one of the two targets was paying it.  `common_prefix_neon`
266/// closes that; it is the arm the tests at the bottom of this file exercise
267/// under `qemu-aarch64`.
268#[inline]
269pub fn common_prefix(a: &[u32], b: &[u32]) -> usize {
270    #[cfg(target_arch = "aarch64")]
271    {
272        // SAFETY: NEON is baseline for every aarch64 target; the kernel reads
273        // only within both slices.
274        unsafe { common_prefix_neon(a, b) }
275    }
276    #[cfg(target_arch = "x86_64")]
277    {
278        // SAFETY: SSE2 is baseline for every x86-64 target, as above.
279        unsafe { common_prefix_sse2(a, b) }
280    }
281    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
282    {
283        common_prefix_scalar(a, b)
284    }
285}

287/// The oracle for [`common_prefix`], and its tail.
288#[inline]
289pub fn common_prefix_scalar(a: &[u32], b: &[u32]) -> usize {
290    let n = a.len().min(b.len());
291    let mut i = 0;
292    while i < n && a[i] == b[i] {
293        i += 1;
294    }
295    i
296}

298/// `common_prefix` on SSE2: four lanes compared for equality, a mask of the
299/// lanes that matched, and the first zero bit of the mask is where the prefix
300/// ends.  Two 16-byte loads per step against the eight bytes a scalar step
301/// compares.
302#[cfg(target_arch = "x86_64")]
303#[inline]
304unsafe fn common_prefix_sse2(a: &[u32], b: &[u32]) -> usize {
305    use core::arch::x86_64::*;
306
307    let n = a.len().min(b.len());
308    let mut i = 0;
309    while i + 4 <= n {
310        let x = _mm_loadu_si128(a.as_ptr().add(i) as *const __m128i);
311        let y = _mm_loadu_si128(b.as_ptr().add(i) as *const __m128i);
312        // One bit per byte, set where the bytes agree; a lane agrees when all
313        // four of its bits do.
314        let same = _mm_movemask_epi8(_mm_cmpeq_epi32(x, y)) as u32;
315        if same != 0xFFFF {
316            return i + (same.trailing_ones() >> 2) as usize;
317        }
318        i += 4;
319    }
320    i + common_prefix_scalar(&a[i..], &b[i..])
321}

323/// `common_prefix` on NEON.
324///
325/// `vceqq_u32` fills each of the four lanes with all ones where the two words
326/// agree, and then the mask has to come back out of the vector register the
327/// same way `digit_run_neon` gets its own out: aarch64 has no `movemask`, so
328/// `vshrn` by 4 narrows the eight `u16` halves of the register into the eight
329/// bytes of one `u64`, one nibble per byte of the input.
330///
331/// The trap is the divisor.  The nibbles count *input bytes*, and a `u32` lane
332/// is four of them, so a lane that matches sets sixteen bits here where a
333/// matching byte in `digit_run_neon` sets four: the first differing lane is at
334/// `trailing_ones() / 16`, not `/ 4`.  Copying either twin's `>> 2` across
335/// unchanged reports four times the answer and the merge reads past the run.
336///
337/// Reading the matches rather than the mismatches, and so `trailing_ones`
338/// against `u64::MAX` rather than `trailing_zeros` against zero, is a deliberate
339/// copy of `common_prefix_sse2`: it saves the `vmvnq_u32` that inverting the
340/// mask would cost, and the two kernels then read as the same kernel twice.
341///
342/// ## What the tests actually pin, and what they do not
343///
344/// Nine mutants of this function, each built and run against the four
345/// `common_prefix` tests under `qemu-aarch64` on 2026-09-07:
346///
347/// ```text
348///   trailing_ones() >> 2, the SSE2 divisor      3 of 4 tests fail
349///   trailing_ones() >> 3                        3 of 4 tests fail
350///   trailing_zeros() vs 0, not ones vs MAX      3 of 4 tests fail
351///   i += 8 in place of i += 4                   4 of 4 tests fail
352///   vceqq_u8 in place of vceqq_u32              all pass
353///   vshrn_n_u16::<1>, ::<2>, ::<8> for ::<4>    all pass
354///   vshrn_n_u16::<9> and ::<16>                 do not compile
355/// ```
356///
357/// The survivors survive for a reason rather than for want of a test, and both
358/// reasons are worth knowing.
359///
360/// The shift amount is inert.  After a compare every lane is `0x0000` or
361/// `0xFFFF`, so any narrow leaves the truncated byte `0x00` or `0xFF` — and
362/// `vshrn_n_u16` accepts only 1 through 8, so *every legal* argument gives this
363/// function the same behaviour.  LLVM sees it too: the emitted instruction is
364/// `xtn v0.8b, v0.8h`, a plain narrow with no shift at all.
365///
366/// The compare width is inert for a different reason: byte equality decides
367/// word equality here, because the first differing byte and the first differing
368/// word are the same quotient by four, which is the divisor already being
369/// applied.  `::<4>` and `vceqq_u32` are kept because they say what is meant —
370/// but no test distinguishes them from the alternatives, since the alternatives
371/// are not behavioural differences, and claiming the tests pin them would be
372/// worse than writing this down.
373#[cfg(target_arch = "aarch64")]
374#[inline]
375unsafe fn common_prefix_neon(a: &[u32], b: &[u32]) -> usize {
376    use core::arch::aarch64::*;
377
378    let n = a.len().min(b.len());
379    let mut i = 0;
380    while i + 4 <= n {
381        let x = vld1q_u32(a.as_ptr().add(i));
382        let y = vld1q_u32(b.as_ptr().add(i));
383        // Sixteen set bits per lane that agrees, zero per lane that does not.
384        let same = vget_lane_u64::<0>(vreinterpret_u64_u8(vshrn_n_u16::<4>(
385            vreinterpretq_u16_u32(vceqq_u32(x, y)),
386        )));
387        if same != u64::MAX {
388            return i + (same.trailing_ones() >> 4) as usize;
389        }
390        i += 4;
391    }
392    i + common_prefix_scalar(&a[i..], &b[i..])
393}

395/// How many leading elements of the ascending `xs` are below `key`.
396///
397/// The merge asks this of the side that is behind: everything up to the other
398/// side's next block is a run to copy whole.  A binary search over the rest of
399/// the array answers in `log(remaining)` probes, but those probes land far ahead
400/// of the position the merge is streaming through, so each is a cache miss on a
401/// colour that does not fit in L2 — and most runs are short.  Galloping probes
402/// at 1, 2, 4, ... from the current position instead, so the cost is
403/// `2 log(run)` touches of memory that is about to be read anyway; only the last
404/// doubling is bisected.  Same answer as `xs.partition_point(|&x| x < key)`,
405/// which the tests assert.
406#[inline]
407pub fn leading_below(xs: &[u32], key: u32) -> usize {
408    let n = xs.len();
409    // `xs[..low]` are known to be below `key`; `high` is the next probe.
410    let (mut low, mut high) = (0usize, 1usize);
411    while high < n && xs[high] < key {
412        low = high + 1;
413        high *= 2;
414    }
415    let high = high.min(n);
416    low + xs[low..high].partition_point(|&x| x < key)
417}

558                let mut b = a.clone();

574            }
575        }

590    /// Differences finer than a lane: the two arrays differ in exactly one
591    /// *byte* of exactly one word, at every position through several 4-lane
592    /// blocks and at each of the four bytes of the word.  A word whose bytes
593    /// mostly agree is the case a vector compare could get wrong at some
594    /// granularity between byte and word, and the merge sees it constantly —
595    /// consecutive block ids differ only in their low byte.  It is one of the
596    /// three tests that the wrong nibbles-per-lane divisor fails; the mutant
597    /// table on `common_prefix_neon` says which mutants no test catches.
598    #[test]
599    fn common_prefix_finds_a_single_differing_byte() {
600        let a: Vec<u32> = (0..24).map(|k| 0x0101_0101 * (k + 1)).collect();
601        for len in 0..24usize {
602            for byte in 0..4u32 {
603                let mut b = a.clone();
604                b[len] ^= 1 << (8 * byte);
605                assert_eq!(
606                    common_prefix(&a, &b),
607                    len,
608                    "differing byte {} of word {}",
609                    byte,
610                    len
611                );
612                assert_eq!(common_prefix(&a, &b), common_prefix_scalar(&a, &b));
613            }
614        }
615    }
