1//! # A perceptual colour space, and the ramp the picture is drawn in
2//!
3//! [`crate::image`]'s weighted pixel is a grey: `weight^(1/2.2)`, dark for a
4//! block that carried most of a transaction's value and pale for one that
5//! carried a trace.  Grey has 254 steps between paper and ink and the eye reads
6//! perhaps thirty of them, which for a quantity whose interesting range is a
7//! fraction of a percent is most of the picture thrown away.
8//!
9//! A colour ramp reads further, but only if it is built in a space where equal
10//! steps *look* equal.  That is the whole reason this module exists and the
11//! reason it is not HSL: HSL's "lightness" is not luminance and its hue circle
12//! is wildly non-uniform — yellow at 60° and blue at 240° are nominally the
13//! same lightness and are nothing of the sort — so a ramp built in it has
14//! bands where the eye sees a step that is not there and stretches where it
15//! misses steps that are.  For a picture whose whole purpose is comparing
16//! magnitudes that is not a cosmetic problem.
17//!
18//! [Oklab](https://bottosson.github.io/posts/oklab/) is a perceptual space of
19//! the CIELAB family, fitted to modern colour-matching data and cheap: two 3x3
20//! matrices and a cube root each way.  `OkLCh` is its polar form — lightness,
21//! chroma, hue — which is the one to build a ramp in, since those are the three
22//! things a ramp wants to vary independently.
23//!
24//! ## Lightness carries the magnitude
25//!
26//! [`ramp`] varies all three, but the load is on lightness, monotonically.
27//! That is what makes it readable photocopied, readable by the eight percent of
28//! men with a colour deficiency, and readable at all where the picture is
29//! folded so small that a cell is a sample rather than a shape.  Hue and chroma
30//! come along to separate levels that lightness alone leaves adjacent; they are
31//! not carrying the signal on their own.
32//!
33//! ## Gamut
34//!
35//! `OkLCh` can name colours sRGB cannot show, and most of the chroma one would
36//! like at the dark end is outside it.  Clamping the channels afterwards is the
37//! obvious fix and the wrong one: it shifts the hue, so a ramp meant to sweep
38//! evenly gets a flat spot where several entries clamp to the same face of the
39//! cube.  [`in_gamut`] instead keeps the lightness and the hue and takes the
40//! chroma down until the colour fits, by bisection — the standard move, and it
41//! is 256 colours computed once at startup, so its cost is nothing.
42
43/// Entries in the ramp, which is one for every value a sample can take.
44pub const RAMP_LEN: usize = 256;

46/// An sRGB colour, gamma-encoded, as a PNG palette holds it.
47pub type Rgb = [u8; 3];

49/// `OkLCh` to sRGB: lightness in `0..=1`, chroma from 0, hue in degrees.
50///
51/// Answers `None` when the colour is outside what sRGB can show, which is what
52/// [`in_gamut`] bisects on.
53fn oklch_to_srgb(lightness: f64, chroma: f64, hue_degrees: f64) -> Option<Rgb> {
54    let hue = hue_degrees.to_radians();
55    let (a, b) = (chroma * hue.cos(), chroma * hue.sin());
56
57    // Oklab to LMS, cubed back out of the cube root the forward direction takes.
58    let l_ = lightness + 0.396_337_777_4 * a + 0.215_803_757_3 * b;
59    let m_ = lightness - 0.105_561_345_8 * a - 0.063_854_172_8 * b;
60    let s_ = lightness - 0.089_484_177_5 * a - 1.291_485_548_0 * b;
61    let (l, m, s) = (l_ * l_ * l_, m_ * m_ * m_, s_ * s_ * s_);
62
63    // LMS to linear sRGB.
64    let linear = [
65        4.076_741_662_1 * l - 3.307_711_591_3 * m + 0.230_969_929_2 * s,
66        -1.268_438_004_6 * l + 2.609_757_401_1 * m - 0.341_319_396_5 * s,
67        -0.004_196_086_3 * l - 0.703_418_614_7 * m + 1.707_614_701_0 * s,
68    ];
69
70    let mut out = [0u8; 3];
71    for (slot, &value) in out.iter_mut().zip(&linear) {
72        // Outside the cube is outside the gamut.  A hair of tolerance, because
73        // the matrices are a fit and the primaries themselves land a few
74        // billionths past 0 and 1.
75        if !(-1e-6..=1.0 + 1e-6).contains(&value) {
76            return None;
77        }
78        let clamped = value.clamp(0.0, 1.0);
79        // The sRGB transfer function, which is a straight line near black and a
80        // 2.4 power above it.
81        let encoded = if clamped <= 0.003_130_8 {
82            12.92 * clamped
83        } else {
84            1.055 * clamped.powf(1.0 / 2.4) - 0.055
85        };
86        *slot = (encoded * 255.0).round().clamp(0.0, 255.0) as u8;
87    }
88    Some(out)
89}

91/// The colour at this lightness and hue with as much of `chroma` as sRGB can
92/// show.
93///
94/// Bisection on the chroma, keeping lightness and hue exactly: sixteen steps
95/// take the interval below a thousandth, which is far under a step of the
96/// 8-bit channels this ends up in.  A lightness of 0 or 1 is black or white and
97/// admits no chroma at all, which falls out of the same search rather than
98/// needing a case.
99fn in_gamut(lightness: f64, chroma: f64, hue_degrees: f64) -> Rgb {
100    if let Some(rgb) = oklch_to_srgb(lightness, chroma, hue_degrees) {
101        return rgb;
102    }
103    let (mut low, mut high) = (0.0, chroma);
104    // Grey is always inside, so this is the answer if every step fails.
105    let mut best = oklch_to_srgb(lightness, 0.0, hue_degrees)
106        .unwrap_or([if lightness > 0.5 { 255 } else { 0 }; 3]);
107    for _ in 0..16 {
108        let middle = 0.5 * (low + high);
109        match oklch_to_srgb(lightness, middle, hue_degrees) {
110            Some(rgb) => {
111                best = rgb;
112                low = middle;
113            }
114            None => high = middle,
115        }
116    }
117    best
118}

120/// The ramp, from paper to the heaviest ink, as [`RAMP_LEN`] sRGB colours.
121///
122/// `ramp()[0]` is what a sample of 0 draws and `ramp()[255]` what a sample of
123/// 255 draws.  [`crate::image`] counts *up* to paper — a heavier pixel is a
124/// smaller sample, so that the union of two transactions is the smaller of
125/// their samples and a row can be blanked to all 1s — so index 255 is the paper
126/// and index 0 the heaviest ink.  The ramp is written in that direction.
127///
128/// What it does between them:
129///
130/// - **lightness** falls from very nearly white to a dark that still has room
131///   for chroma, and falls monotonically, which is the property the whole ramp
132///   rests on. See the module docs.
133/// - **hue** sweeps 250° of the circle and no more, from a cold blue through
134///   green and gold to a warm red. It is an arc rather than the full circle
135///   because the ends have to stay apart: a ramp that wraps puts its lightest
136///   and darkest entries at the same hue, which is exactly the confusion it is
137///   there to prevent.
138/// - **chroma** rises off the paper and falls again into the dark, since
139///   neither end of the lightness axis has gamut to spare. The peak sits
140///   towards the heavy end, where the levels most need separating.
141pub fn ramp() -> [Rgb; RAMP_LEN] {
142    let mut out = [[0u8; 3]; RAMP_LEN];
143    for (index, slot) in out.iter_mut().enumerate() {
144        // 0 at paper, 1 at the heaviest ink -- the reverse of the index.
145        let t = (RAMP_LEN - 1 - index) as f64 / (RAMP_LEN - 1) as f64;
146
147        let lightness = 0.985 - 0.66 * t;
148        let hue = 250.0 - 250.0 * t.powf(0.85);
149        // Zero at the paper end so the lightest entries are true neutrals, and
150        // eased off again into the dark where there is no gamut for it.
151        let chroma = 0.16 * (t.powf(0.55)) * (1.0 - 0.45 * t * t);
152
153        *slot = in_gamut(lightness, chroma, hue);
154    }
155    out
156}

158/// The spread, in blocks, at which half the chroma is gone.  About 6% of the
159/// 2022 chain: measured colours run from 0 for a coinbase to some 260,000 at
160/// the tip, so this puts the interesting range across the whole saturation
161/// axis rather than crushing it at one end.
162pub const SPREAD_HALF: f64 = 45_000.0;
163
164/// Held constant across every tint, so neither thing being shown is confounded
165/// with lightness and the picture is readable at any size.
166pub const LIGHTNESS: f64 = 0.72;
167
168/// The chroma of a perfectly concentrated colour, before the gamut mapping.
169pub const CHROMA: f64 = 0.15;
170
171/// The hue of block 0, in degrees: cold, and the start of the arc.
172pub const HUE_FROM: f64 = 250.0;
173
174/// How far round the circle the chain runs, stopping short of a full turn so
175/// the ends stay apart.
176pub const HUE_ARC: f64 = 250.0;
177
178/// How concentrated a colour is, in `(0, 1]`: 1 for a coinbase, a half at
179/// [`SPREAD_HALF`], and never quite 0.
180///
181/// This is the mix axis, and the reason one division has a name is that
182/// [`tint`], [`tint_index`] and [`tints`] must agree about it exactly, or a
183/// quantised colour is not a quantisation of anything.
184#[allow(dead_code)]
185pub fn concentration(spread_blocks: f64) -> f64 {
186    1.0 / (1.0 + spread_blocks.max(0.0) / SPREAD_HALF)
187}
188
189/// The spread [`concentration`] answers `wanted` for — its inverse, and the
190/// only reason it exists is so [`tints`] can evaluate [`tint`] at a bin centre
191/// rather than keep a second copy of the colour.
192fn spread_at(wanted: f64) -> f64 {
193    SPREAD_HALF * (1.0 / wanted.clamp(f64::MIN_POSITIVE, 1.0) - 1.0)
194}

196/// The colour of a transaction, from the three numbers `--moments` prints.
197///
198/// Used by `examples/tint.rs` rather than by the driver, which draws pictures
199/// pixel-per-block through [`ramp`] instead; a binary crate cannot see an
200/// example's use of its modules, hence the allow.
201///
202/// A colour in this crate has always been a distribution over block ids, and
203/// this is the obvious picture of one: *when* the coins came from, and *how
204/// concentrated* that was.
205///
206/// - **hue** carries the mean block over a 250 degree arc — early chain cold,
207///   late chain warm.  An arc rather than the circle so that block 0 and the
208///   chain tip do not land on the same hue, which is the one confusion a
209///   cyclic channel invites.
210/// - **chroma** carries the [`concentration`], `1 / (1 + spread /
211///   SPREAD_HALF)`.  A coinbase has a spread of zero and comes out fully
212///   saturated; a colour smeared across a third of chain history comes out
213///   nearly grey.  Mixing literally desaturates, which is the property worth
214///   having: a washed-out pixel is a transaction whose coins have been through
215///   everything.
216/// - **lightness** is held constant, so neither of the two things being shown
217///   is confounded with it and the picture is readable at any size.
218///
219/// Gamut-mapped by [`in_gamut`], so a saturated hue keeps its hue.
220#[allow(dead_code)]
221pub fn tint(mean_block: f64, spread_blocks: f64, chain_blocks: f64) -> Rgb {
222    let where_ = if chain_blocks > 0.0 {
223        (mean_block / chain_blocks).clamp(0.0, 1.0)
224    } else {
225        0.0
226    };
227    in_gamut(
228        LIGHTNESS,
229        CHROMA * concentration(spread_blocks),
230        HUE_FROM - HUE_ARC * where_,
231    )
232}

234/// Hue bins along the chain.  See [`tints`] for why it is 32.
235pub const TINT_HUES: usize = 32;
236
237/// Mixing levels across the [`concentration`] axis, counting *up* with mixing:
238/// slot 0 is a coinbase and slot 7 a colour that has been through everything.
239/// That is the direction `src/bin/tx-view.rs`'s `slot` counts in, and keeping
240/// it means the two pictures cannot be read backwards against each other.
241pub const TINT_MIXES: usize = 8;
242
243/// Entries in the tint table, which has to fit in a byte.
244pub const TINTS_LEN: usize = TINT_HUES * TINT_MIXES;
245
246/// Which entry of [`tints`] stands for this transaction's colour.
247///
248/// The hue axis is cut into [`TINT_HUES`] equal slices of the chain and the mix
249/// axis into [`TINT_MIXES`] equal slices of the [`concentration`] — *not* of
250/// the spread.
251///
252/// Spread is the wrong thing to cut evenly, and not by a little.  The chroma is
253/// proportional to `1 / (1 + spread / 45000)`, so the first eighth of a spread
254/// axis running to 260,000 blocks — 0 to 32,500 — covers chroma from 0.150 down
255/// to 0.087, which is *half of the whole colour axis in one bin*: a worst case
256/// of 0.031 in Oklab, past the 0.02 an eye can find, before the other seven
257/// bins have said anything.  The measurements agree from the other side: over
258/// the first 60,000,000 records of the 2022 chain, 87.70% of transactions have
259/// a spread between 16,384 and 262,144 blocks and 5.78% have none at all, so
260/// even bins of spread would sort almost the whole chain into two of them.  The
261/// concentration is the compressive coordinate already, and it is exactly the
262/// one the chroma is linear in, so equal slices of it are equal slices of what
263/// the eye is shown.
264///
265/// The index is `hue * TINT_MIXES + mix`, so consecutive entries are one hue at
266/// eight degrees of mixing — the order [`tints`] writes them in, and the order
267/// a PNG `PLTE` chunk wants them.
268#[allow(dead_code)]
269pub fn tint_index(mean_block: f64, spread_blocks: f64, chain_blocks: f64) -> u8 {
270    let where_ = if chain_blocks > 0.0 {
271        (mean_block / chain_blocks).clamp(0.0, 1.0)
272    } else {
273        0.0
274    };
275    let hue = ((where_ * TINT_HUES as f64) as usize).min(TINT_HUES - 1);
276    // Rounds down, so a concentration of exactly 1 -- which takes a spread of
277    // exactly zero, nothing near it -- would fall one past the end without the
278    // clamp.  That is 5.78% of the first 60,000,000 records: 3,466,043 of them
279    // in the `--bands 1024 --moments` run cited below.  It is not the coinbase
280    // count, which is 344,032 lines with no inputs at all, 0.5734% of the same
281    // prefix; the rest of the 5.78% is every other transaction that drew all
282    // of its value from a single block.  Nor is it the 6.02% whose spread is
283    // under one block: the 147,561 records between the two come out at a
284    // concentration just short of 1, land in the last slot unaided, and are no
285    // part of what the clamp is for.
286    let step = ((concentration(spread_blocks) * TINT_MIXES as f64) as usize).min(TINT_MIXES - 1);
287    (hue * TINT_MIXES + (TINT_MIXES - 1 - step)) as u8
288}

290/// The [`TINTS_LEN`] colours a [`tint_index`] byte points at.
291///
292/// **This is not [`ramp`] and must not be confused with it.**  `ramp` is keyed
293/// by a *weight* sample on the block axis and is what the `PLTE` of the
294/// (transaction, block) picture holds; this is keyed by a *whole colour*
295/// quantised on two axes, and there is no block axis in it at all.  A tint
296/// index fed through `ramp` would draw a legend that lies about what every
297/// pixel means, which is why this is a second table under a second name rather
298/// than a second use of the first.
299///
300/// Every entry is [`tint`] evaluated at the centre of its bin, so there stays
301/// exactly one definition of the colour: change [`HUE_ARC`] or [`LIGHTNESS`]
302/// and the table moves with it, because the table is not a table — it is the
303/// same arithmetic, run 256 times at startup.  It takes no `chain_blocks`,
304/// because `tint` reads the mean block only as a fraction of the chain: the
305/// height moves which transactions land in which bin, never what a bin looks
306/// like.
307///
308/// ## Why 32 by 8, and what it costs
309///
310/// The two axes compete for one byte, and half a bin of either is worth about
311/// the same thing in Oklab, so the split that minimises the worst case is the
312/// one that balances them.  A hue bin is `250 / H` degrees wide, so half of one
313/// is `125 / H`, and at the full chroma of 0.15 that is a distance of
314/// `0.15 * (125 / H) * pi / 180 = 0.327 / H`.  A mix bin is `1 / M` of the
315/// concentration, so half of one is `0.15 / (2 M) = 0.075 / M` of chroma
316/// directly.  With `H * M = 256` the two are equal at `H = 33.4`, predicting a
317/// worst case of 0.0139 in quadrature; 32 is the power of two beside it.
318///
319/// Measured over the first 60,000,000 records of the 2022 chain
320/// (`--bands 1024 --moments`, 4m23s, 11.3 GB peak), as the mean and the worst
321/// Oklab distance between a transaction's true `tint` and the entry its index
322/// points at:
323///
324/// ```text
325///       H    M   mix axis         mean dE    worst dE
326///     128    2   concentration   0.020889    0.039607
327///      85    3   concentration   0.014990    0.027193
328///      64    4   concentration   0.007640    0.021206
329///      51    5   concentration   0.007976    0.017383
330///      42    6   concentration   0.007443    0.015563
331///      32    8   concentration   0.005328    0.014707
332///      25   10   concentration   0.005081    0.015956
333///      16   16   concentration   0.005891    0.023166
334///      64    4   doublings       0.009728    0.039572
335///      32    8   doublings       0.007344    0.015578
336///      16   16   doublings       0.008407    0.023166
337/// ```
338///
339/// 32 by 8 has the smallest worst case of the lot, and 0.0139 predicted against
340/// 0.014707 measured says the balance argument is the reason rather than a
341/// coincidence.  Only 25 by 10 beats it on the mean, by 0.0002, and pays 0.0012
342/// of worst case for it.  It matters that the *worst* case is what was
343/// minimised: 0.02 in Oklab is about a just-noticeable difference, a picture is
344/// read one transaction at a time, and at 32 by 8 no transaction anywhere moves
345/// by a difference an eye can find.
346///
347/// The same table over 2,000,000 records of `--weighted --moments`, where the
348/// moments are exact rather than binned into 1024 bands — 936,288 of those
349/// 2,000,000 lines carry a different mean or spread from the binned run over
350/// the same records — puts 32 by 8 at 0.004765 mean and **0.014703** worst.
351/// The same worst case to five digits off half a million different numbers is
352/// the point: it is a property of the geometry and not of the data, so it is a
353/// guarantee and not an observation.
354///
355/// ## The mix ladder that was rejected
356///
357/// `src/bin/tx-view.rs` gives one slot per doubling — 1 block, 2 or 3, 4 to 7 —
358/// which is the right ladder for a *count of blocks* starting at one.  Tried
359/// here as bins doubling out from `SPREAD_HALF / 2^(M-4)`, it is worse on both
360/// statistics at every split that fits in a byte: 0.007344 mean and 0.015578
361/// worst at 32 by 8, against 0.005328 and 0.014707.  The reason is in the
362/// paragraph above `tint_index`.  A doubling ladder is compressive in the
363/// spread; the chroma is not a function of the spread but of the
364/// [`concentration`], which is compressive already.  Compressing a compressed
365/// axis crowds the concentrated end — where 5.78% of those 60,000,000 records
366/// sit at a spread of exactly zero — and stretches the mixed end, where there
367/// is nothing.
368///
369/// ## The 32 entries that never come up
370///
371/// Mix slot 7 is every concentration under 1/8, which needs a spread past
372/// 315,000 blocks; the largest in the 2022 chain is about 260,000, so 32 of the
373/// 256 entries stand for a colour that chain never shows.  They are kept.
374/// Fitting the axis to the widest spread actually seen would buy about an
375/// eighth of the mix error and would make the byte a function of the file it
376/// was computed over — the same transaction would get one index in a run over a
377/// prefix and another in a run over the whole chain, which is exactly the
378/// property that makes a byte stream unreadable six months later.
379#[allow(dead_code)]
380pub fn tints() -> [Rgb; TINTS_LEN] {
381    let mut out = [[0u8; 3]; TINTS_LEN];
382    for (index, slot) in out.iter_mut().enumerate() {
383        let (hue, mix) = (index / TINT_MIXES, index % TINT_MIXES);
384        // The centre of the bin, on both axes.  `tint` reads its first argument
385        // only as `mean / chain`, so a chain of 1 makes that argument the
386        // fraction itself and the table independent of the chain height.
387        let where_ = (hue as f64 + 0.5) / TINT_HUES as f64;
388        let step = TINT_MIXES - 1 - mix;
389        let middle = (step as f64 + 0.5) / TINT_MIXES as f64;
390        *slot = tint(where_, spread_at(middle), 1.0);
391    }
392    out
393}

395/// An sRGB colour as Oklab: lightness, then the two opponent axes.
396///
397/// The forward half of what [`oklch_to_srgb`] undoes, and it is here so that
398/// the cost of quantising a colour can be *measured* rather than asserted — a
399/// distance in this space is the only number available that means "how
400/// different do these look".
401#[allow(dead_code)]
402pub fn oklab(rgb: Rgb) -> [f64; 3] {
403    let linear = |c: u8| {
404        let c = c as f64 / 255.0;
405        if c <= 0.040_45 {
406            c / 12.92
407        } else {
408            ((c + 0.055) / 1.055).powf(2.4)
409        }
410    };
411    let (r, g, b) = (linear(rgb[0]), linear(rgb[1]), linear(rgb[2]));
412    let l = (0.412_221_470_8 * r + 0.536_332_536_3 * g + 0.051_445_992_9 * b).cbrt();
413    let m = (0.211_903_498_2 * r + 0.680_699_545_1 * g + 0.107_396_956_6 * b).cbrt();
414    let s = (0.088_302_461_9 * r + 0.281_718_837_6 * g + 0.629_978_700_5 * b).cbrt();
415    [
416        0.210_454_255_3 * l + 0.793_617_785_0 * m - 0.004_072_046_8 * s,
417        1.977_998_495_1 * l - 2.428_592_205_0 * m + 0.450_593_709_9 * s,
418        0.025_904_037_1 * l + 0.782_771_766_2 * m - 0.808_675_766_0 * s,
419    ]
420}

422/// How far apart two colours look: the Euclidean distance in [`oklab`], which
423/// is the whole reason for being in that space.
424///
425/// About 0.02 is a just-noticeable difference for a pair of large patches, and
426/// rather less for two colours touching.  The quantisation [`tints`] does is
427/// held under it, which is what the numbers on that page are.
428#[allow(dead_code)]
429pub fn difference(one: Rgb, other: Rgb) -> f64 {
430    let (a, b) = (oklab(one), oklab(other));
431    ((a[0] - b[0]).powi(2) + (a[1] - b[1]).powi(2) + (a[2] - b[2]).powi(2)).sqrt()
432}

444    /// Relative luminance, which is what "lighter" means to an eye and what the
445    /// ramp has to be monotone in.  Not the mean of the channels: green carries
446    /// most of the luminance and blue almost none.
447    fn luminance(rgb: Rgb) -> f64 {
448        let linear = |c: u8| {
449            let c = c as f64 / 255.0;
450            if c <= 0.040_45 {
451                c / 12.92
452            } else {
453                ((c + 0.055) / 1.055).powf(2.4)
454            }
455        };
456        0.2126 * linear(rgb[0]) + 0.7152 * linear(rgb[1]) + 0.0722 * linear(rgb[2])
457    }
458
459    /// The property the ramp rests on: heavier is darker, every single step.
460    /// Without it the picture cannot be read photocopied, by a colour-deficient
461    /// eye, or where it is folded small enough that a cell is one sample.
462    #[test]
463    fn the_ramp_darkens_monotonically_from_paper_to_ink() {
464        let ramp = ramp();
465        for index in (1..RAMP_LEN).rev() {
466            let lighter = luminance(ramp[index]);
467            let darker = luminance(ramp[index - 1]);
468            assert!(
469                darker <= lighter + 1e-9,
470                "entry {} is lighter than {}: {:?} against {:?}",
471                index - 1,
472                index,
473                ramp[index - 1],
474                ramp[index]

624    /// The one property the whole byte rests on: the entry a transaction's
625    /// index points at is a colour it can be mistaken for.
626    ///
627    /// The bound is the worst case `tints` claims -- 0.014707 measured over
628    /// 60,000,000 real records, 0.014703 over 2,000,000 exact ones -- and this
629    /// sweeps a grid finer than the bins to catch the corners the chain never
630    /// happened to visit.  A change to `TINT_HUES`, `TINT_MIXES`, `HUE_ARC` or
631    /// `CHROMA` that spent more than a just-noticeable difference would fail
632    /// here rather than in six months in a picture.
633    #[test]
634    fn a_tint_index_points_at_a_colour_the_tint_can_be_mistaken_for() {
635        let chain = 762_261.0;
636        let table = tints();
637        let mut worst = 0.0f64;
638        let (mut where_worst, mut spread_worst) = (0.0, 0.0);
639        for step in 0..=400 {
640            let mean = chain * step as f64 / 400.0;
641            // Zero, then out past the widest spread the 2022 chain shows.
642            for spread in [0.0, 1.0, 500.0, 5_000.0, 22_500.0, 45_000.0, 90_000.0,
643                           150_000.0, 260_000.0, 315_000.0, 1_000_000.0] {
644                let index = tint_index(mean, spread, chain) as usize;
645                let d = difference(tint(mean, spread, chain), table[index]);
646                if d > worst {
647                    worst = d;
648                    where_worst = mean;
649                    spread_worst = spread;
650                }
651            }
652        }
653        assert!(
654            worst <= 0.0148,
655            "quantising cost {} at mean {} spread {}, past the 0.014707 the docs claim",
656            worst,
657            where_worst,
658            spread_worst
659        );
660    }

662    /// Each entry is the tint of its own bin's centre, so the centre has to map
663    /// back to the entry.  Without this the table and the index are two
664    /// descriptions of two different partitions and every byte is off by one
665    /// somewhere.
666    #[test]
667    fn every_entry_is_the_colour_of_a_bin_that_indexes_back_to_it() {
668        for index in 0..TINTS_LEN {
669            let (hue, mix) = (index / TINT_MIXES, index % TINT_MIXES);
670            let where_ = (hue as f64 + 0.5) / TINT_HUES as f64;
671            let middle = ((TINT_MIXES - 1 - mix) as f64 + 0.5) / TINT_MIXES as f64;
672            assert_eq!(
673                tint_index(where_, spread_at(middle), 1.0) as usize,
674                index,
675                "entry {} is drawn for a bin that indexes elsewhere",
676                index
677            );
678        }
679    }

681    /// [`concentration`] and [`spread_at`] are the mix axis and its inverse, and
682    /// `tints` evaluates `tint` through the pair.  A drift between them would
683    /// move every entry off its bin centre without moving the index, which the
684    /// test above would catch only once it had grown past half a bin.
685    #[test]
686    fn the_mix_axis_and_its_inverse_are_the_same_axis() {
687        for spread in [0.0, 1.0, 45_000.0, 260_000.0, 1e9] {
688            let there_and_back = spread_at(concentration(spread));
689            assert!(
690                (there_and_back - spread).abs() <= 1e-6 * spread.max(1.0),
691                "spread {} came back as {}",
692                spread,
693                there_and_back
694            );
695        }
696        assert_eq!(concentration(0.0), 1.0, "a coinbase is perfectly concentrated");
697        assert!(concentration(SPREAD_HALF) - 0.5 < 1e-12, "half the chroma at SPREAD_HALF");
698        // Monotone, which is what makes an ordered cut of it an ordered cut of
699        // the mixing.
700        let mut previous = f64::INFINITY;
701        for k in 0..1000 {
702            let c = concentration(k as f64 * 500.0);
703            assert!(c < previous, "concentration must fall with spread");
704            previous = c;
705        }
706    }

708    /// The byte's two axes read the way the tint's two channels do: slot 0 of
709    /// the mix axis is a coinbase, the last slot is a colour that has been
710    /// everywhere, and the hue slot climbs with the chain.
711    #[test]
712    fn the_index_counts_the_way_the_colour_does() {
713        let chain = 762_261.0;
714        let mix = |mean, spread| tint_index(mean, spread, chain) as usize % TINT_MIXES;
715        let hue = |mean, spread| tint_index(mean, spread, chain) as usize / TINT_MIXES;
716
717        assert_eq!(mix(400_000.0, 0.0), 0, "a coinbase is the unmixed end");
718        assert!(mix(400_000.0, 45_000.0) > 0, "a spread of SPREAD_HALF is mixed");
719        assert!(
720            mix(400_000.0, 260_000.0) > mix(400_000.0, 45_000.0),
721            "more spread is more mixed"
722        );
723        assert_eq!(
724            mix(400_000.0, 1e12),
725            TINT_MIXES - 1,
726            "and it saturates rather than overflowing"
727        );
728
729        assert_eq!(hue(0.0, 0.0), 0, "block 0 is the cold end");
730        assert_eq!(hue(chain, 0.0), TINT_HUES - 1, "the tip is the warm end");
731        assert_eq!(
732            hue(chain * 10.0, 0.0),
733            TINT_HUES - 1,
734            "past the tip clamps rather than wrapping"
735        );
736        let mut previous = 0;
737        for k in 0..=100 {
738            let h = hue(chain * k as f64 / 100.0, 0.0);
739            assert!(h >= previous, "the hue slot must not go backwards along the chain");
740            previous = h;
741        }

744    /// The mistake this table exists to prevent.
745    ///
746    /// [`ramp`] is keyed by a weight on the block axis and carries its signal in
747    /// lightness; [`tints`] is keyed by a whole colour and is at constant
748    /// lightness by construction.  Running an index through `ramp` would draw a
749    /// legend that lies, and the two are different sizes of different things --
750    /// so they must not be interchangeable, and the numbers say so.
751    #[test]
752    fn the_tint_table_is_not_the_weight_ramp() {
753        let (table, ramp) = (tints(), ramp());
754        assert_ne!(&table[..], &ramp[..], "two tables, two meanings");
755
756        let lightness = |c: Rgb| oklab(c)[0];
757        let spanned = |t: &[Rgb]| {
758            let (mut lo, mut hi) = (f64::INFINITY, f64::NEG_INFINITY);
759            for &c in t {
760                lo = lo.min(lightness(c));
761                hi = hi.max(lightness(c));
762            }
763            hi - lo
764        };
765        // The ramp spends nearly the whole lightness axis; the tints spend the
766        // gamut mapping's leftovers and nothing else.
767        assert!(
768            spanned(&ramp) > 0.5,
769            "the ramp carries its signal in lightness: spanned {}",
770            spanned(&ramp)
771        );
772        assert!(
773            spanned(&table) < 0.1,
774            "the tints are at one lightness, so hue and chroma carry both axes: spanned {}",
775            spanned(&table)
776        );
777    }

793    fn the_two_directions_of_the_space_agree() {
