1//! # Knowing when a colour is dead rather than guessing
2//!
3//! The driver holds a colour for every transaction with an unspent output,
4//! because any of those outputs might still be spent.  That is the only answer
5//! available to a program reading a stream, and over the 2022 chain it means
6//! holding 92,735,490 colours at the end and 92,753,575 at peak — the
7//! multiplier in the memory wall, since the store is that count times the size
8//! of a colour.
9//!
10//! It is also the wrong question.  What the fold needs is not *could this be
11//! spent* but *will anything later in this file read it again*, and the two
12//! differ enormously: 51,815,075 of the 136 million outputs unspent at the end
13//! of the chain are `OP_RETURN`, provably unspendable and untouched by any of
14//! the file's 2,044,897,328 inputs, and millions more are simply never spent
15//! within it.  Every colour held for one of those is held for nothing.
16//!
17//! A file has a future, so the right question can be answered — by reading the
18//! records once before colouring them.  `examples/lastspend.rs` does that and
19//! writes one `u32` a transaction: the record that spends it last, or
20//! [`NEVER`].  This reads that back.  The peak drops 4.79x, which is what makes
21//! `--bands 1024` comfortable and `--bands 4096` possible at all.
22//!
23//! ## It must not change a single byte of output
24//!
25//! This decides *when a colour is freed*, never what it contains, so a run with
26//! the oracle and one without have to agree exactly.  Two things protect that.
27//!
28//! The oracle is refused unless it covers at least the records the run will
29//! read.  Longer is safe -- an oracle for the whole file, used on a prefix,
30//! only says a colour is read later than the prefix reaches, so the fold holds
31//! it longer than it had to -- which is what lets one oracle serve every prefix
32//! of the same file.  Shorter is not: it answers `NEVER` where the truth is
33//! "later", and the driver would then either fail its lookup loudly or, worse,
34//! be handed a displaced entry quietly.  And [`Oracle::last_reads`] is deliberately
35//! conservative in the one place it could be wrong: it frees on the *lowest*
36//! input index that names a transaction, because the fold walks its inputs from
37//! the last to the first, so the lowest index is the one it reaches last.
38//! Freeing on the highest would drop the colour while inputs below it still
39//! wanted it.
40
41use std::io::{self, Read};
42
43/// A transaction nothing in the file ever spends.
44pub const NEVER: u32 = u32::MAX;
45
46const MAGIC: &[u8; 8] = b"LASTSPN1";
47
48/// For each transaction, the record that spends it last.
49pub struct Oracle {
50    last: Vec<u32>,
51    records: u64,
52}
53
54impl Oracle {
55    /// Read an oracle, and refuse one that was not written for these records.
56    pub fn load(path: &str, records_expected: Option<u64>) -> io::Result<Oracle> {
57        let mut file = io::BufReader::with_capacity(1 << 22, std::fs::File::open(path)?);
58        let mut head = [0u8; 24];
59        file.read_exact(&mut head)?;
60        if &head[..8] != MAGIC {
61            return Err(io::Error::new(
62                io::ErrorKind::InvalidData,
63                "not a release oracle".to_string(),
64            ));
65        }
66        let records = u64::from_le_bytes(head[8..16].try_into().unwrap());
67        let count = u64::from_le_bytes(head[16..24].try_into().unwrap()) as usize;
68
69        // Longer than the run is safe and shorter is not, so this is an
70        // inequality rather than a match.  An oracle written for the whole file
71        // says a colour's last read is at some record the run may never reach,
72        // which only makes the fold hold it longer than it needed to -- the
73        // conservative direction, and exactly what makes one oracle usable for
74        // every prefix of the same file.  One written for fewer records answers
75        // `NEVER` where the truth is "later", and frees colours the run still
76        // wants.
77        if let Some(expected) = records_expected {
78            if records < expected {
79                return Err(io::Error::new(
80                    io::ErrorKind::InvalidData,
81                    format!(
82                        "written for {} records and this run reads {}; \
83                         it would free colours that are still wanted",
84                        records, expected
85                    ),
86                ));
87            }
88        }
89
90        let mut last = vec![0u32; count];
91        // Read straight into the vector's bytes: 3 GB through a `u32`-at-a-time
92        // loop is a minute of nothing.
93        let bytes = unsafe {
94            std::slice::from_raw_parts_mut(last.as_mut_ptr() as *mut u8, count * 4)
95        };
96        file.read_exact(bytes)?;
97        if cfg!(target_endian = "big") {
98            for v in &mut last {
99                *v = v.swap_bytes();
100            }
101        }
102        Ok(Oracle { last, records })
103    }
104
105    pub fn records(&self) -> u64 {
106        self.records
107    }
108
109    /// The record that spends `tx` last, or [`NEVER`].
110    #[inline]
111    pub fn last(&self, tx: usize) -> u32 {
112        self.last.get(tx).copied().unwrap_or(NEVER)
113    }
114
115    /// Whether a colour for `tx` is worth keeping once its record is emitted.
116    #[inline]
117    pub fn ever_read(&self, tx: usize) -> bool {
118        self.last(tx) != NEVER
119    }
120
121    /// Which of this record's inputs is the last read of the transaction it
122    /// spends, written into `out`.
123    ///
124    /// An input is the last read when this record is the last that spends its
125    /// source *and* no earlier input of this same record spends it too.  The
126    /// second half matters: the fold walks inputs from the last to the first,
127    /// so the earliest index is the one it reaches last, and that is the one
128    /// that may take the colour.  Marking the later index instead would free
129    /// the colour while the earlier input still wanted it, and the lookup that
130    /// followed would fail.
131    pub fn last_reads(&self, record: u64, inputs: &[crate::sexp::Input], out: &mut Vec<bool>) {
132        out.clear();
133        out.resize(inputs.len(), false);
134        let index = record as u32;
135        for (k, input) in inputs.iter().enumerate() {
136            if self.last(input.prev_tx_id) != index {
137                continue;
138            }
139            // The commonest record has one or two inputs, so the scan below is
140            // over nothing; a consolidation with thousands is rare enough that
141            // its quadratic worst case never shows against the merge it feeds.
142            let earlier = inputs[..k].iter().any(|e| e.prev_tx_id == input.prev_tx_id);
143            out[k] = !earlier;
144        }
145    }
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151    use crate::sexp::Input;
152
153    fn oracle(last: &[u32]) -> Oracle {
154        Oracle {
155            last: last.to_vec(),
156            records: 0,
157        }
158    }
159
160    fn input(prev: usize) -> Input {
161        Input {
162            prev_tx_id: prev,
163            amount: 1,
164        }
165    }
166
167    /// The plain case: an input whose source nothing later reads is the last
168    /// read; one whose source is read again is not.
169    #[test]
170    fn the_last_record_to_spend_a_transaction_may_take_its_colour() {
171        // tx 0 is spent last by record 5, tx 1 by record 9.
172        let o = oracle(&[5, 9]);
173        let mut out = Vec::new();
174        o.last_reads(5, &[input(0), input(1)], &mut out);
175        assert_eq!(out, vec![true, false], "record 5 is the last read of tx 0 only");
176        o.last_reads(9, &[input(1)], &mut out);
177        assert_eq!(out, vec![true]);
178    }
179
180    /// The case the ordering comment is about.  A record that spends the same
181    /// transaction twice must free it on the input the fold reaches *last*,
182    /// which is the lowest index, because the fold walks inputs in reverse.
183    #[test]
184    fn a_record_spending_one_transaction_twice_frees_it_on_the_lowest_input() {
185        let o = oracle(&[7]);
186        let mut out = Vec::new();
187        o.last_reads(7, &[input(0), input(0), input(0)], &mut out);
188        assert_eq!(
189            out,
190            vec![true, false, false],
191            "only the input the reverse fold reaches last may take the colour"
192        );
193    }
194
195    /// A transaction nothing spends is dead the moment its own line is written.
196    #[test]
197    fn a_transaction_nothing_spends_is_never_read() {
198        let o = oracle(&[NEVER, 3]);
199        assert!(!o.ever_read(0));
200        assert!(o.ever_read(1));
201        // And one past the end of the table -- a prefix run whose oracle was
202        // written for fewer transactions -- is treated as never read rather
203        // than panicking.
204        assert!(!o.ever_read(99));
205    }
206
207    /// An input whose source is spent again later is not a last read even when
208    /// this record spends it more than once.
209    #[test]
210    fn a_source_read_again_later_is_never_taken_here() {
211        let o = oracle(&[12]);
212        let mut out = Vec::new();
213        o.last_reads(7, &[input(0), input(0)], &mut out);
214        assert_eq!(out, vec![false, false], "record 12 still wants tx 0");
215    }
