1//! A streaming reader for the transaction records the driver eats.
2//!
3//! The Scheme reads them with plain `(read)`, so the format is whitespace
4//! agnostic and a record may be split across lines; this reader is too, rather
5//! than assuming the one-record-per-line the sample files happen to use.
6//!
7//! Records look like
8//!
9//! ```text
10//! ((timestamp block-id tx-id _ _ _ _) (input ...) (output ...))
11//! input  = (addr-id amount prev-tx-id vout)
12//! output = (addr-id amount _)
13//! ```
14//!
15//! Only four things are ever used: the header's `block-id` and `tx-id`, each
16//! input's `prev-tx-id`, and how many outputs there are.  Fields are picked out
17//! by position and the rest of each list is skipped, so a record with extra
18//! trailing fields still reads — unlike the Scheme's fixed-arity `match/first`.
19//! Nothing is allocated per record.
20//!
21//! The buffer is ours rather than a `BufReader`, because at 149 GB the hot path
22//! is "look at one byte" and it needs to inline down to a bounds check.
23//!
24//! ## Skipped means skipped
25//!
26//! Those four values are a small minority of what a record spells out: the
27//! header gives 2 of its 7 fields to the driver, an input 1 of its 4, an output
28//! none of its 3 — an output is read only for existing.  So the wanted
29//! positions are named in a bitmask ([`HEADER_FIELDS`] and friends) and
30//! everything else goes through [`Reader::skip_int`], which walks the digits
31//! without converting them.  Turning a number into a `usize` costs a multiply
32//! and an add per digit, and there is no point paying it for a number that is
33//! dropped on the next line.
34//!
35//! ## Digits are scanned wide
36//!
37//! Finding where a run of digits ends is the one thing this reader does to
38//! nearly every byte, and it is a poor fit for a loop and a good fit for a
39//! vector compare.  [`simd::digit_run`] answers it sixteen bytes at a time, and
40//! [`simd::eight_digits`] folds eight digits into a value in three
41//! multiply-shift steps, so [`Reader::read_int`] checks for overflow once per
42//! eight digits rather than once per digit.  [`Reader::digit_run`] is the seam
43//! between those kernels and this buffer: it is what handles a run of digits
44//! that straddles a refill, so neither kernel has to know the buffer exists.
45
46use crate::simd;
47use std::io::{self, Read};
48
49const BUF_SIZE: usize = 1 << 20;
50
51/// Positions the record header is read for: the block id at 1, the tx id at 2.
52const HEADER_FIELDS: u32 = (1 << 1) | (1 << 2);
53
54/// Position an input is read for: the previous tx id at 2.
55const INPUT_FIELDS: u32 = 1 << 2;
56
57/// An output is read for nothing at all — only for there being one.
58const OUTPUT_FIELDS: u32 = 0;
59
60pub struct Record {
61    pub block_id: usize,
62    pub tx_id: usize,
63    pub outputs: usize,
64}
65
66pub struct Reader<R: Read> {
67    inner: R,
68    buf: Box<[u8]>,
69    pos: usize,
70    end: usize,
71    /// Bytes consumed strictly before `buf[pos]`, for error messages.
72    consumed: usize,
73}
74
75impl<R: Read> Reader<R> {
76    pub fn new(inner: R) -> Self {
77        Reader {
78            inner,
79            buf: vec![0u8; BUF_SIZE].into_boxed_slice(),
80            pos: 0,
81            end: 0,
82            consumed: 0,
83        }
84    }
85
86    #[inline]
87    fn peek(&mut self) -> io::Result<Option<u8>> {
88        if self.pos < self.end {
89            return Ok(Some(self.buf[self.pos]));
90        }
91        self.refill()
92    }
93
94    #[cold]
95    fn refill(&mut self) -> io::Result<Option<u8>> {
96        self.consumed += self.end;
97        self.pos = 0;
98        self.end = 0;
99        loop {
100            match self.inner.read(&mut self.buf) {
101                Ok(0) => return Ok(None),
102                Ok(n) => {
103                    self.end = n;
104                    return Ok(Some(self.buf[0]));
105                }
106                Err(e) if e.kind() == io::ErrorKind::Interrupted => continue,
107                Err(e) => return Err(e),
108            }
109        }
110    }
111
112    fn offset(&self) -> usize {
113        self.consumed + self.pos
114    }
115
116    fn err<T>(&self, msg: String) -> io::Result<T> {
117        Err(io::Error::new(
118            io::ErrorKind::InvalidData,
119            format!("{} at byte offset {}", msg, self.offset()),
120        ))
121    }
122
123    /// The next non-whitespace byte, left unconsumed.
124    #[inline]
125    fn skip_ws(&mut self) -> io::Result<Option<u8>> {
126        loop {
127            match self.peek()? {
128                Some(b) if b.is_ascii_whitespace() => self.pos += 1,
129                other => return Ok(other),
130            }
131        }
132    }
133
134    fn expect(&mut self, want: u8) -> io::Result<()> {
135        match self.skip_ws()? {
136            Some(b) if b == want => {
137                self.pos += 1;
138                Ok(())
139            }
140            Some(b) => self.err(format!(
141                "expected {:?}, found {:?}",
142                want as char, b as char
143            )),
144            None => self.err(format!("expected {:?}, found end of input", want as char)),
145        }
146    }
147
148    /// Walk the run of digits at the cursor, handing each contiguous slice of it
149    /// to `absorb`, and answer how many digits there were altogether.
150    ///
151    /// `absorb` is called more than once when the run straddles a refill.  The
152    /// buffer is a window onto the stream and [`simd::digit_run`] can only speak
153    /// for what is inside it, so a run reaching the end of the window is not
154    /// known to have ended: refill and scan again.  Keeping that here is what
155    /// lets the vector kernel work in whole 16-byte blocks without any caller
156    /// having to think about the buffer boundary.
157    #[inline]
158    fn digit_run(&mut self, mut absorb: impl FnMut(&[u8])) -> io::Result<usize> {
159        let mut digits = 0;
160        loop {
161            let run = simd::digit_run(&self.buf[self.pos..self.end]);
162            if run > 0 {
163                absorb(&self.buf[self.pos..self.pos + run]);
164                self.pos += run;
165                digits += run;
166            }
167            if self.pos < self.end {
168                // Stopped on a byte that is not a digit: the run is over.
169                return Ok(digits);
170            }
171            // The window is spent.  `peek` refills it, or says the stream has
172            // ended and the run with it.
173            if self.peek()?.is_none() {
174                return Ok(digits);
175            }
176        }
177    }
178
179    /// Read one non-negative integer.  Nothing in these records is ever
180    /// negative — the fields are timestamps, ids, counts and satoshi amounts —
181    /// so a leading `-` is a malformed record rather than a number to accept.
182    ///
183    /// The digits are folded eight at a time, which is also how often the
184    /// overflow check runs; the remainder is finished off one at a time.
185    fn read_int(&mut self) -> io::Result<usize> {
186        let first = match self.skip_ws()? {
187            Some(b) => b,
188            None => return self.err("expected an integer, found end of input".into()),
189        };
190        if first == b'-' {
191            return self.err("negative integer, but every field here is unsigned".into());
192        }
193
194        let mut value: usize = 0;
195        let mut overflowed = false;
196        let digits = self.digit_run(|run| {
197            if overflowed {
198                return;
199            }
200            let mut chunks = run.chunks_exact(8);
201            for chunk in &mut chunks {
202                let eight = simd::eight_digits(u64::from_le_bytes(chunk.try_into().unwrap()));
203                match value
204                    .checked_mul(100_000_000)
205                    .and_then(|v| v.checked_add(eight as usize))
206                {
207                    Some(v) => value = v,
208                    None => {
209                        overflowed = true;
210                        return;
211                    }
212                }
213            }
214            for &b in chunks.remainder() {
215                match value
216                    .checked_mul(10)
217                    .and_then(|v| v.checked_add((b - b'0') as usize))
218                {
219                    Some(v) => value = v,
220                    None => {
221                        overflowed = true;
222                        return;
223                    }
224                }
225            }
226        })?;
227
228        if overflowed {
229            return self.err("integer does not fit in a usize".into());
230        }
231        if digits == 0 {
232            return self.err(format!("expected an integer, found {:?}", first as char));
233        }
234
235        Ok(value)
236    }
237
238    /// Consume one integer without working out what it is.
239    ///
240    /// The cheapest way to read a number nobody wants is not to convert it, and
241    /// most of the numbers in a record are numbers nobody wants — see the module
242    /// docs.  Every malformed-input check that does not need the value is still
243    /// made: a leading `-` is refused, and so is an empty field.
244    ///
245    /// The one check given up is the overflow test, which cannot be made without
246    /// doing the arithmetic this exists to avoid.  A field too wide for a
247    /// `usize` therefore passes here where [`Reader::read_int`] would reject it.
248    /// Nothing downstream ever sees the value, so that is a rejection the run
249    /// did not depend on — and satoshi amounts, the widest fields in the format,
250    /// have four digits of headroom against a 64-bit `usize` anyway.
251    fn skip_int(&mut self) -> io::Result<()> {
252        let first = match self.skip_ws()? {
253            Some(b) => b,
254            None => return self.err("expected an integer, found end of input".into()),
255        };
256        if first == b'-' {
257            return self.err("negative integer, but every field here is unsigned".into());
258        }
259        if self.digit_run(|_| {})? == 0 {
260            return self.err(format!("expected an integer, found {:?}", first as char));
261        }
262        Ok(())
263    }
264
265    /// Read `( int int ... )`, handing the elements whose position is set in
266    /// `wanted` to `visit`, skipping the rest, and answering how many elements
267    /// there were in total.
268    ///
269    /// `wanted` is a bitmask over positions: position `n` is wanted when bit `n`
270    /// is set.  Positions from 32 up are never wanted, which is the existing
271    /// rule that fields beyond the named ones are ignored, arriving at the same
272    /// place by a cheaper route.
273    fn read_flat_list(
274        &mut self,
275        wanted: u32,
276        mut visit: impl FnMut(usize, usize),
277    ) -> io::Result<usize> {
278        self.expect(b'(')?;
279        let mut n = 0;
280        loop {
281            match self.skip_ws()? {
282                Some(b')') => {
283                    self.pos += 1;
284                    return Ok(n);
285                }
286                Some(_) => {
287                    if n < 32 && wanted & (1 << n) != 0 {
288                        let v = self.read_int()?;
289                        visit(n, v);
290                    } else {
291                        self.skip_int()?;
292                    }
293                    n += 1;
294                }
295                None => return self.err("unterminated list".into()),
296            }
297        }
298    }
299
300    /// The next record, or `None` at end of input.  `inputs` is cleared and
301    /// refilled with the previous-transaction id of each input; pass the same
302    /// vector every time to keep this allocation-free.
303    pub fn next_record(&mut self, inputs: &mut Vec<usize>) -> io::Result<Option<Record>> {
304        inputs.clear();
305
306        if self.skip_ws()?.is_none() {
307            return Ok(None);
308        }
309        self.expect(b'(')?;
310
311        let (mut block_id, mut tx_id) = (None, None);
312        let fields = self.read_flat_list(HEADER_FIELDS, |i, v| match i {
313            1 => block_id = Some(v),
314            2 => tx_id = Some(v),
315            _ => {}
316        })?;
317        let (block_id, tx_id) = match (block_id, tx_id) {
318            (Some(b), Some(t)) => (b, t),
319            _ => {
320                return self.err(format!(
321                    "record header has {} fields, expected the block id at position 1 and the tx id at position 2",
322                    fields
323                ))
324            }
325        };
326
327        self.expect(b'(')?;
328        loop {
329            match self.skip_ws()? {
330                Some(b')') => {
331                    self.pos += 1;
332                    break;
333                }
334                Some(b'(') => {
335                    let mut prev_tx_id = None;
336                    let fields = self.read_flat_list(INPUT_FIELDS, |i, v| {
337                        if i == 2 {
338                            prev_tx_id = Some(v)
339                        }
340                    })?;
341                    match prev_tx_id {
342                        Some(p) => inputs.push(p),
343                        None => {
344                            return self.err(format!(
345                                "input has {} fields, expected the previous tx id at position 2",
346                                fields
347                            ))
348                        }
349                    }
350                }
351                _ => return self.err("expected an input".into()),
352            }
353        }
354
355        self.expect(b'(')?;
356        let mut outputs = 0;
357        loop {
358            match self.skip_ws()? {
359                Some(b')') => {
360                    self.pos += 1;
361                    break;
362                }
363                Some(b'(') => {
364                    self.read_flat_list(OUTPUT_FIELDS, |_, _| {})?;
365                    outputs += 1;
366                }
367                _ => return self.err("expected an output".into()),
368            }
369        }
370
371        self.expect(b')')?;
372        Ok(Some(Record {
373            block_id,
374            tx_id,
375            outputs,
376        }))
377    }
378}

384    /// What a record boils down to, for comparing two reads of the same bytes.
385    type Parsed = (usize, usize, Vec<usize>, usize);

387    /// A `Read` that hands back one byte per call, so the reader refills its
388    /// window at every single position in the stream.
389    ///
390    /// This is the point of the whole test module.  [`Reader::digit_run`] has to
391    /// stitch a run of digits back together when it straddles a refill, and with
392    /// a 1 MiB buffer that happens about once per million bytes — far too rare
393    /// for a corpus to be trusted to cover it, and the corpus cannot say *which*
394    /// digit the split fell on.  Reading a byte at a time makes every number in
395    /// the input take the straddling path, at every offset within itself.
396    struct Dribble<'a>(&'a [u8]);
397
398    impl Read for Dribble<'_> {
399        fn read(&mut self, out: &mut [u8]) -> io::Result<usize> {
400            if self.0.is_empty() || out.is_empty() {
401                return Ok(0);
402            }
403            out[0] = self.0[0];
404            self.0 = &self.0[1..];
405            Ok(1)
406        }
407    }
408
409    fn read_all(inner: impl Read) -> io::Result<Vec<Parsed>> {
410        let mut reader = Reader::new(inner);
411        let mut inputs = Vec::new();
412        let mut records = Vec::new();
413        while let Some(r) = reader.next_record(&mut inputs)? {
414            records.push((r.block_id, r.tx_id, inputs.clone(), r.outputs));
415        }
416        Ok(records)
417    }
418
419    /// Read `text` both in one gulp and a byte at a time, insist the two agree,
420    /// and answer what they said.
421    fn parse(text: &str) -> io::Result<Vec<Parsed>> {
422        let whole = read_all(text.as_bytes());
423        let dribbled = read_all(Dribble(text.as_bytes()));
424        match (&whole, &dribbled) {
425            (Ok(a), Ok(b)) => assert_eq!(a, b, "buffered and dribbled reads disagree"),
426            (Err(a), Err(b)) => assert_eq!(
427                a.to_string(),
428                b.to_string(),
429                "buffered and dribbled reads disagree on the error"
430            ),
431            _ => panic!("one read succeeded and the other did not: {:?}", whole),
432        }
433        whole
434    }

581    /// A long run of records with fields of mixed width, read both ways.  The
582    /// widths are chosen so digit runs land on every offset relative to the
583    /// eight-digit stride, and the whole thing is long enough that the buffered
584    /// read does several thousand records between refills while the dribbled one
585    /// refills constantly.
586    #[test]
587    fn a_long_mixed_width_stream_reads_the_same_both_ways() {
588        let mut text = String::new();
589        let mut expected = Vec::new();
590        for i in 0..400usize {
591            let block = 10usize.pow((i % 19) as u32 / 2);
592            let tx = 10usize.pow((i % 18) as u32) + i;
593            let prev = 10usize.pow((i % 17) as u32) + i;
594            let amount = "7".repeat(i % 19 + 1);
595            text.push_str(&format!(
596                "(({} {} {} {} 0 {} 0) ((3 {} {} 0) (4 {} {} 1)) ((1 {} 2) (2 {} 3)))\n",
597                i, block, tx, amount, amount, amount, prev, amount, prev, amount, amount
598            ));
599            expected.push((block, tx, vec![prev, prev], 2));
600        }
601        assert_eq!(parse(&text).unwrap(), expected);
602    }
