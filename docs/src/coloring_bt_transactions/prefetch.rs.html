1//! # Reading the records on a thread of its own
2//!
3//! [`sexp::Reader`](crate::sexp) is the other half of what a run does to every
4//! byte, and unlike the fold it depends on nothing the fold produces: a record
5//! can be parsed at any time after the bytes arrive and before it is coloured.
6//! So it does not have to happen between two colours, and this is where it goes
7//! instead.
8//!
9//! What that is worth depends entirely on how big the colours are, which is why
10//! it is a switch rather than a change:
11//!
12//! - Colours of a few blocks — `--example records -- --window 0`, whose
13//!   ancestry stays inside a block — and the fold is almost nothing, so parsing
14//!   is nearly the whole run: 1,000,001 records of 95 MB in 0.58s, and the fold
15//!   behind them is noise.
16//! - Colours of a few thousand blocks — `--window 4000`, which is what a real
17//!   chain grows — and the merge dwarfs it: at 150,000 records the parse is
18//!   around 0.1s of a 3.4s fold, so hiding it is worth a few percent and no
19//!   more.
20//!
21//! Both are real workloads and neither number is the interesting one on its
22//! own.  What matters is that the cost is *off the critical path* either way,
23//! and that hiding it costs one thread and one bounded channel.
24//!
25//! ## Batches, and one arena for their inputs
26//!
27//! A record is three `usize` and a slice of inputs, so sending them one at a
28//! time would spend more on the channel than on the record.  They go in batches
29//! of [`BATCH`], and a batch is three flat vectors rather than a vector of
30//! records each owning a vector of inputs: one allocation a field instead of
31//! one a record, and the whole thing is refilled rather than rebuilt once the
32//! fold thread hands it back.
33//!
34//! ## The SIMD is exactly where it was
35//!
36//! [`crate::simd::digit_run`] and [`crate::simd::eight_digits`] run inside
37//! `next_record`, and `next_record` is called here the way it was called from
38//! the loop — same reader, same buffer, same 16-byte stride.  A batch boundary
39//! falls between records and never inside one, so no run of digits is ever cut
40//! in half by it; the vector kernels cannot tell which thread they are on.
41
42use std::io;
43use std::sync::mpsc::{sync_channel, Receiver, SyncSender};
44use std::thread;
45
46use crate::sexp;

48/// Records per batch.
49///
50/// Large enough that the channel costs nothing per record, small enough that
51/// the fold thread is not left waiting on a batch at the start of a run and
52/// that the pipeline holds a bounded and modest number of records.
53const BATCH: usize = 1024;
54
55/// How many batches may be waiting.  Two is enough to keep the reader working
56/// while the fold thread consumes; more only buys latency nobody is watching.
57const QUEUE: usize = 4;
58
59/// A run of records, with all of their inputs in one arena.
60pub struct Batch {
61    records: Vec<sexp::Record>,
62    /// Every record's inputs, back to back.
63    inputs: Vec<sexp::Input>,
64    /// `ends[i]` is where record `i`'s inputs stop in `inputs`; record `i`'s
65    /// inputs start where record `i - 1`'s stopped, and the first at 0.
66    ends: Vec<usize>,
67}
68
69impl Batch {
70    fn new() -> Self {
71        Batch {
72            records: Vec::with_capacity(BATCH),
73            inputs: Vec::new(),
74            ends: Vec::with_capacity(BATCH),
75        }
76    }
77
78    fn clear(&mut self) {
79        self.records.clear();
80        self.inputs.clear();
81        self.ends.clear();
82    }
83
84    fn len(&self) -> usize {
85        self.records.len()
86    }
87
88    /// Record `i` and the slice of inputs that go with it.
89    #[inline]
90    fn get(&self, i: usize) -> (&sexp::Record, &[sexp::Input]) {
91        let start = if i == 0 { 0 } else { self.ends[i - 1] };
92        (&self.records[i], &self.inputs[start..self.ends[i]])
93    }
94}

96/// Either the records, one at a time off the reader, or the same records off a
97/// thread that is already several batches ahead.
98///
99/// An enum rather than a trait because the choice is made once, before the
100/// first record, and what it decides is where a `&[Input]` comes from — a
101/// borrow that outlives neither arm and so is easier to hand back from one
102/// method than to hide behind a `dyn`.
103pub enum Records {
104    /// Parsed here, between one colour and the next.  What runs without
105    /// `--threads`, and exactly what the driver has always done.
106    Here {
107        reader: sexp::Reader<Box<dyn io::Read + Send>>,
108        inputs: Vec<sexp::Input>,
109        record: Option<sexp::Record>,
110    },
111    /// Parsed on a thread that keeps [`QUEUE`] batches ahead of the fold.
112    Ahead {
113        batches: Receiver<io::Result<Batch>>,
114        /// Batches the fold thread has finished with, to be refilled.
115        spare: SyncSender<Batch>,
116        batch: Batch,
117        at: usize,
118        /// Set when the reader has said there are no more, so a run that stops
119        /// on the record limit does not go looking for another batch.
120        done: bool,
121    },
122}
123
124impl Records {
125    pub fn here(input: Box<dyn io::Read + Send>) -> Self {
126        Records::Here {
127            reader: sexp::Reader::new(input),
128            inputs: Vec::new(),
129            record: None,
130        }
131    }
132
133    /// The same records, read on a thread of its own.
134    pub fn ahead(input: Box<dyn io::Read + Send>) -> Self {
135        let (send_batch, batches) = sync_channel::<io::Result<Batch>>(QUEUE);
136        let (spare, used) = sync_channel::<Batch>(QUEUE + 1);
137
138        thread::spawn(move || {
139            let mut reader = sexp::Reader::new(input);
140            let mut inputs: Vec<sexp::Input> = Vec::new();
141            loop {
142                // A batch the fold thread has finished with, or a fresh one:
143                // `try_recv`, because waiting on recycling would be waiting on
144                // the consumer, and the queue is what paces this already.
145                let mut batch = used.try_recv().unwrap_or_else(|_| Batch::new());
146                batch.clear();
147
148                let mut ended = false;
149                while batch.len() < BATCH {
150                    match reader.next_record(&mut inputs) {
151                        Err(e) => {
152                            // The error goes down the channel rather than up a
153                            // return, so that the fold thread reports it in the
154                            // order it would have hit it -- and *behind* the
155                            // records already in hand, which is why the partial
156                            // batch is sent first.  Dropping it here would lose
157                            // every good record between the last batch boundary
158                            // and the bad one, and a run that fails on record
159                            // 1,000,000 would print nine hundred and ninety-nine
160                            // thousand fewer lines than the serial reader does.
161                            if !batch.records.is_empty() {
162                                let _ = send_batch.send(Ok(batch));
163                            }
164                            let _ = send_batch.send(Err(e));
165                            return;
166                        }
167                        Ok(None) => {
168                            ended = true;
169                            break;
170                        }
171                        Ok(Some(record)) => {
172                            batch.records.push(record);
173                            batch.inputs.extend_from_slice(&inputs);
174                            batch.ends.push(batch.inputs.len());
175                        }
176                    }
177                }
178
179                let last = ended;
180                if !batch.records.is_empty() && send_batch.send(Ok(batch)).is_err() {
181                    // The fold thread stopped early -- a record limit, or a
182                    // closed pipe.  Nothing left to read for.
183                    return;
184                }
185                if last {
186                    return;
187                }
188            }
189        });
190
191        Records::Ahead {
192            batches,
193            spare,
194            batch: Batch::new(),
195            at: 0,
196            done: false,
197        }
198    }
199
200    /// The next record and its inputs, or `None` at the end of the stream.
201    ///
202    /// The borrow is of whichever buffer the record was parsed into, so it ends
203    /// at the next call — which is what the fold wants anyway, since it turns
204    /// the inputs into a colour before asking for another record.
205    #[inline]
206    pub fn next(&mut self) -> io::Result<Option<(&sexp::Record, &[sexp::Input])>> {
207        match self {
208            Records::Here {
209                reader,
210                inputs,
211                record,
212            } => {
213                *record = reader.next_record(inputs)?;
214                Ok(record.as_ref().map(|r| (r, inputs.as_slice())))
215            }
216            Records::Ahead {
217                batches,
218                spare,
219                batch,
220                at,
221                done,
222            } => {
223                if *at == batch.len() {
224                    if *done {
225                        return Ok(None);
226                    }
227                    match batches.recv() {
228                        // Every sender is gone, so the reader finished and said
229                        // so with the last batch it sent.
230                        Err(_) => {
231                            *done = true;
232                            return Ok(None);
233                        }
234                        Ok(Err(e)) => {
235                            *done = true;
236                            return Err(e);
237                        }
238                        Ok(Ok(next)) => {
239                            // The one we have just finished with goes back to be
240                            // refilled; best-effort, since a full queue only
241                            // means the reader has spares already.
242                            let spent = std::mem::replace(batch, next);
243                            let _ = spare.try_send(spent);
244                            *at = 0;
245                        }
246                    }
247                }
248                let i = *at;
249                *at += 1;
250                Ok(Some(batch.get(i)))
251            }
252        }
253    }
254}

260    /// The records the driver's own tests use, in the shape [`sexp`] reads.
261    fn records(n: usize) -> String {
262        (0..n)
263            .map(|tx| {
264                let inputs: String = (0..tx % 4)
265                    .map(|k| format!("(7 {} {} 0)", 100 + k, tx.saturating_sub(k + 1)))
266                    .collect();
267                format!(
268                    "((1 {} {} 0 0 0 0) ({}) ((7 1 0)(7 1 0)))\n",
269                    tx / 7,
270                    tx,
271                    inputs
272                )
273            })
274            .collect()
275    }
276
277    /// Everything a record carries, so the two readers can be compared as
278    /// values rather than by eye.
279    fn drain(mut source: Records) -> Vec<(usize, usize, usize, Vec<sexp::Input>)> {
280        let mut all = Vec::new();
281        while let Some((record, inputs)) = source.next().expect("the records are well formed") {
282            all.push((
283                record.block_id,
284                record.tx_id,
285                record.outputs,
286                inputs.to_vec(),
287            ));
288        }
289        all
290    }
291
292    /// The property: a thread changes when the records are read and nothing at
293    /// all about what they say.  Lengths either side of a batch boundary,
294    /// because that is the seam the arena has to get right.
295    #[test]
296    fn reading_ahead_yields_exactly_the_same_records() {
297        for n in [0usize, 1, 2, 1023, 1024, 1025, 2048, 2049, 5000] {
298            let text = records(n);
299            let here = drain(Records::here(Box::new(io::Cursor::new(text.clone()))));
300            let ahead = drain(Records::ahead(Box::new(io::Cursor::new(text.clone()))));
301            assert_eq!(here.len(), n, "{} records read one at a time", n);
302            assert_eq!(ahead, here, "{} records disagree across the two readers", n);
303        }
304    }
305
306    /// A malformed record has to come back as an error from the call that
307    /// reaches it, not be swallowed by the thread that found it first.
308    #[test]
309    fn a_bad_record_is_reported_by_both_readers() {
310        let text = format!("{}((1 0 nonsense", records(3));
311        for mut source in [
312            Records::here(Box::new(io::Cursor::new(text.clone()))),
313            Records::ahead(Box::new(io::Cursor::new(text.clone()))),
314        ] {
315            let mut good = 0;
316            let outcome = loop {
317                match source.next() {
318                    Ok(Some(_)) => good += 1,
319                    Ok(None) => break Ok(()),
320                    Err(e) => break Err(e),
321                }
322            };
323            assert_eq!(good, 3, "the records before the bad one still arrive");
324            assert!(outcome.is_err(), "the bad record has to be reported");
325        }
326    }
327
328    /// Stopping early -- a record limit -- must not hang: the reader thread is
329    /// blocked on a full queue and has to notice the receiver going away.
330    #[test]
331    fn stopping_early_lets_the_reader_thread_go() {
332        let text = records(200_000);
333        let mut source = Records::ahead(Box::new(io::Cursor::new(text)));
334        for _ in 0..10 {
335            assert!(source.next().expect("well formed").is_some());
336        }
337        drop(source);
338    }
