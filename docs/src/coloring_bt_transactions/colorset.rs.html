1//! # Colors as sorted sets of block ids
2//!
3//! The fast backend behind [`crate::store::ColorStore`].  See that module for
4//! why there are two and what keeps them honest.
5//!
6//! A color here is an `Rc<[u32]>`: block ids, ascending, no duplicates.  Three
7//! things follow from that shape, and they matter in roughly the reverse of the
8//! order one expects.
9//!
10//! ## The representation is most of the speed
11//!
12//! Four bytes a block against the ring's 24-byte `Node`, laid out contiguously
13//! instead of chased through `next`.  A union is then two sequential reads and
14//! one sequential write, which the prefetcher handles and the vector unit can
15//! work on; the ring's merge is a dependent load per term and can be neither.
16//! Before reaching for intrinsics it is worth being clear that this is where the
17//! bulk of the difference comes from.
18//!
19//! ## Sharing removes most of the unions
20//!
21//! `Rc` is what [`crate::store::ColorStore::share`] wants: a second handle on a
22//! color is a refcount bump rather than a copy.  [`crate::poly`]'s objection to
23//! reference counting — that it cannot reclaim a cycle — is about rings, and a
24//! set is acyclic, so it does not apply here.
25//!
26//! ## The merge itself is the smallest of the three
27//!
28//! And before the merge runs at all, [`SetStore::combine`] tries to avoid it:
29//! two handles on one allocation, an empty side, or two ranges that do not
30//! overlap are each answered without comparing a single pair of elements.  The
31//! last of those is common in this workload — a transaction's ancestry often
32//! sits entirely above or below another's — and turns the union into a
33//! `memcpy`.
34//!
35//! ## Limit
36//!
37//! Block ids must fit in a `u32`, which is the point of the representation.
38//! Bitcoin is around 900,000 blocks, so the headroom is four thousandfold, but
39//! the backend asserts rather than silently truncating.
40
41use crate::store::ColorStore;
42use std::collections::HashSet;
43use std::rc::Rc;

45/// One color: block ids ascending, no duplicates.
46pub type Set = Rc<[u32]>;
47
48pub struct SetStore {
49    /// Union builds here and copies into the `Rc` once the length is known, so
50    /// the merge itself never reallocates.
51    scratch: Vec<u32>,
52    /// Elements in allocations that are still referenced, and the high-water
53    /// mark of that.  Not node counts — the two backends measure different
54    /// things and `--stats` says which.
55    live: usize,
56    peak: usize,
57}
58
59impl SetStore {
60    /// Wrap the scratch buffer's contents as a color, and count it.
61    fn intern(&mut self, len_hint: usize) -> Set {
62        debug_assert_eq!(self.scratch.len(), len_hint);
63        let set: Set = Rc::from(&self.scratch[..]);
64        self.live += set.len();
65        self.peak = self.peak.max(self.live);
66        set
67    }
68
69    fn track(&mut self, set: Set) -> Set {
70        self.live += set.len();
71        self.peak = self.peak.max(self.live);
72        set
73    }
74}
75
76/// Merge two ascending duplicate-free slices into `out`, which is cleared first.
77///
78/// Branchless in the sense that matters: the comparison decides how far each
79/// cursor moves rather than which branch runs, so equal elements collapse to one
80/// output without a separate dedup pass.  Since neither input repeats, the
81/// output cannot either.
82///
83/// This is the control the vector kernel has to beat.
84fn merge_into(out: &mut Vec<u32>, a: &[u32], b: &[u32]) {
85    out.clear();
86    out.reserve(a.len() + b.len());
87
88    let (mut i, mut j) = (0, 0);
89    while i < a.len() && j < b.len() {
90        let (x, y) = (a[i], b[j]);
91        out.push(x.min(y));
92        i += (x <= y) as usize;
93        j += (y <= x) as usize;
94    }
95    out.extend_from_slice(&a[i..]);
96    out.extend_from_slice(&b[j..]);
97}

99impl ColorStore for SetStore {
100    type Color = Set;
101
102    /// This backend stores no coefficients at all -- a block is in the set or it
103    /// is not -- so there is nothing for a weight to multiply.  See
104    /// [`crate::weighted`] for the one that does carry them.
105    const WEIGHTED: bool = false;
106
107    fn new() -> Self {
108        SetStore {
109            scratch: Vec::new(),
110            live: 0,
111            peak: 0,
112        }
113    }
114
115    fn singleton(&mut self, block: usize) -> Set {
116        let block = u32::try_from(block)
117            .unwrap_or_else(|_| panic!("block id {} does not fit in a u32 — see colorset", block));
118        let set: Set = Rc::from(&[block][..]);
119        self.track(set)
120    }
121
122    fn combine(&mut self, a: &Set, _wa: f64, b: &Set, _wb: f64) -> Set {
123        // Two handles on one allocation.  Common: several inputs of one
124        // transaction often trace back to a single ancestor.
125        if Rc::ptr_eq(a, b) {
126            return self.share(a);
127        }
128        if a.is_empty() {
129            return self.share(b);
130        }
131        if b.is_empty() {
132            return self.share(a);
133        }
134
135        // Disjoint ranges: the union is a concatenation, so no element is ever
136        // compared and the whole thing is two `memcpy`s.
137        let (lo, hi) = if a[a.len() - 1] < b[0] {
138            (a, b)
139        } else if b[b.len() - 1] < a[0] {
140            (b, a)
141        } else {
142            self.scratch.clear();
143            merge_into(&mut self.scratch, a, b);
144            let len = self.scratch.len();
145            return self.intern(len);
146        };
147
148        self.scratch.clear();
149        self.scratch.reserve(lo.len() + hi.len());
150        self.scratch.extend_from_slice(lo);
151        self.scratch.extend_from_slice(hi);
152        let len = self.scratch.len();
153        self.intern(len)
154    }
155
156    fn scale(&mut self, color: &Set, _w: f64) -> Set {
157        self.share(color)
158    }
159
160    fn share(&mut self, color: &Set) -> Set {
161        Rc::clone(color)
162    }
163
164    fn release(&mut self, color: Set) {
165        // The allocation only goes away when the last handle does.
166        if Rc::strong_count(&color) == 1 {
167            self.live -= color.len();
168        }
169    }
170
171    fn for_each_term(&self, color: &Set, mut f: impl FnMut(usize, f64)) {
172        // Stored ascending because that is what a merge wants; the output format
173        // is decreasing exponents, so it is read back the other way.  Every
174        // coefficient is 1 by construction.
175        for &block in color.iter().rev() {
176            f(block as usize, 1.0);
177        }
178    }
179
180    fn usage(&self) -> (usize, usize) {
181        (self.live, self.peak)
182    }
183
184    fn usage_labels(&self) -> (&'static str, &'static str) {
185        ("live set elems", "peak set elems")
186    }
187
188    fn audit(&self, live: &mut dyn Iterator<Item = &Set>) -> String {
189        // Colors share allocations, so summing lengths over the live handles
190        // would count a shared set once per holder.  Count each allocation once,
191        // by address.
192        let mut seen: HashSet<*const u32> = HashSet::new();
193        let mut reachable = 0;
194        for set in live {
195            if seen.insert(set.as_ptr()) {
196                reachable += set.len();
197            }
198        }
199        if reachable == self.live {
200            format!("audit: {} live set elems, all reachable", self.live)
201        } else {
202            format!(
203                "audit: LEAK -- {} live set elems but only {} reachable ({} lost)",
204                self.live,
205                reachable,
206                self.live as isize - reachable as isize
207            )
208        }
209    }
210}

217    /// `BTreeSet` is the oracle: it defines what a union is, slowly and
218    /// obviously, so [`merge_into`] only has to agree with it.
219    fn oracle(a: &[u32], b: &[u32]) -> Vec<u32> {
220        a.iter()
221            .chain(b.iter())
222            .copied()
223            .collect::<BTreeSet<u32>>()
224            .into_iter()
225            .collect()
226    }
227
228    fn check(a: &[u32], b: &[u32]) {
229        let mut got = Vec::new();
230        merge_into(&mut got, a, b);
231        assert_eq!(got, oracle(a, b), "a={:?} b={:?}", a, b);
232        // Union is commutative, and the two argument orders take different
233        // routes through the cursor arithmetic.
234        let mut swapped = Vec::new();
235        merge_into(&mut swapped, b, a);
236        assert_eq!(swapped, got, "not commutative: a={:?} b={:?}", a, b);
237    }

239    #[test]
240    fn merges_the_shapes_the_fast_paths_are_about() {
241        let dense: Vec<u32> = (0..40).collect();
242        check(&[], &[]);
243        check(&[], &dense);
244        check(&dense, &[]);
245        // identical
246        check(&dense, &dense);
247        // disjoint, either way round
248        check(&(0..20).collect::<Vec<_>>(), &(20..40).collect::<Vec<_>>());
249        check(&(20..40).collect::<Vec<_>>(), &(0..20).collect::<Vec<_>>());
250        // touching at exactly one element
251        check(&(0..21).collect::<Vec<_>>(), &(20..40).collect::<Vec<_>>());
252        // one a subset of the other
253        check(&dense, &[5, 6, 7]);
254        // interleaved, no element shared
255        check(
256            &(0..40).filter(|v| v % 2 == 0).collect::<Vec<_>>(),
257            &(0..40).filter(|v| v % 2 == 1).collect::<Vec<_>>(),
258        );
259        // wildly lopsided
260        check(&dense, &[u32::MAX]);
261        check(&[0], &(1..500).collect::<Vec<_>>());
262    }

278    struct Rng(u64);
279
280    impl Rng {
281        fn next(&mut self) -> u64 {
282            self.0 ^= self.0 << 13;
283            self.0 ^= self.0 >> 7;
284            self.0 ^= self.0 << 17;
285            self.0
286        }
287
288        fn set(&mut self, n: u64, universe: u64) -> Vec<u32> {
289            let mut s: Vec<u32> = (0..n).map(|_| (self.next() % universe) as u32).collect();
290            s.sort_unstable();
291            s.dedup();
292            s
293        }
294    }

296    #[test]
297    fn merges_random_sets() {
298        let mut rng = Rng(0x9E37_79B9_7F4A_7C15);
299        for _ in 0..3000 {
300            // A small universe makes collisions -- and so the dedup path --
301            // common; a large one makes them rare.  Both matter.
302            let universe = if rng.next().is_multiple_of(2) { 32 } else { 100_000 };
303            let (na, nb) = (rng.next() % 60, rng.next() % 60);
304            let a = rng.set(na, universe);
305            let b = rng.set(nb, universe);
306            check(&a, &b);
307        }

310    /// The store's own bookkeeping: `live` must come back to zero once every
311    /// color is released, or `--stats` would report a leak that is not there
312    /// (or, worse, miss one that is).
313    #[test]
314    fn releasing_everything_brings_live_back_to_zero() {
315        let mut store = SetStore::new();
316        let a = store.singleton(3);
317        let b = store.singleton(9);
318        let c = store.combine(&a, 1.0, &b, 1.0);
319        let shared = store.share(&c);
320
321        assert_eq!(store.usage().0, 1 + 1 + 2);
322        // A shared handle owns no new storage, so releasing it frees nothing.
323        store.release(shared);
324        assert_eq!(store.usage().0, 4);
325
326        store.release(a);
327        store.release(b);
328        store.release(c);
329        assert_eq!(store.usage().0, 0);
330    }
