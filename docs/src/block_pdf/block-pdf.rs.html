1//! # The blocks of a neighbourhood, written as a page
2//!
3//! The sibling of `tree-pdf` whose unit is the **block** K(I, O) of the
4//! payments graph — one transaction, its inputs and its outputs, every arc
5//! between them — rather than the node:
6//!
7//! ```text
8//! block-pdf <graph-basename> <transpose-basename> --root <id>[,<id>...] -o <file>
9//!           [--depth <n>] [--max-blocks <n>] [--max-nodes <n>]
10//!           [--fanout <n>] [--fanout-in <n>] [--fanout-out <n>]
11//!           [--seed producing|consuming]
12//!           [--horizontal] [--fill] [--width <pt>] [--max-height <pt>]
13//!           [--mark <id>[,<id>...]] [--labels [compact]] [--label-base <id>]
14//!           [--dash-cross] [--no-check]
15//! ```
16//!
17//! A spanning tree keeps one arc per node, so of a block with two inputs it
18//! draws one input feeding everything and the other a bare leaf, its arcs
19//! dropped and counted; three arcs in four of this graph go that way, and the
20//! picture structurally hides that the graph is made of complete bipartite
21//! pieces.  This binary walks the graph block by block — [`blocks`] says how —
22//! and draws each one it admits as a gadget: its inputs on one row, its
23//! outputs on the row below, all |I|·|O| arcs between them.  A node is drawn
24//! once, so consecutive blocks chain through the nodes they share; nothing is
25//! ever dropped, and the report says `dropped=0` so that a caption can quote
26//! it.
27//!
28//! The transpose is mandatory: a block cannot be found without knowing what
29//! points at a node, and that is what the transpose's successors are.
30//!
31//! # The ink
32//!
33//! The same ink as `tree-pdf`, from the same file, so that a reader carries
34//! one legend across the figures: a filled disc has drawn out-arcs, a hollow
35//! grey ring is a sink, a hollow ring in the warning colour is a *frontier*
36//! — its out-arcs exist and its consuming block is not on the page, whether
37//! the depth stopped there or a budget did — and a marked node wears the
38//! caller's blue over anything else.  Three things are this binary's own.  An
39//! *outside input* — an input no drawn block produced — sits on the row above
40//! the outputs it feeds like every other input, beside the block that fed the
41//! others, and carries a short dashed stub from the side it came from when
42//! something did produce it, and no stub when it is a true source.  A run of
43//! inputs or outputs the fanout hid is three dots in the warning colour at a
44//! slot of their own, with the hidden count at their side under `--labels`.
45//! And the one arc between two drawn nodes that no drawn block owns — from a
46//! frontier output to an outside input, an arc of a block the scissors cut —
47//! is dotted in the warning colour and counted as *stray*.
48//!
49//! An arc entering a block from a drawn block other than the one above it is
50//! a *cross* arc.  It is inked exactly like every other arc — two arcs of one
51//! block must never look different because of a layout accident — and
52//! counted, and `--dash-cross` dashes it for a reader who wants to see the
53//! quotient's non-tree edges.  It is also the one kind of arc that can skip a
54//! row, so the layout counts the arcs that cross a row over a node they do
55//! not end at and reports them as `over_nodes=`; a caption can quote the 0.
56//!
57//! # The labels
58//!
59//! A label is written *beside* its node, on the row, in the direction the
60//! row runs: arcs meet a row only at its nodes, so the paper between two
61//! nodes of a row is the one place near a node that no arc of the gadget can
62//! cross, whereas above or below the node is exactly where its arcs go.  A
63//! row label under `--labels compact` sits beside the row's last slot for the
64//! same reason, and the hidden count of an ellipsis on the outward side of
65//! the row, where the dots have no arcs of their own.  The page keeps a
66//! margin on that side wide enough for the longest label.
67//!
68//! # The page
69//!
70//! Sized, scaled and written as `tree-pdf` does it: one scale that hugs, or
71//! `--fill` for two scales to the page asked for; the same PDF writer, the
72//! same nothing to install.

95const USAGE: &str = "usage: block-pdf <graph-basename> <transpose-basename> --root <id>[,<id>...] -o <file>\n\
96       [--depth <n>] [--max-blocks <n>] [--max-nodes <n>]\n\
97       [--fanout <n>] [--fanout-in <n>] [--fanout-out <n>]\n\
98       [--seed producing|consuming]\n\
99       [--horizontal] [--fill] [--width <pt>] [--max-height <pt>]\n\
100       [--mark <id>[,<id>...]] [--labels [compact]] [--label-base <id>]\n\
101       [--dash-cross] [--no-check]";

103/// How much of a node unit an outside input's stub reaches back, along the
104/// depth axis: past the box's edge, short of the row above.
105const STUB: f64 = 0.6;
106
107/// The largest a label is set, in points, and the width of a Helvetica digit
108/// in ems — all a graph id is made of: enough metrics to place a label, and
109/// to keep a margin for it, without carrying a font.
110const LABEL_SIZE: f64 = 7.0;
111const DIGIT: f64 = 0.556;
112
113/// What `--labels` writes: nothing, every node's id, or one label per row.
114#[derive(Clone, Copy, PartialEq)]
115enum Labels {
116    None,
117    Each,
118    /// The first id of each row and, for a contiguous output row, the last:
119    /// one label a row instead of one a node, for the hubs where per-node
120    /// labels would overprint.
121    Compact,
122}
123
124/// How the page is settled: everything about the drawing that is not a block.
125struct Layout {
126    width: f64,
127    max_height: f64,
128    fill: bool,
129    vertical: bool,
130    labels: Labels,
131    /// Printed ids are `id - base`, for nine-digit ids at seven points.
132    label_base: Option<i64>,
133    dash_cross: bool,
134}
135
136/// Where the baseline of a label of `width` points starts when it is set
137/// beside the mark at `(x, y)` that reaches `gap` points from its centre: to
138/// the right of it when depth runs down the page, under it when depth runs
139/// across.
140fn beside(vertical: bool, x: f64, y: f64, gap: f64, width: f64, size: f64) -> (f64, f64) {
141    if vertical {
142        (x + gap + 1.0, y - 0.36 * size)
143    } else {
144        (x - width / 2.0, y - gap - 1.0 - 0.72 * size)
145    }
146}
147
148/// Where a label starts when it is set on the outward side of a row's mark:
149/// the past's side for an input row, the future's for an output row.
150fn outward(vertical: bool, row: Row, x: f64, y: f64, gap: f64, width: f64, size: f64) -> (f64, f64) {
151    match (vertical, row) {
152        (true, Row::Input) => (x - width / 2.0, y + gap + 1.0),
153        (true, Row::Output) => (x - width / 2.0, y - gap - 1.0 - 0.72 * size),
154        (false, Row::Input) => (x - gap - 1.0 - width, y - 0.36 * size),
155        (false, Row::Output) => (x + gap + 1.0, y - 0.36 * size),
156    }
157}

159/// Draws a laid-out block scene as one page, `marks` being the ids the
160/// caller asked to have inked apart.
161///
162/// The scale is settled as `tree-pdf` settles it — one scale that hugs, or
163/// two that fill — and the node radius keys on the tighter of the two
164/// spacings exactly as there: rows sit a unit apart along depth and slots two
165/// units apart along breadth.
166fn draw(scene: &BlockScene, marks: &HashSet<usize>, opts: &Layout) -> Page {
167    let bounds = scene.bounds;
168    let id = |g: usize| match opts.label_base {
169        None => g.to_string(),
170        Some(base) => (g as i64 - base).to_string(),
171    };
172
173    // What a row says under compact labels: the block's range and side, or
174    // for an input row what is on it out of what the block has.
175    let row_text = |bi: usize, row: Row| -> Option<String> {
176        let b = &scene.blocks[bi];
177        match row {
178            Row::Output => {
179                let n = b.outputs.len();
180                Some(if n == 1 {
181                    id(b.first_out)
182                } else {
183                    format!("{}-{} ({})", id(b.outputs[0]), id(b.outputs[n - 1]), n)
184                })
185            }
186            Row::Input => {
187                let mut shown: Vec<usize> = scene
188                    .nodes
189                    .iter()
190                    .filter(|p| matches!(p.role, Role::Outside { block, .. } if block as usize == bi))
191                    .map(|p| p.graph)
192                    .collect();
193                if shown.is_empty() {
194                    return None;
195                }
196                shown.sort_unstable();
197                let n = b.inputs.len();
198                // The kept inputs not on this row are drawn where their
199                // producers put them; the hidden ones are on this row's
200                // ellipsis, and so count as the row's.
201                let elsewhere = b.kept_in.count(n) - shown.len();
202                let here = n - elsewhere;
203                let span = if shown.len() > 1 {
204                    format!("{} ... {}", id(shown[0]), id(shown[shown.len() - 1]))
205                } else {
206                    id(shown[0])
207                };
208                Some(if n == 1 {
209                    span
210                } else if here == n {
211                    format!("{span} ({n})")
212                } else {
213                    format!("{span} ({here} of {n})")
214                })
215            }
216        }
217    };
218
219    // The longest label hangs out of the drawing on the breadth-positive
220    // side; the page keeps room for it there, and a little along depth for
221    // the hidden counts on the outward side of a cut row.
222    let longest = match opts.labels {
223        Labels::None => 0,
224        Labels::Each => scene.nodes.iter().map(|p| id(p.graph).len()).max().unwrap_or(0),
225        Labels::Compact => (0..scene.blocks.len())
226            .flat_map(|bi| [Row::Input, Row::Output].into_iter().filter_map(move |row| row_text(bi, row)))
227            .map(|t| t.len())
228            .max()
229            .unwrap_or(0),
230    };
231    let overhang =
232        if longest > 0 { DIGIT * LABEL_SIZE * longest as f64 + MAX_RADIUS + 1.0 } else { 0.0 };
233    let along_depth = if opts.labels != Labels::None && !scene.ellipses.is_empty() {
234        0.72 * LABEL_SIZE + 3.0
235    } else {
236        0.0
237    };
238    let (ml, mr, mt, mb) = if opts.vertical {
239        (MARGIN, MARGIN + overhang, MARGIN + along_depth, MARGIN + along_depth)
240    } else {
241        (MARGIN + along_depth, MARGIN + along_depth, MARGIN, MARGIN + overhang)
242    };
243
244    let (scale_x, scale_y, page_w, page_h) = if opts.fill {
245        (
246            (opts.width - ml - mr) / bounds.width(),
247            (opts.max_height - mt - mb) / bounds.height(),
248            opts.width,
249            opts.max_height,
250        )
251    } else {
252        let scale =
253            ((opts.width - ml - mr) / bounds.width()).min((opts.max_height - mt - mb) / bounds.height());
254        (scale, scale, bounds.width() * scale + ml + mr, bounds.height() * scale + mt + mb)
255    };
256
257    // PDF's y runs up the page and the layout's runs down the drawing, so the
258    // flip lives here, in the one mapping every mark goes through.
259    let at = |x: f64, y: f64| {
260        (ml + (x - bounds.min_x) * scale_x, page_h - mt - (y - bounds.min_y) * scale_y)
261    };
262    let node_at = |p: &Placed| at(p.x, p.y);
263
264    let (depth_scale, breadth_scale) = if opts.vertical {
265        (scale_y, scale_x)
266    } else {
267        (scale_x, scale_y)
268    };
269    let spacing = depth_scale.min(2.0 * breadth_scale);
270    let radius = (0.5 * spacing * INK).clamp(MIN_RADIUS, MAX_RADIUS);
271    let hollow = radius >= MIN_HOLLOW;
272    // An ellipsis: three dots along the row, far enough apart to stay three.
273    let dot = radius / 3.0;
274    let step = (0.5 * breadth_scale).max(2.0 * dot);
275
276    // The depth-negative direction on the page: up when vertical, left when
277    // not.  Stubs point that way.
278    let (back_x, back_y) = if opts.vertical { (0.0, 1.0) } else { (-1.0, 0.0) };
279
280    let mut page = Page::new(page_w, page_h);
281
282    // The arcs first, behind everything: tree and cross arcs in one grey,
283    // solid --- a cross arc dashed only when asked --- then the stubs, dashed,
284    // then the stray arcs dotted in the cut's colour.
285    page.set_stroke(LINK);
286    page.set_line_width((radius / 3.0).clamp(0.16, 1.2));
287    let batch = |page: &mut Page, kind: ArcKind| -> usize {
288        let mut in_path = 0;
289        let mut total = 0;
290        for a in scene.arcs.iter().filter(|a| a.kind == kind) {
291            let (x0, y0) = node_at(&scene.nodes[a.from as usize]);
292            let (x1, y1) = node_at(&scene.nodes[a.to as usize]);
293            page.segment(x0, y0, x1, y1);
294            in_path += 1;
295            total += 1;
296            if in_path == BATCH {
297                page.stroke();
298                in_path = 0;
299            }
300        }
301        if in_path > 0 {
302            page.stroke();
303        }
304        total
305    };
306    batch(&mut page, ArcKind::Tree);
307    if scene.arcs.iter().any(|a| a.kind == ArcKind::Cross) {
308        if opts.dash_cross {
309            page.set_dash(2.0 * radius, 1.5 * radius);
310        }
311        batch(&mut page, ArcKind::Cross);
312        if opts.dash_cross {
313            page.solid();
314        }
315    }
316
317    let stubbed: Vec<&Placed> = scene
318        .nodes
319        .iter()
320        .filter(|p| matches!(p.role, Role::Outside { producer: Producer::Undrawn, .. }))
321        .collect();
322    if !stubbed.is_empty() {
323        page.set_dash(2.0 * radius, 1.5 * radius);
324        let mut in_path = 0;
325        for p in &stubbed {
326            let (x, y) = node_at(p);
327            page.segment(x + back_x * STUB * depth_scale, y + back_y * STUB * depth_scale, x, y);
328            in_path += 1;
329            if in_path == BATCH {
330                page.stroke();
331                in_path = 0;
332            }
333        }
334        if in_path > 0 {
335            page.stroke();
336        }
337        page.solid();
338    }
339
340    if scene.arcs.iter().any(|a| a.kind == ArcKind::Stray) {
341        page.set_stroke(CUT);
342        page.set_dash(0.5 * radius, 1.5 * radius);
343        batch(&mut page, ArcKind::Stray);
344        page.solid();
345    }
346
347    if !scene.ellipses.is_empty() {
348        page.set_fill(CUT);
349        let (sx, sy) = if opts.vertical { (1.0, 0.0) } else { (0.0, 1.0) };
350        let mut in_path = 0;
351        for e in &scene.ellipses {
352            let (x, y) = at(e.x, e.y);
353            for k in [-1.0, 0.0, 1.0] {
354                page.circle(x + sx * k * step, y + sy * k * step, dot);
355            }
356            in_path += 3;
357            if in_path >= BATCH {
358                page.fill();
359                in_path = 0;
360            }
361        }
362        if in_path > 0 {
363            page.fill();
364        }
365    }
366
367    // The nodes, in the six classes `tree-pdf` sorts them into: filled says
368    // the node has drawn out-arcs, hollow says it has none; the warning colour
369    // says the graph goes on where the drawing does not; the caller's mark
370    // outranks both.
371    let class = |p: &Placed| {
372        let leaf = usize::from(!p.consumed);
373        if marks.contains(&p.graph) {
374            4 + leaf
375        } else if p.outdeg > 0 && !p.consumed {
376            2 + leaf
377        } else {
378            leaf
379        }
380    };
381    for wanted in 0..=5 {
382        if !scene.nodes.iter().any(|p| class(p) == wanted) {
383            continue;
384        }
385        let colour = [INNER, LEAF, CUT, CUT, MARK, MARK][wanted];
386        let rings = hollow && wanted % 2 == 1;
387        if rings {
388            page.set_fill(PAPER);
389            page.set_stroke(colour);
390            page.set_line_width((radius / 3.0).clamp(0.3, 1.6));
391        } else {
392            page.set_fill(colour);
393        }
394        let mut in_path = 0;
395        for p in scene.nodes.iter().filter(|p| class(p) == wanted) {
396            let (x, y) = node_at(p);
397            page.circle(x, y, radius);
398            in_path += 1;
399            if in_path == BATCH {
400                if rings {
401                    page.fill_and_stroke();
402                } else {
403                    page.fill();
404                }
405                in_path = 0;
406            }
407        }
408        if in_path > 0 {
409            if rings {
410                page.fill_and_stroke();
411            } else {
412                page.fill();
413            }
414        }
415    }
416
417    // The labels last, over everything, beside what they name (the module's
418    // third section); the size follows the circles and never drops below
419    // legibility.
420    if opts.labels != Labels::None {
421        let size = (radius * 2.2).clamp(3.5, LABEL_SIZE);
422        let digit = DIGIT * size;
423        page.set_fill(LABEL);
424        let write_beside = |page: &mut Page, x: f64, y: f64, gap: f64, text: &str| {
425            let (lx, ly) = beside(opts.vertical, x, y, gap, digit * text.len() as f64, size);
426            page.text(lx, ly, size, text);
427        };
428        let write_outward = |page: &mut Page, row: Row, x: f64, y: f64, gap: f64, text: &str| {
429            let (lx, ly) = outward(opts.vertical, row, x, y, gap, digit * text.len() as f64, size);
430            page.text(lx, ly, size, text);
431        };
432        for e in &scene.ellipses {
433            let (x, y) = at(e.x, e.y);
434            write_outward(&mut page, e.row, x, y, dot, &format!("+{}", e.hidden));
435        }
436        match opts.labels {
437            Labels::Each => {
438                for p in &scene.nodes {
439                    let (x, y) = node_at(p);
440                    write_beside(&mut page, x, y, radius, &id(p.graph));
441                }
442            }
443            Labels::Compact => {
444                // One label a row, beside its last slot --- the node or the
445                // ellipsis furthest along the row.
446                for (bi, _) in scene.blocks.iter().enumerate() {
447                    for row in [Row::Input, Row::Output] {
448                        let Some(text) = row_text(bi, row) else { continue };
449                        // The row's marks as (along the row, x, y, reach).
450                        let along = |x: f64, y: f64| if opts.vertical { x } else { -y };
451                        let mut end: Option<(f64, f64, f64, f64)> = None;
452                        let mut consider = |x: f64, y: f64, gap: f64| {
453                            if end.is_none_or(|(a, ..)| along(x, y) > a) {
454                                end = Some((along(x, y), x, y, gap));
455                            }
456                        };
457                        for p in scene.nodes.iter().filter(|p| match p.role {
458                            Role::Output { block } => row == Row::Output && block as usize == bi,
459                            Role::Outside { block, .. } => row == Row::Input && block as usize == bi,
460                        }) {
461                            let (x, y) = node_at(p);
462                            consider(x, y, radius);
463                        }
464                        for e in scene.ellipses.iter().filter(|e| e.block as usize == bi && e.row == row) {
465                            let (x, y) = at(e.x, e.y);
466                            consider(x, y, step + dot);
467                        }
468                        if let Some((_, x, y, gap)) = end {
469                            write_beside(&mut page, x, y, gap, &text);
470                        }
471                    }
472                }
473            }
474            Labels::None => {}
475        }
476    }
477
478    page
479}

481fn run() -> Result<(), String> {
482    let argv: Vec<String> = env::args().skip(1).collect();
483
484    let mut basenames: Vec<&str> = Vec::new();
485    let mut out_path: Option<&str> = None;
486    let mut roots: Vec<usize> = Vec::new();
487    let mut prune = BlockPrune { max_nodes: DEFAULT_MAX_NODES, ..BlockPrune::default() };
488    let mut seed = Seed::Producing;
489    let mut width = DEFAULT_WIDTH;
490    let mut max_height = DEFAULT_MAX_HEIGHT;
491    let mut vertical = true;
492    let mut fill = false;
493    let mut labels = Labels::None;
494    let mut label_base: Option<i64> = None;
495    let mut dash_cross = false;
496    let mut check = true;
497    let mut marks: HashSet<usize> = HashSet::new();
498
499    let mut i = 0;
500    while i < argv.len() {
501        match argv[i].as_str() {
502            "-o" | "--out" => {
503                i += 1;
504                out_path = Some(argv.get(i).ok_or_else(|| format!("-o wants a file\n{USAGE}"))?);
505            }
506            "--root" => {
507                i += 1;
508                let v = argv.get(i).ok_or_else(|| format!("--root wants a node id\n{USAGE}"))?;
509                for part in v.split(',') {
510                    roots.push(
511                        part.parse::<usize>().map_err(|_| format!("--root {part}: not a node id"))?,
512                    );
513                }
514            }
515            "--depth" => {
516                i += 1;
517                let v = argv.get(i).ok_or_else(|| format!("--depth wants a number\n{USAGE}"))?;
518                prune.depth =
519                    Some(v.parse::<usize>().map_err(|_| format!("--depth {v}: not a number"))?);
520            }
521            flag @ ("--max-blocks" | "--max-nodes") => {
522                i += 1;
523                let v = argv.get(i).ok_or_else(|| format!("{flag} wants a number\n{USAGE}"))?;
524                let n = v
525                    .parse::<usize>()
526                    .ok()
527                    .filter(|&n| n > 0)
528                    .ok_or_else(|| format!("{flag} {v}: wants a positive number"))?;
529                if flag == "--max-blocks" {
530                    prune.max_blocks = n;
531                } else {
532                    prune.max_nodes = n;
533                }
534            }
535            flag @ ("--fanout" | "--fanout-in" | "--fanout-out") => {
536                i += 1;
537                let v = argv.get(i).ok_or_else(|| format!("{flag} wants a number\n{USAGE}"))?;
538                let n = v
539                    .parse::<usize>()
540                    .ok()
541                    .filter(|&n| n > 0)
542                    .ok_or_else(|| format!("{flag} {v}: wants a positive number"))?;
543                // `--fanout` sets both sides; a side's own flag overrides it
544                // whichever order they come in.
545                match flag {
546                    "--fanout-in" => prune.fanout_in = Some(n),
547                    "--fanout-out" => prune.fanout_out = Some(n),
548                    _ => {
549                        prune.fanout_in.get_or_insert(n);
550                        prune.fanout_out.get_or_insert(n);
551                    }
552                }
553            }
554            "--seed" => {
555                i += 1;
556                let v = argv.get(i).ok_or_else(|| format!("--seed wants producing or consuming\n{USAGE}"))?;
557                seed = match v.as_str() {
558                    "producing" => Seed::Producing,
559                    "consuming" => Seed::Consuming,
560                    _ => return Err(format!("--seed {v}: producing or consuming\n{USAGE}")),
561                };
562            }
563            flag @ ("--width" | "--max-height") => {
564                i += 1;
565                let v = argv.get(i).ok_or_else(|| format!("{flag} wants a number\n{USAGE}"))?;
566                let pt = v
567                    .parse::<f64>()
568                    .ok()
569                    .filter(|&pt| pt > 2.0 * MARGIN)
570                    .ok_or_else(|| format!("{flag} {v}: a page needs a positive size"))?;
571                if flag == "--width" {
572                    width = pt;
573                } else {
574                    max_height = pt;
575                }
576            }
577            // Depth across the page instead of down it: what a chain of
578            // K(1,k) blocks wants.  Down is the default here, the past above
579            // the event, unlike `tree-pdf`, whose subjects are chains.
580            "--horizontal" => vertical = false,
581            "--vertical" => vertical = true,
582            "--fill" => fill = true,
583            "--labels" => {
584                labels = Labels::Each;
585                if argv.get(i + 1).is_some_and(|v| v == "compact") {
586                    labels = Labels::Compact;
587                    i += 1;
588                }
589            }
590            "--label-base" => {
591                i += 1;
592                let v = argv.get(i).ok_or_else(|| format!("--label-base wants a node id\n{USAGE}"))?;
593                label_base =
594                    Some(v.parse::<i64>().map_err(|_| format!("--label-base {v}: not a node id"))?);
595            }
596            "--mark" => {
597                i += 1;
598                let v = argv.get(i).ok_or_else(|| format!("--mark wants a node id\n{USAGE}"))?;
599                for part in v.split(',') {
600                    marks.insert(
601                        part.parse::<usize>().map_err(|_| format!("--mark {part}: not a node id"))?,
602                    );
603                }
604            }
605            "--dash-cross" => dash_cross = true,
606            "--no-check" => check = false,
607            "-h" | "--help" => return Err(USAGE.to_string()),
608            other if other.starts_with('-') => {
609                return Err(format!("unknown flag {other}\n{USAGE}"))
610            }
611            other => basenames.push(other),
612        }
613        i += 1;
614    }
615
616    let [graph_name, transpose_name] = basenames.as_slice() else {
617        return Err(format!(
618            "expected a graph basename and its transpose's, got {}\n{USAGE}",
619            basenames.len()
620        ));
621    };
622    let Some(out_path) = out_path else {
623        return Err(format!("-o is required\n{USAGE}"));
624    };
625    if roots.is_empty() {
626        return Err(format!("--root is required\n{USAGE}"));
627    }
628
629    eprintln!("reading {graph_name}");
630    let pg = BvGraph::with_basename(graph_name)
631        .load()
632        .map_err(|e| format!("{graph_name}: {e:#}"))?;
633    eprintln!("reading {transpose_name}");
634    let pgt = BvGraph::with_basename(transpose_name)
635        .load()
636        .map_err(|e| format!("{transpose_name}: {e:#}"))?;
637
638    let walked = blocks::walk(&pg, &pgt, &roots, &prune, seed, check)?;
639    for line in &walked.seeds {
640        eprintln!("{line}");
641    }
642    eprintln!("{}", prune.summary());
643    let scene = blocks::lay_out_blocks(&pg, &pgt, walked, vertical)?;
644    eprintln!("{}", scene.report.lines());
645
646    let opts = Layout { width, max_height, fill, vertical, labels, label_base, dash_cross };
647    let page = draw(&scene, &marks, &opts);
648    let (w, h) = (page.width(), page.height());
649    page.write(out_path)?;
650
651    let size = std::fs::metadata(out_path)
652        .map(|m| m.len())
653        .map_err(|e| format!("{out_path}: {e}"))?;
654    eprintln!(
655        "{} nodes, {} arcs on a {:.0} by {:.0} pt page, {} bytes written to {out_path}",
656        scene.report.nodes_drawn, scene.report.arcs_drawn, w, h, size
657    );
658    Ok(())
659}
660
661fn main() -> ExitCode {
662    match run() {
663        Ok(()) => ExitCode::SUCCESS,
664        Err(message) => {
665            eprintln!("{message}");
666            ExitCode::FAILURE
667        }
668    }
669}

694    /// The deflated content stream of a written page, as text.
695    fn ops_of(page: Page, name: &str) -> String {
696        use std::io::Read;
697        let path = std::env::temp_dir()
698            .join(format!("block-pdf-ops-{}-{name}.pdf", std::process::id()));
699        let path = path.to_string_lossy().into_owned();
700        page.write(&path).unwrap();
701        let bytes = std::fs::read(&path).unwrap();
702        std::fs::remove_file(&path).ok();
703        let start = bytes.windows(7).position(|w| w == b"stream\n").unwrap() + 7;
704        let end = bytes.windows(10).position(|w| w == b"\nendstream").unwrap();
705        let mut ops = String::new();
706        flate2::read::ZlibDecoder::new(&bytes[start..end])
707            .read_to_string(&mut ops)
708            .unwrap();
709        ops
710    }

714    /// U7, the page half: a chain with nothing dashed carries no dash
715    /// operator; a diamond's cross arc is dashed only when asked; a source

717    #[test]
718    fn dashes_only_where_something_is_dashed() {
719        let scene = scene_of(6, &CHAIN, &[1], &BlockPrune::default());
720        let ops = ops_of(draw(&scene, &HashSet::new(), &layout(Labels::None, false)), "chain");
721        assert!(!ops.contains(" d\n"), "nothing to dash: {ops}");
722        assert_eq!(ops.matches("c h").count(), 6, "six nodes: {ops}");
723        assert_eq!(ops.matches(" l\n").count(), 8, "eight arcs");
724        assert!(!ops.contains("0.82 0.24 0.1"), "nothing was cut");
725
726        let diamond = [(0, 1), (0, 2), (1, 3), (2, 4), (3, 5), (4, 5)];
727        let scene = scene_of(6, &diamond, &[1], &BlockPrune::default());
728        let solid = ops_of(draw(&scene, &HashSet::new(), &layout(Labels::None, false)), "solid");
729        assert!(!solid.contains(" d\n"), "a cross arc is solid by default: {solid}");
730        let dashed = ops_of(draw(&scene, &HashSet::new(), &layout(Labels::None, true)), "dashed");
731        assert_eq!(dashed.matches("] 0 d\n").count(), 2, "one dashed batch, then solid: {dashed}");
732
733        // Rooted one block down, the seed's inputs come from an undrawn
734        // block: stubs, dashed, and the frontier colour nowhere since
735        // nothing leads on from the drawn nodes.
736        let scene = scene_of(6, &CHAIN, &[3], &BlockPrune { depth: Some(0), ..BlockPrune::default() });
737        assert_eq!(scene.report.outside_from_undrawn, 2);
738        let ops = ops_of(draw(&scene, &HashSet::new(), &layout(Labels::None, false)), "stub");
739        assert_eq!(ops.matches("] 0 d\n").count(), 2, "the stubs' dash and its undoing: {ops}");
740        assert!(ops.contains("0.82 0.24 0.1"), "3 and 4 lead on to a block the depth stopped");
741    }

743    /// Labels are opt-in, name every node when asked, and one row at a time

777    /// A label sits beside its node along the row --- right of it when depth
778    /// runs down, under it when depth runs across --- and never on the
779    /// node's column, where its arcs are.
780    #[test]
781    fn labels_sit_beside_their_nodes() {
782        let (x, y, size) = (100.0, 200.0, 7.0);
783        let (lx, ly) = beside(true, x, y, 5.0, 20.0, size);
784        assert_eq!(lx, 106.0, "past the circle");
785        assert!(ly < y && ly > y - size, "the baseline a little under the centre line");
786        let (lx, ly) = beside(false, x, y, 5.0, 20.0, size);
787        assert_eq!(lx, 90.0, "centred");
788        assert!(ly < y - 5.0 - size * 0.72, "under the circle");
789
790        // Outward of an input row is the past's side: up when vertical, left
791        // when horizontal.
792        assert!(outward(true, Row::Input, x, y, 2.0, 20.0, size).1 > y);
793        assert!(outward(true, Row::Output, x, y, 2.0, 20.0, size).1 < y);
794        assert!(outward(false, Row::Input, x, y, 2.0, 20.0, size).0 < x - 20.0);
795        assert!(outward(false, Row::Output, x, y, 2.0, 20.0, size).0 > x);
796
797        // On the page: every node's label starts right of that node's
798        // circle, on its own row.
799        let scene = scene_of(6, &CHAIN, &[1], &BlockPrune::default());
800        let ops = ops_of(draw(&scene, &HashSet::new(), &layout(Labels::Each, false)), "beside");
801        // A circle's path starts at its rightmost point, (cx + r, cy).
802        let circles: Vec<(f64, f64)> = ops
803            .lines()
804            .filter(|l| l.ends_with(" c h"))
805            .map(|l| {
806                let v: Vec<f64> = l.split(' ').take(2).map(|s| s.parse().unwrap()).collect();
807                (v[0], v[1])
808            })
809            .collect();
810        assert_eq!(circles.len(), 6);
811        let labels: Vec<(f64, f64)> = ops
812            .lines()
813            .filter(|l| l.starts_with("BT"))
814            .map(|l| {
815                let v: Vec<&str> = l.split(' ').collect();
816                (v[4].parse().unwrap(), v[5].parse().unwrap())
817            })
818            .collect();
819        assert_eq!(labels.len(), 6);
820        for (lx, ly) in labels {
821            assert!(
822                circles.iter().any(|&(rx, cy)| (cy - ly - 0.36 * 7.0).abs() < 0.05 && (lx - rx - 1.0).abs() < 0.05),
823                "label at ({lx}, {ly}) starts one point right of a circle: {circles:?}"
824            );
825        }
826    }
