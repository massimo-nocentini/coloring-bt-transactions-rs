1//! # Where the drawing is looked at from
2//!
3//! The layout in [`forest`](super) puts every node at a place measured in
4//! *nodes*: level `d` sits at `x = d`, and two nodes at the same depth are at
5//! least a diameter and a margin apart.  A window, meanwhile, is measured in
6//! pixels and is a few hundred of them across, while the drawings this is for
7//! are millions of nodes wide.  A camera is the one thing standing between the
8//! two: it says which rectangle of the drawing the window is showing, and
9//! everything else — what to cull, what to draw, what a click landed on — is
10//! read off it.
11//!
12//! It is deliberately a plain value with no idea that GTK exists.  That is what
13//! lets the whole of the interaction — zooming, panning, framing a subtree — be
14//! tested without opening a window, which is the only way any of it *can* be
15//! tested here.
16//!
17//! # The one invariant
18//!
19//! [`Camera::scale`] is pixels per node, and it is kept inside
20//! [`MIN_SCALE`]`..=`[`MAX_SCALE`].  The bounds are not a policy about how far
21//! one may zoom — [`Camera::frame`] will happily fit a drawing at either end —
22//! they are there so that a scale can never become zero, infinite, or NaN, since
23//! every other method divides by it.
24//!
25//! # Zooming is anchored
26//!
27//! [`Camera::zoom`] takes the point on the screen to hold still.  Given the
28//! pointer, the node under the cursor stays under the cursor and the drawing
29//! grows around it, which is the gesture that lets one dive into a subtree
30//! without also having to chase it across the window.  Passing the centre of the
31//! viewport gives the other reading, zooming into the middle.
32//!
33//! Because the scale is clamped, the anchor is honoured against the scale the
34//! camera *ended up with* rather than the one it was asked for: at the stops,
35//! zooming further does nothing at all instead of sliding the drawing sideways.
36
37// A small library with no `lib.rs` to live in, and the one file of it that more
38// than the viewers want: the main binary's `--view` measures its window with the
39// same camera, and `src/*.rs` is where a file the main binary names has to be.
40// So this sits at the root and everything else reaches it by `#[path]` --
41// `tree-view` and `tx-view` from `src/bin/`, `tests/viewer_geometry.rs` from
42// outside, all under the name `crate::camera` either way.  Compiled into a
43// binary crate, anything one frame does not happen to call reads as dead; the
44// surface is the point, so the lint goes rather than the surface.
45#![allow(dead_code)]
46
47/// An axis-aligned rectangle in node units.
48///
49/// Half-open in neither direction and empty when a maximum is below its minimum,
50/// which is what [`Rect::nothing`] builds and what lets a bounding box be
51/// accumulated by [`Rect::add`] from no points at all.
52#[derive(Clone, Copy, PartialEq, Debug)]
53pub struct Rect {
54    pub min_x: f64,
55    pub min_y: f64,
56    pub max_x: f64,
57    pub max_y: f64,
58}
59
60impl Rect {
61    pub fn new(min_x: f64, min_y: f64, max_x: f64, max_y: f64) -> Self {
62        Rect { min_x, min_y, max_x, max_y }
63    }
64
65    /// The empty rectangle: it contains no point and intersects nothing, and
66    /// growing it by [`Rect::add`] gives the bounding box of what was added.
67    pub fn nothing() -> Self {
68        Rect::new(f64::INFINITY, f64::INFINITY, f64::NEG_INFINITY, f64::NEG_INFINITY)
69    }
70
71    /// The square of side `side` centred on `(x, y)`.
72    pub fn square(x: f64, y: f64, side: f64) -> Self {
73        let half = side / 2.0;
74        Rect::new(x - half, y - half, x + half, y + half)
75    }
76
77    pub fn is_empty(self) -> bool {
78        self.max_x < self.min_x || self.max_y < self.min_y
79    }
80
81    pub fn width(self) -> f64 {
82        self.max_x - self.min_x
83    }
84
85    pub fn height(self) -> f64 {
86        self.max_y - self.min_y
87    }
88
89    pub fn centre(self) -> (f64, f64) {
90        ((self.min_x + self.max_x) / 2.0, (self.min_y + self.max_y) / 2.0)
91    }
92
93    /// Grows the rectangle to hold `(x, y)`.
94    pub fn add(&mut self, x: f64, y: f64) {
95        self.min_x = self.min_x.min(x);
96        self.min_y = self.min_y.min(y);
97        self.max_x = self.max_x.max(x);
98        self.max_y = self.max_y.max(y);
99    }
100
101    pub fn contains(self, x: f64, y: f64) -> bool {
102        x >= self.min_x && x <= self.max_x && y >= self.min_y && y <= self.max_y
103    }
104
105    /// Whether the two rectangles share any point.  Touching edges count, which
106    /// costs a node on a boundary being visited by both sides of a split and
107    /// buys never dropping one that sits exactly on it.
108    pub fn intersects(self, other: Rect) -> bool {
109        self.min_x <= other.max_x
110            && other.min_x <= self.max_x
111            && self.min_y <= other.max_y
112            && other.min_y <= self.max_y
113    }
114
115    /// The same rectangle grown by `by` on every side.
116    pub fn grown(self, by: f64) -> Self {
117        Rect::new(self.min_x - by, self.min_y - by, self.max_x + by, self.max_y + by)
118    }
119
120    /// The smallest square holding this rectangle, centred on it.
121    ///
122    /// What a quadtree wants for a root: quartering a square gives squares, so
123    /// the cells stay square all the way down and one number describes a cell's
124    /// size on both axes.
125    pub fn to_square(self) -> Self {
126        let (cx, cy) = self.centre();
127        Rect::square(cx, cy, self.width().max(self.height()))
128    }
129}
130
131/// Pixels per node the camera will not go below, or above.
132///
133/// The low end is a drawing a thousand million nodes wide shown in a thousand
134/// pixels; the high end is a single node filling a window.  Neither is a limit
135/// anybody meets by scrolling — they are there so the scale stays a number one
136/// can divide by.
137pub const MIN_SCALE: f64 = 1e-6;
138pub const MAX_SCALE: f64 = 1e3;
139
140/// How much one notch of the wheel multiplies or divides the scale by.
141///
142/// Small enough that a zoom reads as a movement rather than a jump, large enough
143/// that crossing four decades of scale — which a graph of transactions asks for —
144/// is a few flicks of the wheel and not a minute of scrolling.
145pub const ZOOM_PER_NOTCH: f64 = 1.25;
146
147/// The fraction of the window [`Camera::frame`] leaves clear around what it is
148/// framing, so that the outermost nodes are not cut in half by the edge.
149const FRAME_MARGIN: f64 = 0.04;
150
151/// The smallest extent [`Camera::frame`] will pretend a rectangle has.
152///
153/// A single node, or a chain, is a rectangle with no width on one axis, and
154/// dividing the window by that gives an infinite scale.  Framing it as if it
155/// were this many nodes across shows it at a size one can see.
156const MIN_FRAMED_EXTENT: f64 = 8.0;
157
158/// Which rectangle of the drawing a window of `width` by `height` pixels shows.
159#[derive(Clone, Copy, Debug)]
160pub struct Camera {
161    /// The node coordinate at the centre of the window.
162    x: f64,
163    y: f64,
164    /// Pixels per node unit, in `MIN_SCALE..=MAX_SCALE`.
165    scale: f64,
166    width: f64,
167    height: f64,
168}
169
170impl Camera {
171    /// A camera on a window of `width` by `height`, framing `subject`.
172    pub fn framing(subject: Rect, width: f64, height: f64) -> Self {
173        let mut camera = Camera { x: 0.0, y: 0.0, scale: 1.0, width, height };
174        camera.resize(width, height);
175        camera.frame(subject);
176        camera
177    }
178
179    pub fn scale(self) -> f64 {
180        self.scale
181    }
182
183    pub fn width(self) -> f64 {
184        self.width
185    }
186
187    pub fn height(self) -> f64 {
188        self.height
189    }
190
191    /// Follows the window changing size, keeping the same point in the middle.
192    ///
193    /// A zero-sized window is what GTK reports before it has laid the drawing
194    /// area out, and a camera whose window has no extent divides by zero in
195    /// [`Camera::visible`]; one pixel is the smallest lie that avoids it.
196    pub fn resize(&mut self, width: f64, height: f64) {
197        self.width = width.max(1.0);
198        self.height = height.max(1.0);
199    }
200
201    /// The rectangle of the drawing now on screen.
202    pub fn visible(self) -> Rect {
203        let half_w = self.width / (2.0 * self.scale);
204        let half_h = self.height / (2.0 * self.scale);
205        Rect::new(self.x - half_w, self.y - half_h, self.x + half_w, self.y + half_h)
206    }
207
208    /// Where a node's coordinate lands in the window.
209    pub fn to_screen(self, x: f64, y: f64) -> (f64, f64) {
210        (
211            (x - self.x) * self.scale + self.width / 2.0,
212            (y - self.y) * self.scale + self.height / 2.0,
213        )
214    }
215
216    /// Which node coordinate a point of the window stands for.
217    pub fn to_node(self, sx: f64, sy: f64) -> (f64, f64) {
218        (
219            (sx - self.width / 2.0) / self.scale + self.x,
220            (sy - self.height / 2.0) / self.scale + self.y,
221        )
222    }
223
224    /// Moves the drawing by a distance measured in pixels.
225    ///
226    /// The arguments are how far the *pointer* went, and the drawing follows it:
227    /// dragging right brings what was off the left edge into view.
228    pub fn pan(&mut self, dx: f64, dy: f64) {
229        self.x -= dx / self.scale;
230        self.y -= dy / self.scale;
231    }
232
233    /// Multiplies the scale by `factor`, holding the point `(sx, sy)` of the
234    /// window over the same node it was over.
235    ///
236    /// The anchor is honoured against the scale actually reached, so at the
237    /// stops this does nothing rather than sliding the drawing sideways.
238    pub fn zoom(&mut self, factor: f64, sx: f64, sy: f64) {
239        let (nx, ny) = self.to_node(sx, sy);
240        let scale = (self.scale * factor).clamp(MIN_SCALE, MAX_SCALE);
241        if scale == self.scale {
242            return;
243        }
244        self.scale = scale;
245        // Put the node that was under (sx, sy) back under it: the offset from the
246        // centre of the window is fixed in pixels, so it is this many nodes now.
247        self.x = nx - (sx - self.width / 2.0) / scale;
248        self.y = ny - (sy - self.height / 2.0) / scale;
249    }
250
251    /// One wheel gesture: `notches` positive scrolls down, and zooms out.
252    pub fn zoom_notches(&mut self, notches: f64, sx: f64, sy: f64) {
253        self.zoom(ZOOM_PER_NOTCH.powf(-notches), sx, sy);
254    }
255
256    /// The node coordinate the middle of the window is on.
257    pub fn eye(self) -> (f64, f64) {
258        (self.x, self.y)
259    }
260
261    /// Puts `(x, y)` in the middle of the window, at the scale already reached.
262    ///
263    /// What [`Camera::frame`] would do if it were not also allowed to zoom, and
264    /// the reason both are here: framing a single node has to invent a size to
265    /// show it at, whereas centring one keeps whatever the eye was already used
266    /// to and only moves it.
267    pub fn look_at(&mut self, x: f64, y: f64) {
268        self.x = x;
269        self.y = y;
270    }
271
272    /// Puts `subject` in the middle of the window, as large as it will go.
273    ///
274    /// An empty rectangle is not something one can look at, and leaves the
275    /// camera alone.
276    pub fn frame(&mut self, subject: Rect) {
277        if subject.is_empty() {
278            return;
279        }
280        let (cx, cy) = subject.centre();
281        self.x = cx;
282        self.y = cy;
283
284        let w = subject.width().max(MIN_FRAMED_EXTENT);
285        let h = subject.height().max(MIN_FRAMED_EXTENT);
286        let clear = 1.0 - 2.0 * FRAME_MARGIN;
287        self.scale = (self.width * clear / w)
288            .min(self.height * clear / h)
289            .clamp(MIN_SCALE, MAX_SCALE);
290    }
291}

297    /// Centring moves the eye and nothing else, which is what tells it apart
298    /// from framing: the zoom one climbed to survives it.
299    #[test]
300    fn looking_at_a_point_keeps_the_zoom() {
301        let mut camera = Camera::framing(Rect::new(0.0, 0.0, 100.0, 100.0), 400.0, 300.0);
302        let was = camera.scale();
303
304        camera.look_at(7.0, 9.0);
305
306        assert_eq!(camera.scale(), was, "the scale moved");
307        assert_eq!(camera.eye(), (7.0, 9.0));
308        assert_eq!(camera.to_screen(7.0, 9.0), (200.0, 150.0), "not in the middle");
309        assert!(camera.visible().contains(7.0, 9.0));
310
311        // And framing the same point would not have: a single point has no
312        // extent, so it is shown at a size of its own invention.
313        camera.frame(Rect::square(7.0, 9.0, 0.0));
314        assert!(camera.scale() > was);
315    }

317    /// A bounding box built from nothing is empty, and holds whatever is added.
318    #[test]
319    fn a_box_grown_from_nothing() {
320        let mut r = Rect::nothing();
321        assert!(r.is_empty());
322        assert!(!r.contains(0.0, 0.0));
323
324        r.add(1.0, 2.0);
325        assert!(!r.is_empty());
326        assert_eq!((r.width(), r.height()), (0.0, 0.0));

329        assert_eq!(r, Rect::new(-1.0, 2.0, 1.0, 6.0));
330        assert_eq!(r.centre(), (0.0, 4.0));
331        assert_eq!(r.to_square(), Rect::new(-2.0, 2.0, 2.0, 6.0));

346    #[test]
347    fn screen_and_node_coordinates_agree() {
348        let camera = Camera::framing(Rect::new(0.0, 0.0, 100.0, 50.0), 800.0, 600.0);
349
350        for &(x, y) in &[(0.0, 0.0), (100.0, 50.0), (-30.0, 12.5)] {
351            let (sx, sy) = camera.to_screen(x, y);
352            let (bx, by) = camera.to_node(sx, sy);
353            assert!((bx - x).abs() < 1e-9 && (by - y).abs() < 1e-9);
354        }

356        // The centre of what is framed is the centre of the window.
357        let (sx, sy) = camera.to_screen(50.0, 25.0);
358        assert!((sx - 400.0).abs() < 1e-9 && (sy - 300.0).abs() < 1e-9);
359    }
